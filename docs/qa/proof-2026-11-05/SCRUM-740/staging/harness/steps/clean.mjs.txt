// Clean-up, own records only: sign every session out, archive the two ZZ TEST tills, then read everything back.
export default async function (S) {
  const f = S.st.fixture;
  const C = S.st.cleanup;
  C.startedAt = C.startedAt ?? new Date().toISOString();
  const act = (what, res, extra = {}) => { C.actions.push({ what, status: res.status, code: res.body?.error?.code ?? null, ...extra }); S.save(); return res; };
  const tills = Object.entries(f.tills);
  for (const [k, t] of tills) if (!t.name.startsWith('ZZ TEST SCRUM-7')) throw new Error('Refusing: ' + t.name);
  // 1. Unpaid sales on this run's tills (none expected).
  C.salesBefore = {};
  for (const [k, t] of tills) {
    const list = (await S.adminApi('GET', `/sales?branchId=${f.branchId}&stationId=${t.id}&limit=50`)).body.sales ?? [];
    C.salesBefore[k] = list.map((s) => ({ id: s.id, receipt: s.receiptNumber ?? null, status: s.status, gross: s.totals?.grossSatang ?? null }));
    for (const s of list.filter((s) => s.status === 'tendering')) act('void unpaid sale', await S.adminApi('POST', `/sales/${s.id}/void`, { reason: 'ZZ TEST SCRUM-740/768 proof: not paid' }), { saleId: s.id });
  }
  // 2. Every session this run opened, signed out.
  for (const label of ['T740', 'T768', 'CON']) {
    if (!S.pages[label]) continue;
    act(`${label} sign-out`, await S.pageApi(label, 'POST', '/auth/sign-out', {}));
    C[`me${label}AfterSignOut`] = (await S.pageApi(label, 'GET', '/me')).status;
  }
  // 3. The two ZZ TEST tills (the receipt printer assignment on the 768 till goes with it).
  for (const [k, t] of tills) act('archive station ' + t.name, await S.adminApi('DELETE', `/stations/${t.id}`));
  // 4. Read back.
  const stations = (await S.adminApi('GET', `/branches/${f.branchId}/stations?includeArchived=true`)).body.stations;
  const mineAfter = tills.map(([k, t]) => { const s = stations.find((x) => x.id === t.id); return { name: s?.name, archived: s?.archived ?? !!s?.archivedAt, devices: (s?.devices ?? []).length }; });
  const rt1 = stations.find((s) => s.name === 'Reception Till 1');
  const regs = (await S.adminApi('GET', `/checkin/registrations?branchId=${f.branchId}`)).body.registrations;
  const mineWaiting = regs.filter((r) => r.id === f.reg1 || r.id === f.reg2).map((r) => ({ guardian: r.guardianName, children: r.children.map((c) => ({ name: c.childName, status: c.status })) }));
  const branches = (await S.adminApi('GET', '/branches')).body.branches.filter((b) => !b.archivedAt);
  const liveFreebies = [];
  for (const b of branches) {
    const pk = await S.adminApi('GET', `/branches/${b.id}/ticket-packages`);
    const list = Array.isArray(pk.body) ? pk.body : (pk.body.packages ?? []);
    for (const p of list.filter((p) => !p.archivedAt)) if (Array.isArray(p.freebies) && p.freebies.length > 0) liveFreebies.push({ branch: b.name, ticket: p.name, freebies: p.freebies.length });
  }
  const pk = await S.adminApi('GET', `/branches/${f.branchId}/ticket-packages?includeArchived=true`);
  const plist = Array.isArray(pk.body) ? pk.body : (pk.body.packages ?? []);
  const t768 = plist.find((p) => p.id === f.ticket768.id);
  const salesAfter = {};
  for (const [k, t] of tills) salesAfter[k] = ((await S.adminApi('GET', `/sales?branchId=${f.branchId}&stationId=${t.id}&limit=50`)).body.sales ?? []).map((s) => ({ receipt: s.receiptNumber ?? null, status: s.status, gross: s.totals?.grossSatang ?? null }));
  C.readBack = {
    at: new Date().toISOString(), stations: mineAfter, salesAfter,
    receptionTill1Devices: (rt1?.devices ?? []).map((d) => d.label),
    myRegistrationsStillWaiting: mineWaiting,
    zzTest768Ticket: { active: t768?.active, archivedAt: t768?.archivedAt },
    liveTicketsWithFreeItems: liveFreebies,
    untouched: stations.filter((s) => /Till 1 \(Pi\)|Till 2 \(test\)|FWBooth1|Reception Till 1/.test(s.name)).map((s) => ({ name: s.name, archived: s.archived ?? !!s.archivedAt })),
  };
  C.finishedAt = new Date().toISOString();
  S.save();
  return C;
}
