import { job } from './job.mjs';
export default async function (S) {
  const { st, save } = S;
  const f = st.fixture, s = st.steps;
  const spec = {
    branch: f.branchId, box: f.boxId, station: f.stationId, terminal: f.terminalId, since: st.steps.passQr.startedAt, date: f.eventsJob.date,
    workshop: f.events.workshop.id, parties: [f.events.party.id, f.events.partyB.id],
    sales: { qr: s.passQr.saleId, cash: s.passCash.passCall.saleId, card: s.passCard.saleId, walkupCash: s.refusals.pass.find((p) => p.label === 'cash').saleId },
    refusedSales: s.refusals.pass.filter((p) => p.label !== 'cash').map((p) => p.saleId),
    refusedPartyPayments: s.refusals.party.filter((p) => p.label !== 'cash').map((p) => p.paymentId),
  };
  const program = `(async()=>{require('tsx/cjs');const assert=require('node:assert/strict'),{Client}=require('pg'),{drizzle}=require('drizzle-orm/node-postgres'),{partyFactsOf}=require('./src/services/analytics-parties.ts');const s=${JSON.stringify(spec)},c=new Client({connectionString:process.env.DATABASE_URL});await c.connect();try{await c.query('BEGIN READ ONLY');await c.query("SET LOCAL statement_timeout='20s'");const pick=(rows,keys)=>rows.map(r=>Object.fromEntries(Object.entries(r).filter(([k])=>keys.test(k))));const q=async(sql,p)=>(await c.query(sql,p)).rows;const saleIds=Object.values(s.sales);const sales=pick(await q('SELECT to_jsonb(x) j FROM pos.sale x WHERE id = ANY($1::uuid[])',[saleIds]).then(r=>r.map(x=>x.j)),/^(id|status|station_id|business_date|channel|receipt_no|receipt_number|code|total_satang|gross_satang|finalised_at|created_at)$/);const stationSales=(await q('SELECT id FROM pos.sale WHERE station_id=$1',[s.station])).map(r=>r.id);const refused=(await q('SELECT count(*)::int n FROM pos.sale WHERE id = ANY($1::uuid[])',[s.refusedSales]))[0].n;const attempts=pick(await q('SELECT to_jsonb(x) j FROM pos.payment_attempt x WHERE sale_id = ANY($1::uuid[]) OR station_id=$2 ORDER BY created_at',[saleIds,s.station]).then(r=>r.map(x=>x.j)),/^(id|sale_id|method|provider|status|amount_satang|device_id|station_id|business_date|created_at|paid_at|updated_at)$/);const holds=pick(await q('SELECT to_jsonb(x) j FROM pos.event_pass_hold x WHERE sale_id = ANY($1::uuid[])',[saleIds]).then(r=>r.map(x=>x.j)),/^(id|sale_id|status|child_id|created_at|applied_at)$/);const links=pick(await q('SELECT to_jsonb(x) j FROM pos.event_attendee_link x WHERE otoapp_event_id=$1 ORDER BY created_at',[s.workshop]).then(r=>r.map(x=>x.j)),/^(id|sale_id|child_id|billing|price_snapshot_satang|created_at)$/);const pp=pick(await q('SELECT to_jsonb(x) j FROM pos.party_payment x WHERE otoapp_event_id = ANY($1::uuid[]) ORDER BY created_at',[s.parties]).then(r=>r.map(x=>x.j)),/^(id|otoapp_event_id|payment_attempt_id|amount_satang|method|station_id|sale_id|created_at|party_date)$/);const refusedPP=(await q('SELECT count(*)::int n FROM pos.party_payment WHERE id = ANY($1::uuid[])',[s.refusedPartyPayments]))[0].n;const cmds=pick(await q("SELECT to_jsonb(x) - 'payload' j, (x.payload->>'deviceId') = $3 AS for_terminal FROM edge.box_command x WHERE box_id=$1 AND created_at >= $2::timestamptz AND kind='terminal_sale' ORDER BY created_at",[s.box,s.since,s.terminal]).then(r=>r.map(x=>({...x.j,for_terminal:x.for_terminal}))),/^(id|kind|status|created_at|completed_at|for_terminal)$/);const op=(await q('SELECT operator_id FROM core.branch WHERE id=$1',[s.branch]))[0].operator_id;const facts=await partyFactsOf(drizzle(c),{id:s.branch,operatorId:op,timezone:'Asia/Bangkok'},s.date);await c.query('COMMIT');console.log(JSON.stringify({event:'s732_db_readback',at:new Date().toISOString(),sales,stationSales,refusedSalesFound:refused,attempts,holds,links,partyPayments:pp,refusedPartyPaymentsFound:refusedPP,terminalCommands:cmds,performanceFacts:facts,passed:true}));}catch(e){await c.query('ROLLBACK').catch(()=>{});console.error(JSON.stringify({event:'s732_db_readback_error',name:e.name,code:e.code??'ASSERTION',message:String(e.message).slice(0,300)}));process.exitCode=1;}finally{await c.end();}})();`;
  const r = await job(program, 's732_db_readback', st.source);
  st.dbReadback = { jobId: r.jobId, status: r.status, result: r.result, errors: r.errors };
  save();
  return st.dbReadback;
}
