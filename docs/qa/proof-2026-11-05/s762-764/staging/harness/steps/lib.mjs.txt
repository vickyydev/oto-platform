const localPhone = async (field, phone) => {
  const prefix = await field.evaluate((el) => /\+(\d{1,4})/.exec(el.parentElement?.textContent ?? '')?.[1] ?? null).catch(() => null);
  if (prefix && phone.startsWith(`+${prefix}`)) return prefix === '66' ? `0${phone.slice(3)}` : phone.slice(prefix.length + 1);
  return phone.replace(/^\+66/, '0');
};
/** Sign in on the POS lock screen as `who` and pick a station by exact name. */
export async function tillSignIn(S, label, stationName, who = 'RECEPTION', viewport = { width: 1600, height: 1000 }) {
  const { page } = await S.newPage(label, viewport);
  await page.goto(S.POS + '/', { waitUntil: 'domcontentloaded' });
  await page.getByRole('heading', { name: 'Oto POS is locked' }).waitFor({ timeout: 60000 });
  const field = page.locator('input[inputmode="tel"], input[type="tel"]').first();
  await field.fill(await localPhone(field, S.need(`STAGING_${who}_PHONE`)));
  await page.locator('input[type="password"]').first().fill(S.need(`STAGING_${who}_PASSWORD`));
  await page.getByRole('button', { name: 'Sign in', exact: true }).click();
  const picker = page.getByRole('heading', { name: 'Which station are you on?' });
  await picker.waitFor({ timeout: 30000 });
  await page.getByRole('button', { name: stationName }).first().click();
  await picker.waitFor({ state: 'hidden', timeout: 30000 });
  return page;
}
export async function unlockIfLocked(S, page, who = 'RECEPTION') {
  if (await page.getByRole('button', { name: 'Unlock', exact: true }).isVisible().catch(() => false)) {
    await page.locator('input[type="password"]').first().fill(S.need(`STAGING_${who}_PASSWORD`));
    await page.getByRole('button', { name: 'Unlock', exact: true }).click();
    await page.getByRole('button', { name: 'Unlock', exact: true }).waitFor({ state: 'hidden', timeout: 30000 });
    await sleep(1500);
    return true;
  }
  return false;
}

export async function readSale(S, id) {
  const r = await S.adminApi('GET', `/sales/${id}`);
  if (r.status !== 200) throw new Error('sale read ' + r.status + ' ' + JSON.stringify(r.body).slice(0, 300));
  return r.body;
}

}
export async function fixScroll(page) {
  await page.evaluate(() => { for (const el of Array.from(document.querySelectorAll('*'))) if (el.scrollLeft) el.scrollLeft = 0; });
}
export async function stepTo(page, label, target) {
  const add = page.getByRole('button', { name: `Add one ${label}` }).first();
  const remove = page.getByRole('button', { name: `Remove one ${label}` }).first();
  const value = add.locator('xpath=preceding-sibling::span[1]');
  for (let i = 0; i < 8; i += 1) {
    const now = Number(await value.textContent());
    if (now === target) return now;
    await (now < target ? add : remove).click();
    await sleep(250);
  }
  return Number(await value.textContent());
}
export async function consoleSignIn(S, label = 'CON', viewport = { width: 1500, height: 1000 }) {
  if (S.pages[label]) return S.pages[label].page;
  const { page } = await S.newPage(label, viewport);
  await page.goto(S.CONSOLE, { waitUntil: 'domcontentloaded' });
  await sleep(4000);
  if (await page.locator('input[type="password"]').count()) {
    const field = page.locator('input[type="tel"], input[inputmode="tel"]').last();
    await field.fill(await localPhone(field, S.need('STAGING_ADMIN_PHONE')));
    await page.locator('input[type="password"]').last().fill(S.need('STAGING_ADMIN_PASSWORD'));
    await page.getByRole('button', { name: /sign in/i }).first().click();
    await sleep(6000);
  }
  if (await page.locator('input[type="password"]').count()) throw new Error('Console still on the sign-in screen');
  return page;
}
