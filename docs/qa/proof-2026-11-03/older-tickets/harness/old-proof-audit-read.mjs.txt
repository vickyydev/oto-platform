import assert from 'node:assert/strict';
import {readFileSync,writeFileSync} from 'node:fs';
import {job} from './qa-render-job.mjs';
const out=process.argv[2],r=JSON.parse(readFileSync(out+'/receipt.json','utf8')),f=r.fixtures;
const s={name:r.name,f,actor:r.identity.accountId,branch:r.identity.branchId};
const program=`(async()=>{const assert=require('node:assert/strict'),{Client}=require('pg'),s=${JSON.stringify(s)},f=s.f,c=new Client({connectionString:process.env.DATABASE_URL,options:'-c default_transaction_read_only=on -c statement_timeout=15000'});await c.connect();try{await c.query('BEGIN READ ONLY');
 const sales=(await c.query("SELECT s.id,s.status,s.gross_satang::int gross, count(p.id)::int attempts,count(p.id) FILTER(WHERE p.method='cash' AND p.status='approved' AND p.amount_satang=s.gross_satang)::int matching_cash FROM pos.sale s LEFT JOIN pos.payment_attempt p ON p.sale_id=s.id WHERE s.id=ANY($1::uuid[]) GROUP BY s.id ORDER BY s.id",[[f.discountSaleId,f.walletSale]])).rows;assert.equal(sales.length,2);for(const sale of sales){assert.equal(sale.status,'finalised');assert.equal(sale.attempts,1);assert.equal(sale.matching_cash,1);assert.equal(sale.gross,sale.id===f.discountSaleId?9000:35000);}
 const wallet=(await c.query('SELECT id,holder_name,balance_satang::int balance,status FROM pos.wallet WHERE id=$1 AND branch_id=$2',[f.walletId,f.walletBranch])).rows[0];assert.equal(wallet.holder_name,s.name+' wallet');assert.equal(wallet.balance,35000);assert.equal(wallet.status,'active');
 const entries=(await c.query('SELECT id,kind,source,amount_satang::int amount,balance_after::int balance,payload,expires_at,business_date FROM pos.wallet_entry WHERE wallet_id=$1 ORDER BY created_at,id',[f.walletId])).rows;assert.deepEqual(entries.map(e=>e.kind),['grant','expire','reactivate']);assert.equal(entries[2].payload.reason,s.name+' reactivation proof');
 const guardian=(await c.query('SELECT id,name,relationship,revoked_at,revoked_by_account_id FROM crm.guardian WHERE id=$1 AND registration_id=$2',[f.pickupId,f.registrationId])).rows[0];assert.equal(guardian.name,s.name+' pickup');assert.equal(guardian.relationship,'ZZ TEST updated family friend');assert.ok(guardian.revoked_at);assert.equal(guardian.revoked_by_account_id,s.actor);
 const family=(await c.query('SELECT r.id,r.branch_id,r.station_id,r.guardian_name,r.guardian_phone,r.photo_file_id,c.id checkin_id,c.child_name,c.status,c.sale_id,c.band_id,c.visit_id,c.checked_in_at FROM crm.registration r JOIN pos.checkin c ON c.registration_id=r.id WHERE r.id=$1',[f.registrationId])).rows;assert.equal(family.length,1);assert.equal(family[0].guardian_name,s.name+' guardian');assert.equal(family[0].child_name,f.childName);assert.equal(family[0].status,'registered');assert.equal(family[0].sale_id,null);assert.equal(family[0].band_id,null);assert.equal(family[0].visit_id,null);assert.equal(family[0].guardian_phone,null);
 const auditTables=(await c.query("SELECT table_schema,table_name FROM information_schema.tables WHERE table_name LIKE '%audit%' ORDER BY table_schema,table_name")).rows;
 const printJobs=(await c.query('SELECT id,kind,status,attempts,error_code FROM edge.print_job WHERE box_id=ANY($1::uuid[]) ORDER BY id',[[f.boxId,f.walletBox]])).rows;
 await c.query('ROLLBACK');console.log(JSON.stringify({event:'old_proof_readback',at:new Date().toISOString(),readOnly:true,sales,wallet,entries:entries.map(e=>({id:e.id,kind:e.kind,source:e.source,amount:e.amount,balance:e.balance,reasonMatches:e.kind==='reactivate'?true:undefined})),guardian:{id:guardian.id,revoked:true,reasonChanged:true,actorMatches:true},family:family.map(x=>({registrationId:x.id,checkinId:x.checkin_id,status:x.status,unpaid:true})),auditTables,printJobs,passed:true}));}finally{await c.end();}})().catch(e=>{console.error(JSON.stringify({event:'old_proof_readback_error',name:e.name,code:e.code??'ASSERTION',message:e.message}));process.exitCode=1;});`;
const result=await job(program,'old_proof_readback',r.deployments.find(d=>d.name==='oto-api-staging').sha);writeFileSync(out+'/audit-readback.json',JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify(result));
