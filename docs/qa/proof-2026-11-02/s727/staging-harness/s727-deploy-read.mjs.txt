import assert from 'node:assert/strict';
import {writeFileSync} from 'node:fs';
import {execFileSync} from 'node:child_process';
import {render,stagingServices} from 'file:///C:/Users/waqar/OneDrive/Desktop/Projects/oto-pos/scripts/agent/render.mjs';
const [sha,ciRun,phase]=process.argv.slice(2);assert.match(sha??'',/^[a-f0-9]{40}$/);assert.match(phase??'',/^[a-z-]+$/);
const root='C:/oto-lanes/codex-727-review-proof-20261103';

const names=['oto-api-staging','oto-pos-web-staging','oto-console-staging'];const services=await stagingServices();

writeFileSync(root+`/docs/qa/proof-2026-11-02/s727/deploy-${phase}.json`,JSON.stringify(receipt,null,2)+'\n');console.log(JSON.stringify(receipt));
