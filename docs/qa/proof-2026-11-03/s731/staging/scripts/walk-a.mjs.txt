// Steps 5 to 9 of the lane's proof plan: F&B and the shop.
import { launch, api, receipt, save, secrets, OUT } from './common.mjs';
import { atTill } from './till.mjs';
import { walker, listening } from './walkcommon.mjs';
const r = receipt();
const s = secrets();
const fam = r.family;
const CHILD = fam.childName;
const browser = await launch();
let page;
let w;
try {
  const t = await atTill(browser, { width: 1440, height: 900 });
  page = t.page;
  w = walker(page, r, 'a');
  w.note('signedIn', t.who);
  const POS = new URL(page.url()).origin;

  // 5. F&B first screen: the band read by the box opens the child's order.
  await page.goto(POS + '/order-station', { waitUntil: 'domcontentloaded' });
  await page.getByRole('heading', { name: 'Scan Wristband' }).waitFor({ timeout: 30000 });
  w.note('wristbandFieldInputMode', await page.getByPlaceholder(/Wristband code/).getAttribute('inputmode'));
  await listening(page);
  await w.scan(s.bandCode, 'F&B first screen: the band');
  await page.getByText(`Allergy / medical alert · ${CHILD}`).first().waitFor({ timeout: 20000 });
  await w.shot('s731-01-fnb-band-read-opens-the-order', [
    ['allergy banner names the child', page.getByText(`Allergy / medical alert · ${CHILD}`)],
    ['allergy text', page.getByText(fam.allergies, { exact: true })],
    ['restriction', page.getByText(`Restriction: ${fam.foodRestrictions}`)],
    ['food not authorized card', page.getByText('Food not authorized')],
  ]);
  // Clear it, then type the short code: the same screen.
  await page.getByRole('button', { name: 'Switch tab' }).first().click();
  await page.getByRole('heading', { name: 'Scan Wristband' }).waitFor({ timeout: 15000 });
  await page.getByPlaceholder(/Wristband code/).fill(fam.bandShortCode);
  await page.getByRole('button', { name: /Load Tab/ }).click();
  await page.getByText(`Allergy / medical alert · ${CHILD}`).first().waitFor({ timeout: 20000 });
  await w.shot('s731-02-fnb-typed-short-code-same-screen', [
    ['allergy banner names the child', page.getByText(`Allergy / medical alert · ${CHILD}`)],
    ['allergy text', page.getByText(fam.allergies, { exact: true })],
    ['restriction', page.getByText(`Restriction: ${fam.foodRestrictions}`)],
    ['food not authorized card', page.getByText('Food not authorized')],
  ]);
  // 6. The credit voucher on that open order: not opened, the order unchanged.
  await listening(page);
  await w.scan(s.walletQr, 'F&B open order: the credit voucher');
  await page.getByText('Credit voucher not opened').first().waitFor({ timeout: 20000 });
  await w.shot('s731-03-fnb-voucher-on-open-order-changes-nothing', [
    ['toast: Credit voucher not opened', page.getByText('Credit voucher not opened')],
    ['toast: Finish or clear it', page.getByText(/This order is still open\. Finish or clear it, then scan again\. Nothing changed\./)],
    ['child still on the order', page.getByText(`Allergy / medical alert · ${CHILD}`)],
  ]);
  // 7. Back on the first screen, the credit voucher opens its tab with its credit.
  await page.getByRole('button', { name: 'Switch tab' }).first().click();
  await page.getByRole('heading', { name: 'Scan Wristband' }).waitFor({ timeout: 15000 });
  await listening(page);
  await w.scan(s.walletQr, 'F&B first screen: the credit voucher');
  await page.getByText(/฿[\d,.]+ credit/).first().waitFor({ timeout: 20000 });
  await w.shot('s731-04-fnb-voucher-opens-its-tab-with-credit', [
    ['credit shown', page.getByText(/฿350(\.00)? credit/)],
    ['first screen gone', page.getByRole('heading', { name: 'Scan Wristband' }), false],
  ]);
  await page.getByRole('button', { name: 'Switch tab' }).first().click().catch(() => {});
  await page.waitForTimeout(1500);

  // 8. The shop's first screen: the band opens the sale on the child's tab.
  await page.goto(POS + '/merch-station', { waitUntil: 'domcontentloaded' });
  await page.getByRole('heading', { name: 'Retail Sale' }).waitFor({ timeout: 30000 });
  await listening(page);
  await w.scan(s.bandCode, 'Shop first screen: the band');
  await page.getByRole('heading', { name: 'Retail Sale' }).waitFor({ state: 'hidden', timeout: 20000 });
  await page.getByText(CHILD).first().waitFor({ timeout: 20000 });
  await w.shot('s731-05-shop-band-read-opens-the-sale', [
    ['child on the sale', page.getByText(CHILD)],
    ['first screen gone', page.getByRole('heading', { name: 'Retail Sale' }), false],
  ]);
  await page.getByRole('button', { name: 'Switch tab' }).first().click().catch(() => {});
  await page.waitForTimeout(1500);

  // 9. A guest sale at payment: the band changes nothing and says so.
  await page.goto(POS + '/merch-station', { waitUntil: 'domcontentloaded' });
  await page.getByRole('heading', { name: 'Retail Sale' }).waitFor({ timeout: 30000 });
  await listening(page);
  await page.getByRole('button', { name: /No wristband/ }).click();
  await page.getByText(r.fixtures.shopItemName).first().click({ timeout: 20000 });
  const sized = page.getByRole('dialog').getByRole('button', { name: /฿|Add/ }).first();
  if (await sized.isVisible().catch(() => false)) await sized.click();
  const charge = page.getByRole('button', { name: /^Charge ฿/ });
  await charge.waitFor({ timeout: 20000 });
  const owed = (await charge.textContent()).trim();
  w.note('shopChargeBefore', owed);
  await charge.click();
  await charge.waitFor({ state: 'hidden', timeout: 20000 });
  await page.waitForTimeout(1500);
  await w.scan(s.bandCode, 'Shop at payment: the band');
  await page.getByText(/^Wristband .* not opened$/).first().waitFor({ timeout: 20000 });
  await w.shot('s731-06-shop-band-at-payment-changes-nothing', [
    ['toast: Wristband not opened', page.getByText(`Wristband ${fam.bandShortCode} not opened`)],
    ['toast: Finish this sale first', page.getByText('Finish this sale first. Nothing changed.')],
    ['child not on the sale', page.getByText(CHILD), false],
  ]);
  await page.getByRole('button', { name: /^Back/ }).first().click();
  await charge.waitFor({ timeout: 15000 });
  const after = (await charge.textContent()).trim();
  w.note('shopChargeAfter', after);
  w.note('shopTotalUnchanged', after === owed);
  // Let the guest sale go: clear its cart.
  const clear = page.getByRole('button', { name: /^Clear$/ }).first();
  if (await clear.isVisible().catch(() => false)) await clear.click();
  await page.waitForTimeout(1000);
  const confirm = page.getByRole('alertdialog').getByRole('button', { name: /clear|discard|yes|confirm/i });
  if (await confirm.count()) await confirm.first().click().catch(() => {});
  await page.waitForTimeout(1500);
  w.note('finishedAt', new Date().toISOString());
  await api(page, 'POST', '/auth/sign-out', {});
  console.log(JSON.stringify(r.walk.a, null, 1));
} catch (e) {
  r.walk ??= {};
  r.walk.a ??= {};
  r.walk.a.error = String(e?.message ?? e).slice(0, 600);
  save(r);
  if (page) await page.screenshot({ path: `${OUT}/s731-error-walk-a.png` }).catch(() => {});
  console.log('ERROR', r.walk.a.error);
  console.log(JSON.stringify(r.walk.a, null, 1).slice(0, 3000));
  if (page) await api(page, 'POST', '/auth/sign-out', {}).catch(() => {});
} finally {
  await browser.close();
}
