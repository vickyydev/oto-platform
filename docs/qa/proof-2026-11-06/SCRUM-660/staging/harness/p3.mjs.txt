// Steps 2-3: the ZZ TEST birthdays-team login, on each conversation: Book Date through its dialog, then the
// team's replies through the inbox composer (the channel is the simulator: nothing reaches a real customer).
import { api, assert, adminConsole, persona, INBOX, loadState, saveState, writeReceipt, chromium, shot, signOut, SCRATCH } from './lib.mjs';
const state = loadState();
const note = (name, detail = {}) => { state.log.push({ step: name, ...detail, at: new Date().toISOString() }); saveState(state); console.log('OK', name, JSON.stringify(detail).slice(0, 900)); };
const PLAN = {
  en: {
    book: { date: '2027-03-21', startTime: '14:00', kids: '12', childName: 'ZZ TEST Child EN', age: '6' },
    replies: [
      'ZZ TEST thank you, let me check the date for you.',
      'ZZ TEST final confirmation: Sunday 21 March 2027 from 14:00 to 16:00, 12 kids and 8 adults, programme ZZ TEST magic show with entertainment host ZZ TEST Mint, food ZZ TEST pizza set and fruit platter, cake at 15:30, Party Host ZZ GM, setup by the Event Department ready by 13:30.',
    ],
  },
  th: {
    book: { date: '2027-03-20', startTime: '14:00', kids: '12', childName: 'ZZ TEST Child TH', age: '5' },
    replies: ['ยืนยัน ZZ TEST วันเสาร์ 20 มีนาคม 2027 เวลา 14.00 น. ถึง 16.00 น. เด็ก 12 คน ผู้ใหญ่ 6 คน อาหาร ZZ TEST พิซซ่า และน้ำผลไม้ เป่าเค้กเวลา 15.30 น. พิธีกร ZZ GM'],
  },
  ru: {
    book: { date: '2027-03-13', startTime: '14:00', kids: '10', childName: 'ZZ TEST Child RU', age: '7' },
    replies: ['ZZ TEST подтверждение: суббота 13 марта 2027 с 14:00 до 16:00, 10 детей и 6 взрослых, программа ZZ TEST шоу мыльных пузырей, еда ZZ TEST пицца и сок, торт в 15:30, ведущий ZZ GM'],
  },
  none: {
    book: { date: '2027-04-10', startTime: '14:00', kids: '12', childName: 'ZZ TEST Child NONE', age: '6' },
    replies: ['ZZ TEST thank you, our birthday packages start from the Jungle package. Which date would you like?'],
  },
};
state.team ??= {};
const browser = await chromium.launch({ channel: 'msedge' });
const admin = await adminConsole(browser);
let ag;
try {
  ag = await persona(browser, admin, state.agent);
  const me = ag.me;
  state.agentMe = { name: me.viewer?.name ?? me.name ?? null, assign: me.capabilities?.assign, writes: me.capabilities?.writes, teams: (me.teams ?? []).map((t) => ({ name: t.name, member: t.member })) };
  saveState(state);
  note('The birthdays-team login in the inbox', state.agentMe);
  const p = ag.page;
  for (const key of ['en', 'th', 'ru', 'none']) {
    const conv = state.convs[key];
    const plan = PLAN[key];
    const T = (state.team[key] ??= {});
    await p.goto(INBOX + '/?c=' + conv.id);
    await p.getByRole('button', { name: /^(Book Date|Date Booked)$/ }).first().waitFor({ timeout: 60000 });
    await p.waitForTimeout(1500);
    if (!T.booked) {
      await p.getByRole('button', { name: /^Book Date$/ }).first().click();
      const dialog = p.getByRole('dialog');
      await dialog.locator('input[aria-label="Date"]').waitFor({ timeout: 30000 });
      await p.waitForTimeout(600);
      const prefilled = {
        date: await dialog.locator('input[aria-label="Date"]').inputValue(),
        startTime: await dialog.locator('input[aria-label="Start time"]').inputValue(),
        park: await dialog.locator('select[aria-label="Park"]').evaluate((el) => el.selectedOptions[0]?.textContent ?? ''),
        kids: await dialog.locator('input[aria-label="Kids"]').inputValue(),
        bookingName: await dialog.locator('input[aria-label="Booking name"]').inputValue(),
      };
      await dialog.locator('input[aria-label="Date"]').fill(plan.book.date);
      await dialog.locator('input[aria-label="Start time"]').fill(plan.book.startTime);
      await dialog.locator('input[aria-label="Kids"]').fill(plan.book.kids);
      await dialog.locator('input[aria-label="Child\'s name"]').fill(plan.book.childName);
      await dialog.locator('input[aria-label="Turning age"]').fill(plan.book.age);
      assert.ok(/Central Floresta/.test(prefilled.park), 'park');
      if (key === 'en') await shot(p, 's660-st-02-book-date-confirm.png');
      await dialog.getByRole('button', { name: /^Book date$/ }).click();
      const outcome = await Promise.race([
        p.waitForSelector('[data-testid="event-chip"]', { timeout: 60000 }).then(() => 'chip'),
        p.waitForSelector('[role="dialog"] [role="alert"]', { timeout: 60000 }).then(() => 'alert'),
      ]);
      if (outcome !== 'chip') {
        const alert = await p.textContent('[role="dialog"] [role="alert"]');
        await shot(p, 'fail-book-date-' + key + '.png');
        throw new Error('Book Date refused (' + key + '): ' + alert);
      }
      await p.waitForTimeout(2000);
      T.booked = { at: new Date().toISOString(), prefilled, set: plan.book, chip: (await p.locator('[data-testid="event-chip"]').textContent())?.trim() };
      saveState(state);
      note('Step 2: Book Date (' + key + ')', T.booked);
    }
    T.sent ??= [];
    for (let i = T.sent.length; i < plan.replies.length; i++) {
      const text = plan.replies[i];
      const box = p.getByLabel('Reply', { exact: true });
      await box.waitFor({ timeout: 30000 });
      await box.fill(text);
      const [resp] = await Promise.all([
        p.waitForResponse((r) => /\/api\/inbox\/conversations\/[^/]+\/messages$/.test(new URL(r.url()).pathname) && r.request().method() === 'POST', { timeout: 60000 }),
        p.getByRole('button', { name: 'Send', exact: true }).click(),
      ]);
      let body = null;
      try { body = await resp.json(); } catch {}
      if (![200, 201].includes(resp.status())) {
        await shot(p, 'fail-reply-' + key + '.png');
        throw new Error('reply refused (' + key + ') ' + resp.status() + ' ' + JSON.stringify(body?.error ?? body).slice(0, 300));
      }
      T.sent.push({ at: new Date().toISOString(), status: resp.status(), messageId: body?.message?.id ?? body?.id ?? null, providerStatus: body?.message?.providerStatus ?? null, chars: text.length });
      saveState(state);
      note('Step 3: the team replied through the composer (' + key + ' #' + (i + 1) + ')', T.sent[i]);
      await p.waitForTimeout(2500);
    }
    // The platform's pick, read as this person (a read; no model is called).
    const pick = await api(ag.ctx, INBOX, 'GET', '/inbox/conversations/' + conv.id + '/beo-confirmation');
    T.pick = { scanned: pick.scanned, eventId: pick.eventId, picked: pick.message ? (pick.message.body === plan.replies[plan.replies.length - 1] ? 'the final confirmation' : 'another message') : null, messageId: pick.message?.id ?? null, authorName: pick.message?.authorName ?? null, bodyLanguage: pick.message?.bodyLanguage ?? null };
    saveState(state);
    note('The platform\'s pick (' + key + ')', T.pick);
    if (key === 'none') assert.equal(T.pick.picked, null);
    else assert.equal(T.pick.picked, 'the final confirmation');
    if (key === 'en') {
      await p.reload();
      await p.getByRole('button', { name: /^Create BEO$/ }).first().waitFor({ timeout: 60000 });
      await p.waitForTimeout(2500);
      await shot(p, 's660-st-03-inbox-chat-with-the-team-confirmation.png');
    }
  }
  writeReceipt('step2-3-book-date-and-replies.json', { login: state.agentMe, conversations: Object.fromEntries(Object.entries(state.team).map(([k, v]) => [k, { conversationId: state.convs[k].id, booked: v.booked, replies: v.sent, platformPick: v.pick }])) });
} catch (e) {
  console.error('FAIL', e.message.split('\n')[0]);
  if (ag) for (const pg of ag.ctx.pages()) await pg.screenshot({ path: SCRATCH + '/fail-p3-' + ag.ctx.pages().indexOf(pg) + '.png' }).catch(() => {});
  process.exitCode = 1;
} finally {
  await signOut(ag);
  await signOut(admin);
  await browser.close();
}
