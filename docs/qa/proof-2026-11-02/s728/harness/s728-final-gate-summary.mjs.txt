import assert from 'node:assert/strict';
import {readFileSync,writeFileSync} from 'node:fs';
import {createHash} from 'node:crypto';
import {execFileSync} from 'node:child_process';
const root='C:/oto-lanes/codex-box-outage-release-20261104';
const out=root+'/docs/qa/proof-2026-11-02/s728/local-gates';
const hash=b=>createHash('sha256').update(b).digest('hex');
const totals={files:0,passed:0,failed:0,skipped:0,todo:0};
const shards=[];
for(const [shard,key,report] of [[1,'frozen-api1','s728-final-frozen-api1.json'],[2,'v4-api2','s728-final-v4-api2.json'],[3,'v4-api3','s728-final-v4-api3.json']]){
 const receipt=JSON.parse(readFileSync('C:/oto-lanes/s728-gates-'+key+'-exit.json','utf8'));
 assert.equal(receipt.exitCode,0);assert.equal(receipt.signal,null);
 const raw=readFileSync('C:/oto-lanes/'+report),r=JSON.parse(raw);
 assert.equal(r.success,true);assert.equal(r.numFailedTests,0);
 const files=r.testResults.map(f=>({file:f.name.split('/apps/api/')[1],passed:f.assertionResults.filter(t=>t.status==='passed').length,failed:f.assertionResults.filter(t=>t.status==='failed').length,skipped:f.assertionResults.filter(t=>['pending','skipped'].includes(t.status)).length,todo:f.assertionResults.filter(t=>t.status==='todo').length}));
 assert.ok(files.every(f=>f.file?.startsWith('test/')&&f.failed===0));
 totals.files+=files.length;totals.passed+=r.numPassedTests;totals.failed+=r.numFailedTests;totals.skipped+=r.numPendingTests;totals.todo+=r.numTodoTests;
 shards.push({shard,source:receipt.sha,exitCode:0,reportSha256:hash(raw),receipt:key+'.json',files});
}
const all=shards.flatMap(s=>s.files.map(f=>f.file));assert.equal(new Set(all).size,all.length);
const original=JSON.parse(readFileSync(out+'/api-complete-run-before-inventory-correction.json','utf8'));
assert.deepEqual([...all].sort(),original.shards.flatMap(s=>s.files.map(f=>f.file)).sort());
assert.deepEqual(totals,{files:444,passed:6174,failed:0,skipped:401,todo:4});
const git=a=>execFileSync('git',a,{cwd:root,encoding:'utf8'}).trim();
assert.equal(git(['diff','--name-only','8306bca8','HEAD','--','apps/api','packages/db','packages/shared','packages/box-agent']),'apps/api/test/routes-guarded.test.ts');
assert.equal(git(['diff','--name-only','682dfdc4','HEAD','--','.',':(exclude)docs']),'');
const result={at:new Date().toISOString(),totals,shards,releaseCandidate:git(['rev-parse','HEAD']),sameFileInventory:true,apiRuntimeUnchanged:true,onlyApiTestChangeSinceOriginalRun:'apps/api/test/routes-guarded.test.ts',interpretation:'Complete shards 2 and 3 pass on unchanged API runtime. Entire shard 1 repeated after correcting the exact route inventory. Original failed and operator-interrupted runs remain separate.'};
writeFileSync(out+'/api-final.json',JSON.stringify(result,null,2)+'\n');
console.log(JSON.stringify({totals,source:result.releaseCandidate,shards:shards.map(s=>({shard:s.shard,source:s.source,exitCode:s.exitCode}))}));
