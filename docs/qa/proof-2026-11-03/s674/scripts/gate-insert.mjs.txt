// Three ZZ TEST gate passages for the ZZ TEST adult band at the ZZ TEST gate lane, written by a one-off job on
// oto-api-staging because no gate on staging can send one (no gate simulator; the park's gate box is not used).
// Every row is checked to be this proof's own before anything is written; detail.zzTest marks them for cleanup.
import { job } from './job.mjs';
import { receipt, save, LIVE } from './common.mjs';
const r = receipt(); const f = r.fixtures;
const adult = r.sale.bands.find((b) => b.kind === 'adult');
const spec = { saleId: r.sale.id, bandId: adult.id, branchId: f.branchId, stationId: f.gateStationId, boxId: f.gateBoxId, mark: 'SCRUM-674 synthetic gate passage' };
const program = `(async()=>{const assert=require('node:assert/strict'),{Client}=require('pg'),{randomUUID}=require('node:crypto');const s=${JSON.stringify(spec)};const c=new Client({connectionString:process.env.DATABASE_URL});await c.connect();try{await c.query('BEGIN');await c.query("SET LOCAL statement_timeout='15s'");const b=(await c.query("select b.id from pos.band b join pos.sale sa on sa.id=b.sale_id where b.id=$1 and b.sale_id=$2 and b.branch_id=$3 and b.kind='adult'",[s.bandId,s.saleId,s.branchId])).rows;assert.equal(b.length,1);const st=(await c.query("select s.id from core.station s join core.box x on x.id=s.box_id where s.id=$1 and s.kind='gate' and s.name like 'ZZ TEST SCRUM-674 %' and s.branch_id=$2 and x.id=$3 and x.name like 'ZZ TEST SCRUM-674 %' and x.status='unclaimed' and x.registered_at is null",[s.stationId,s.branchId,s.boxId])).rows;assert.equal(st.length,1);assert.equal((await c.query("select count(*)::int n from pos.band_event where band_id=$1 and kind in ('entry','exit','denied','timeout','alarm')",[s.bandId])).rows[0].n,0);const rows=[['entry',{direction:'entry',side:'left'},180],['denied',{direction:'entry',side:'left',reason:'ANTI_PASSBACK'},120],['exit',{direction:'exit',side:'right'},60]];const ids=[];for(const [kind,d,ago] of rows){const id=randomUUID();ids.push(id);await c.query("insert into pos.band_event (id,band_id,kind,station_id,box_id,detail,created_at) values ($1,$2,$3,$4,$5,$6::jsonb,now()-make_interval(secs=>$7))",[id,s.bandId,kind,s.stationId,s.boxId,JSON.stringify({...d,zzTest:s.mark}),ago]);}const back=(await c.query("select id,kind,detail->>'direction' direction,detail->>'reason' reason,created_at from pos.band_event where band_id=$1 order by created_at",[s.bandId])).rows;await c.query('COMMIT');console.log(JSON.stringify({event:'s674_gate_insert',at:new Date().toISOString(),ids,back}));}catch(e){await c.query('ROLLBACK').catch(()=>{});console.log(JSON.stringify({event:'s674_gate_insert_error',name:e.name,code:e.code??null,message:String(e.message).slice(0,300)}));process.exitCode=1;}finally{await c.end();}})();`;
const res = await job(program, 's674_gate_insert', LIVE);
r.gateInsert = { jobId: res.jobId, source: res.source, status: res.status, startedAt: res.startedAt, rows: res.rows };
save(r);
console.log(JSON.stringify(r.gateInsert, null, 1));
