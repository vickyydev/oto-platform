// Execute the actual nested request and bounded body reader; inject transport/receipt sinks only.
// This intentionally avoids reimplementing cookie capture, unknown-issuance accounting or dispatch order.
function actualRequestHarness(kind) {
  const source = fs.readFileSync(new URL('./admin639-staging-proof.mjs', import.meta.url), 'utf8');
  const requestStart = source.indexOf('  async function request(c, method, pathname,');
  const requestEnd = source.indexOf('\n  async function census(c)', requestStart);
  const readerStart = source.indexOf('async function boundedJson(response)');
  const readerEnd = source.indexOf('\n/** Local preflight seam', readerStart);
  assert.ok(requestStart >= 0 && requestEnd > requestStart && readerStart >= 0 && readerEnd > readerStart);
  const requestBody = source.slice(requestStart, requestEnd), readerBody = source.slice(readerStart, readerEnd);
  const receipt = { operations: [], pending: 0, unresolvedAuth: 0 }, events = [], transport = [];
  const fetch = async (url, options) => {
    const pathname = new URL(url).pathname;
    transport.push({ method: options.method, path: pathname, observedCookie: options.headers.cookie === 'oto_session=ZZ_TEST_SYNTHETIC_COOKIE' });
    assert.equal(events.at(-1), 'intent', 'The actual request must durably record intent before transport.');
    if (pathname === '/api/auth/sign-in') {
      const body = kind === 'malformed' ? 'ZZ TEST non-json response' : new ReadableStream({ start(controller) {
        controller.enqueue(new TextEncoder().encode('{'));
        const keepAlive = setTimeout(() => {}, 1_000); // AbortSignal.timeout is unref'ed; keep only this synthetic stalled read alive.
        options.signal.addEventListener('abort', () => { clearTimeout(keepAlive); controller.error(new Error('SYNTHETIC_BODY_TIMEOUT')); }, { once: true });
      } });
      return new Response(body, { status: 200, headers: { 'set-cookie': 'oto_session=ZZ_TEST_SYNTHETIC_COOKIE; Path=/; HttpOnly; Secure' } });
    }
    if (pathname === '/api/auth/sign-out') return Response.json({ ok: true });
    if (pathname === '/api/me') return Response.json({ error: { code: 'UNAUTHENTICATED' } }, { status: 401 });
    throw new Error('UNEXPECTED_SYNTHETIC_TRANSPORT_PATH');
  };
  class SyntheticProofError extends Error { constructor(code) { super(code); this.code = code; } }
  const factory = new Function('requestAllowed', 'ORIGIN', 'TIMEOUT', 'MAX_BODY', 'ProofError', 'need', 'codeOf', 'now', 'receipt', 'persist', 'save', 'safe', 'fetch',
    `${readerBody}\nreturn ${requestBody.trim()};`);
  const request = factory(requestAllowed, ORIGIN, 20, 2_000_000, SyntheticProofError,
    (condition, code) => { if (!condition) throw new SyntheticProofError(code); }, error => error.code ?? 'UNCLASSIFIED_PROOF_ERROR',
    () => new Date().toISOString(), receipt, row => events.push(row.terminal ? 'terminal' : 'intent'), () => {},
    () => { assert.equal(receipt.pending, 0); assert.equal(receipt.unresolvedAuth, 0); }, fetch);
  return { request, receipt, transport };
}
