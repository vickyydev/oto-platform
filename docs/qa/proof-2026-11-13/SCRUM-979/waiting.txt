S.phase = 'Simulated card silence reaches staff confirmation';
const p = S.till;
await S.unlock('till');
S.receipt.numbers.waiting = S.free[1];
await S.walkIn(p, S.free[1]);
await S.ticket(p, 1, 1);
const sale = await S.saleWritten('sale-waiting');
S.waiting = { sale: sale.id, startedAt: new Date().toISOString() };
const configured = await S.call('POST', '/payments/terminal-simulator', { action: 'terminal.outcome', deviceId: S.dev.card.id, outcome: 'no_response' });
S.assert.equal(configured.status, 200);
S.assert.equal(configured.body.applied, true);
await p.getByRole('heading', { name: 'Card', exact: true }).first().click();
const next = p.waitForResponse(r => r.request().method() === 'POST' && new URL(r.url()).pathname === '/api/payments/attempts');
await p.getByRole('button', { name: 'Start card payment', exact: true }).click();
const response = await next;
S.assert.equal(response.status(), 200);
S.waiting.attempt = (await response.json()).attempt.id;
S.receipt.reads.waiting = S.waiting; S.save();
await S.expect.poll(async () => (await S.attempt(S.waiting.attempt)).body?.attempt?.status, { timeout: 180000, intervals: [5000] }).toBe('awaiting_staff_confirmation');
await S.shot(p, '02-card-waiting-for-staff.png');
S.waiting.heldAt = new Date().toISOString(); S.save();
return S.waiting;
