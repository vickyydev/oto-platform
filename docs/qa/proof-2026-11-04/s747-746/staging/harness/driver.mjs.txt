const tillPage = async (label) => {

  const page = await ctx.newPage();
  page.on('response', async (res) => {
    const req = res.request();
    const path = new URL(res.url()).pathname;
    if (!path.startsWith('/api/') || !watched.test(path)) return;
    let body = null; try { body = await res.json(); } catch {}
    let reqBody = null; try { reqBody = req.postDataJSON(); } catch {}
    st.network.push({ at: new Date().toISOString(), till: label, method: req.method(), path, status: res.status(), request: shrink(reqBody), response: shrink(body) });
    save();
  });
  pages[label] = { ctx, page };
  return pages[label];
};

};
