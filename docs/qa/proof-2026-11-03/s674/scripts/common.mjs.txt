const typedPhone = async (field, phone) => {
  const prefix = await field.evaluate((el) => /\+(\d{1,4})/.exec(el.parentElement?.textContent ?? '')?.[1] ?? null).catch(() => null);
  if (!prefix || !phone.startsWith(`+${prefix}`)) return phone;
  const national = phone.slice(prefix.length + 1);
  return prefix === '66' ? `0${national}` : national;
};

/** Sign in on the Console's own sign-in screen with a phone and password (never printed). */
export async function consoleSignIn(browser, phone, password, viewport = { width: 1500, height: 1000 }) {
  const ctx = await browser.newContext({ viewport });
  const page = await ctx.newPage();
  await page.goto(CONSOLE, { waitUntil: 'domcontentloaded' });
  await page.waitForTimeout(4000);
  if (await page.locator('input[type="password"]').count()) {
    const field = page.locator('input[type="tel"], input[inputmode="tel"]').last();
    await field.fill(await typedPhone(field, phone));
    await page.locator('input[type="password"]').last().fill(password);
    await page.getByRole('button', { name: /sign in/i }).first().click();
    await page.waitForTimeout(6000);
  }
  return { ctx, page };
}
export async function consoleSession(browser, viewport) {
  const s = await consoleSignIn(browser, need('STAGING_ADMIN_PHONE'), need('STAGING_ADMIN_PASSWORD'), viewport);
  if (await s.page.locator('input[type="password"]').count()) throw new Error('Console still on the sign-in screen');
  return s;
}

/** fetch from inside a signed-in page, with an Idempotency-Key on every write. */
export const api = (page, method, path, body) =>
  page.evaluate(async ({ method, path, body }) => {
    const headers = body === undefined ? {} : { 'content-type': 'application/json' };
    if (method !== 'GET') headers['idempotency-key'] = crypto.randomUUID();
    const r = await fetch('/api' + path, { method, credentials: 'include', headers, body: body === undefined ? undefined : JSON.stringify(body) });
    const text = await r.text();
    let json = null;
    try { json = JSON.parse(text); } catch { json = text.slice(0, 400); }
    return { status: r.status, body: json };
  }, { method, path, body });

export const ok = (res, what) => {
  if (res.status !== 200 && res.status !== 201) throw new Error(what + ' status ' + res.status + ' ' + JSON.stringify(res.body).slice(0, 400));
  return res.body;
};
