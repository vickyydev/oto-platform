// Step 10: the OTO App activity logbook, filtered to BEO AI Fill, as the ZZ TEST login (manager) sees it.
import { adminConsole, persona, hand, APP, loadState, saveState, chromium, shot, signOut, SCRATCH } from './lib.mjs';
const state = loadState();
const note = (name, detail = {}) => { state.log.push({ step: name, ...detail, at: new Date().toISOString() }); saveState(state); console.log('OK', name, JSON.stringify(detail).slice(0, 1500)); };
const browser = await chromium.launch({ channel: 'msedge' });
const admin = await adminConsole(browser);
let ag;
try {
  ag = await persona(browser, admin, state.agent);
  const a = await hand(ag.ctx, 'oto_app');
  await a.setViewportSize({ width: 1440, height: 1100 });
  let rows = null;
  a.on('response', async (r) => { if (new URL(r.url()).pathname === '/api/activity-logs' && new URL(r.url()).searchParams.get('types') === 'beo_ai_fill') { try { rows = await r.json(); } catch {} } });
  await a.goto(APP + '/activity-logbook');
  await a.getByTestId('select-activity-type').waitFor({ timeout: 90000 });
  await a.waitForTimeout(1500);
  await a.getByTestId('select-activity-type').click();
  await a.getByRole('option', { name: 'BEO AI Fill' }).click();
  await a.waitForTimeout(3000);
  await shot(a, 's660-st-16-activity-logbook-beo-ai-fill-rows.png');
  state.logbook = { shown: rows ? rows.logs.map((l) => ({ type: l.activityType, summary: l.summaryText, createdAt: l.createdAt })) : null, totalCount: rows?.totalCount ?? null };
  saveState(state);
  note('Step 10: the activity logbook, BEO AI Fill', state.logbook);
} catch (e) {
  console.error('FAIL', e.message.split('\n')[0]);
  if (ag) for (const pg of ag.ctx.pages()) await pg.screenshot({ path: SCRATCH + '/fail-p6-' + ag.ctx.pages().indexOf(pg) + '.png' }).catch(() => {});
  process.exitCode = 1;
} finally {
  await signOut(ag);
  await signOut(admin);
  await browser.close();
}
