export async function consoleSession() {
  const phone = need('STAGING_ADMIN_PHONE');
  const password = need('STAGING_ADMIN_PASSWORD');
  const browser = await launch();
  const ctx = await browser.newContext({ viewport: { width: 1500, height: 1000 } });
  const page = await ctx.newPage();
  await page.goto(CONSOLE, { waitUntil: 'domcontentloaded' });
  await page.waitForTimeout(4000);
  if (await page.locator('input[type="password"]').count()) {
    const field = page.locator('input[type="tel"], input[inputmode="tel"]').last();
    const prefix = await field.evaluate((el) => /\+(\d{1,4})/.exec(el.parentElement?.textContent ?? '')?.[1] ?? null).catch(() => null);
    const typed = prefix && phone.startsWith(`+${prefix}`) ? (prefix === '66' ? `0${phone.slice(3)}` : phone.slice(prefix.length + 1)) : phone;
    await field.fill(typed);
    await page.locator('input[type="password"]').last().fill(password);
    await page.getByRole('button', { name: /sign in/i }).first().click();
    await page.waitForTimeout(6000);
  }
  if (await page.locator('input[type="password"]').count()) throw new Error('Console still on the sign-in screen');
  return { browser, ctx, page };
}

/** fetch from inside the signed-in Console page. */
export const api = (page, method, path, body) =>
  page.evaluate(async ({ method, path, body }) => {
    const headers = body === undefined ? {} : { 'content-type': 'application/json' };
    if (method !== 'GET') headers['idempotency-key'] = crypto.randomUUID();
    const r = await fetch('/api' + path, { method, credentials: 'include', headers, body: body === undefined ? undefined : JSON.stringify(body) });
    const text = await r.text();
    let json = null;
    try { json = JSON.parse(text); } catch { json = text.slice(0, 400); }
    return { status: r.status, body: json };
  }, { method, path, body });

export async function openTemplates(page, branchName) {
  await page.goto(`${CONSOLE}/catalogue/print-templates`, { waitUntil: 'domcontentloaded' });
  await page.waitForTimeout(5000);
  const picker = page.locator('select').filter({ has: page.locator('option', { hasText: branchName }) }).first();
  await picker.selectOption({ label: branchName });
  await page.waitForTimeout(5000);
}

export const mainText = async (page) => (await page.locator('main').innerText().catch(() => '')).slice(0, 1500);
