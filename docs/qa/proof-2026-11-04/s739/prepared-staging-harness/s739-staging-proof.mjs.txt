async function live() {
  const rows = [];
  for (const name of ['oto-api-staging', 'oto-pos-web-staging', 'oto-pos-staging', 'oto-console-staging', 'oto-launcher-staging', 'oto-booth-staging', 'oto-app-staging']) {
    const service = services.find(s => s.name === name); assert.ok(service);
    const d = (await render('GET', '/services/' + service.id + '/deploys?limit=1'))[0]?.deploy;
    assert.equal(d?.status, 'live', 'Deployment is not ready: ' + name);
    let unchangedBuildContextTree;
    if (name === 'oto-app-staging') {
      // This independent app is path-filtered. Verify its entire actual build
      // context is identical instead of restarting an unrelated service.
      assert.equal(service.rootDir, 'apps/oto-app');
      const tree = commit => execFileSync('git', ['rev-parse', commit + ':apps/oto-app'], { cwd: root, encoding: 'utf8' }).trim();
      unchangedBuildContextTree = tree(sha);
      assert.equal(tree(d.commit.id), unchangedBuildContextTree);
    } else assert.equal(d.commit.id, sha, 'Live source differs: ' + name);
    rows.push({ name, serviceId: service.id, deployId: d.id, source: d.commit.id,
      ...(unchangedBuildContextTree ? { unchangedBuildContextTree } : {}) });
  }
  return rows;
}
async function settings() {
  const apiService = services.find(s => s.name === 'oto-api-staging');

  assert.equal(rows.length, 57, 'API setting count changed');
  assert.ok(rows.some(r => r.key === 'PGW_PROVIDER' && r.value === 'simulator'), 'Staging gateway is not the simulator');
  return rows.map(r => ({ name: r.key, sha256: createHash('sha256').update(r.value).digest('hex') })).sort((a, b) => a.name.localeCompare(b.name));
}
