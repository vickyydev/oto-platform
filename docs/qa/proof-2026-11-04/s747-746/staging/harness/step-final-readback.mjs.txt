// Read only, after cleanup: everything this proof made is retired, nothing waits, nobody is signed in on our tills;
// and the invoice segments as the LIVE deployed code computes them for T1, T2, PI1, PI2 and our two codes.
import { job } from './job.mjs?v=1';
export default async function (S) {
  const { st, save, SOURCE } = S;
  const f = st.fixture;
  const spec = { branch: f.branchId, box: f.boxId, stations: [f.tills.four.id, f.tills.six.id], def: f.voucher.definitionId, checkin: f.family.checkinId, holdSale: st.steps.typedVoucher.saleId };
  const program = `(async()=>{require('tsx/cjs');const {Client}=require('pg');const pay=require('../../packages/shared/src/payments.ts');const s=${JSON.stringify(spec)},c=new Client({connectionString:process.env.DATABASE_URL});await c.connect();try{await c.query('BEGIN READ ONLY');await c.query("SET LOCAL statement_timeout='20s'");const q=async(sql,p)=>(await c.query(sql,p)).rows;
const liveSegments=Object.fromEntries(['T1','T2','PI1','PI2','ZQ7C','ZZ747A','TOOLONG7'].map(k=>[k,pay.invoiceStationSegment(k)]));
const stations=await q("SELECT name,code_prefix,archived_at IS NOT NULL archived FROM core.station WHERE id = ANY($1::uuid[])",[s.stations]);
const box=(await q("SELECT name,archived_at IS NOT NULL archived FROM core.box WHERE id=$1",[s.box]))[0]??null;
const sessions=(await q("SELECT count(*)::int n FROM core.session WHERE station_id = ANY($1::uuid[]) AND expires_at>now() AND revoked_at IS NULL",[s.stations]))[0].n;
const leases=await q("SELECT lease_id IS NULL AS released FROM edge.station_session WHERE station_id = ANY($1::uuid[])",[s.stations]);
const pending=(await q("SELECT count(*)::int n FROM pos.payment_attempt WHERE station_id = ANY($1::uuid[]) AND status NOT IN ('approved','declined','cancelled','not_found')",[s.stations]))[0].n;
const attempts=await q("SELECT invoice_no,provider,status FROM pos.payment_attempt WHERE station_id = ANY($1::uuid[]) ORDER BY created_at",[s.stations]);
const openSales=(await q("SELECT count(*)::int n FROM pos.sale WHERE station_id = ANY($1::uuid[]) AND status <> 'finalised' AND status <> 'void' AND status <> 'voided'",[s.stations]))[0].n;
const vouchers=await q("SELECT status,held_sale_id IS NULL AS not_held,right(code,4) last4 FROM promo.voucher WHERE voucher_definition_id=$1",[s.def]);
const def=(await q("SELECT active,archived_at IS NOT NULL archived FROM promo.voucher_definition WHERE id=$1",[s.def]))[0];
const heldAnywhere=(await q("SELECT count(*)::int n FROM promo.voucher WHERE held_sale_id=$1",[s.holdSale]))[0].n;
const stay=(await q("SELECT status FROM pos.checkin WHERE id=$1",[s.checkin]))[0]??null;
const qaAudit=(await q("SELECT count(*)::int n FROM core.audit_log WHERE action='qa.voucher.void' AND entity_id::text IN (SELECT id::text FROM promo.voucher WHERE voucher_definition_id=$1)",[s.def]))[0].n;
await c.query('COMMIT');console.log(JSON.stringify({event:'s747_final_readback',at:new Date().toISOString(),liveSegments,stations,box,sessions,leases,pending,attempts,openSales,vouchers,def,heldAnywhere,stay,qaAudit}));}catch(e){await c.query('ROLLBACK').catch(()=>{});console.log(JSON.stringify({event:'s747_final_readback_error',name:e.name,code:e.code??null,message:String(e.message).slice(0,400)}));process.exitCode=1;}finally{await c.end();}})();`;
  const r = await job(program, 's747_final_readback', SOURCE);
  st.finalReadback = { jobId: r.jobId, status: r.status, result: r.result, errors: r.errors };
  save();
  return st.finalReadback;
}
