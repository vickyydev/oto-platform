const redact = text => String(text)
  .replace(/postgres(?:ql)?:\/\/[^\s"'`<>]+/gi, '<redacted database URL>')

function call(command, args, options = {}) {
  const result = spawnSync(command, args, { cwd: ROOT, encoding: 'utf8', maxBuffer: 256 * 1024 * 1024, ...options });
  need(result.status === 0 && result.signal === null, command === 'docker' ? 'DOCKER_OPERATION_FAILED' : 'LOCAL_COMMAND_FAILED');
  return result.stdout;
}
const docker = (...args) => call('docker', args).trim();

// Shared verbatim between host and runtime census. The root example is an explicit non-secret input.
function sourcePath(p) {
  if (!p || !/^[A-Za-z0-9_./-]+$/.test(p) || p.split('/').includes('..')) return false;
  if (/^(?:docs|imports|services)\//.test(p) || /(?:^|\/)(?:node_modules|dist|\.git)(?:\/|$)/.test(p)) return false;
  if (p === '.env.example') return true;
  if (/(?:^|\/)\.env(?:\.|\/|$)/.test(p)) return false;
  return path.posix.basename(p) === '.gitignore' || /^(?:apps|packages|scripts)\//.test(p)
    || /^\.github\/workflows\//.test(p)
    || /^(?:package\.json|pnpm-lock\.yaml|pnpm-workspace\.yaml|turbo\.json|eslint\.config\.mjs|tsconfig[^/]*\.json|render\.yaml|\.npmrc|\.pnpmfile\.cjs)$/.test(p);
}
function fileRow(base, p) {
  const absolute = path.resolve(base, p), realBase = fs.realpathSync(base), realFile = fs.realpathSync(absolute);
  need(realFile.startsWith(realBase + path.sep) && fs.lstatSync(absolute).isFile(), 'SOURCE_FILE_BOUNDARY_REQUIRED');
  return { path: p, sha256: hash(fs.readFileSync(absolute)) };
}
function sources() {
  const names = call('git', ['ls-files', '--cached', '--others', '--exclude-standard', '-z']).split('\0').filter(Boolean);
  need(names.filter(p => sourcePath(p)).length > 3000, 'FULL_SOURCE_REQUIRED');
  return [...new Set(names.filter(p => sourcePath(p)))].sort().map(p => fileRow(ROOT, p));
}
const dependencies = rows => rows.filter(row => /(?:^|\/)(?:package\.json|package-lock\.json|pnpm-lock\.yaml|pnpm-workspace\.yaml)$/.test(row.path));

function checkedPg() {
  const receiptFile = path.join(OUT, 'api-runtime.json'), receipt = read(receiptFile), current = inspect(PG_ID);
  need(receipt.id === PG_ID && receipt.name === PG && receipt.mode === 'api' && receipt.temporary === true
    && receipt.ready === true && receipt.user === 'oto' && receipt.database === 'postgres'
    && ['localhost', '127.0.0.1'].includes(receipt.host)
    && Number.isInteger(receipt.port) && receipt.port >= 1024 && receipt.port <= 65535
    && receipt.baselineContainerIds?.length === 10 && new Set(receipt.baselineContainerIds).size === 10
    && receipt.baselineContainerIds.every(id => /^[a-f0-9]{64}$/.test(id) && id !== PG_ID), 'OWN_PG_RECEIPT_REQUIRED');
  need(current.Id === PG_ID && current.Name === '/' + PG && current.State.Running
    && Object.entries(LABELS).every(([k, v]) => receipt.labels?.[k] === v && current.Config.Labels?.[`oto.agent.${k}`] === v)
    && current.Mounts.length === 0 && Object.hasOwn(current.HostConfig.Tmpfs ?? {}, '/var/lib/postgresql/data'), 'OWN_TMPFS_PG_REQUIRED');
  const bindings = current.HostConfig.PortBindings ?? {}, bound = bindings['5432/tcp'];
  const actualBindings = current.NetworkSettings.Ports ?? {}, actualBound = actualBindings['5432/tcp'];
  need(Object.keys(bindings).length === 1 && bound?.length === 1
    && ['127.0.0.1', '::1'].includes(bound[0].HostIp)
    && (bound[0].HostPort === '' || Number(bound[0].HostPort) === receipt.port)
    && Object.keys(actualBindings).length === 1 && actualBound?.length === 1
    && ['127.0.0.1', '::1'].includes(actualBound[0].HostIp)
    && Number(actualBound[0].HostPort) === receipt.port, 'OWN_PG_LOOPBACK_BINDING_REQUIRED');
  const allIds = new Set(docker('ps', '-aq', '--no-trunc').split(/\s+/));
  need(receipt.baselineContainerIds.every(id => allIds.has(id)), 'BASELINE_CONTAINERS_MUST_REMAIN');
  return { receiptSha256: hash(fs.readFileSync(receiptFile)), baselineContainerIds: receipt.baselineContainerIds,
    id: PG_ID, image: current.Image, priorNetworkIds: Object.values(current.NetworkSettings.Networks).map(n => n.NetworkID).sort() };
}
function checkedApp(setup) {
  const current = inspect(setup.appId), net = JSON.parse(docker('network', 'inspect', NETWORK))[0];
  need(current.Id === setup.appId && current.Name === '/' + APP && current.Image === IMAGE && current.State.Running
    && current.Mounts.length === 0 && Object.keys(current.HostConfig.PortBindings ?? {}).length === 0
    && current.Config.WorkingDir === '/workspace' && current.Path === 'bash'
    && equal(current.Args, ['-c', 'exec tail -f /dev/null'])
    && Object.entries(LABELS).every(([k, v]) => current.Config.Labels?.[`oto.agent.${k}`] === v)
    && net.Id === setup.networkId && net.Name === NETWORK && net.Internal === true
    && Object.entries(LABELS).every(([k, v]) => net.Labels?.[`oto.agent.${k}`] === v)
    && Object.keys(current.NetworkSettings.Networks).length === 1
    && current.NetworkSettings.Networks[NETWORK]?.NetworkID === net.Id
    && inspect(PG_ID).NetworkSettings.Networks[NETWORK]?.Aliases?.includes('pg'), 'OWN_APP_BOUNDARY_REQUIRED');
  return current;
}
function idle(setup) {
  checkedApp(setup);
  const commands = docker('top', setup.appId, '-eo', 'pid,comm').split(/\r?\n/).slice(1).map(s => s.trim()).filter(Boolean);
  need(commands.length === 1 && /^\d+\s+tail$/.test(commands[0]), 'OWN_APP_MUST_BE_IDLE');
}

/** Recovery only records these exact already-created resources; it never creates or removes one. */
export function recoverSetupFromIntent({ explicitExecute, expectedNetworkId } = {}) {
  need(explicitExecute === true && /^[a-f0-9]{64}$/.test(expectedNetworkId ?? '')
    && expectedNetworkId.startsWith('5395752fd165'), 'EXPLICIT_RECOVERY_NETWORK_REQUIRED');
  need(!fs.existsSync(path.join(OUT, 'integration-setup.json')), 'PRESERVE_EXISTING_SETUP');
  const intentFile = path.join(OUT, 'integration-setup-intent.json'), intent = read(intentFile), pg = checkedPg();
  need(intent.ticket === 'SCRUM-637' && intent.run === LABELS.run && intent.root === ROOT
    && intent.network === NETWORK && intent.app === APP && intent.image === IMAGE
    && intent.pg?.id === PG_ID && intent.pg.receiptSha256 === pg.receiptSha256 && intent.pg.image === pg.image
    && equal(intent.pg.baselineContainerIds, pg.baselineContainerIds)
    && equal(pg.priorNetworkIds, [...new Set([...intent.pg.priorNetworkIds, expectedNetworkId])].sort()), 'EXACT_SETUP_INTENT_REQUIRED');
  const appId = '7fd9b57eca90a3a4462fc29e073669384d1329cf53a3eed555baacea86e11eae';
  const setup = { ticket: 'SCRUM-637', run: LABELS.run, at: new Date().toISOString(), networkId: expectedNetworkId,
    network: NETWORK, appId, app: APP, image: IMAGE, pg: intent.pg, appHostPorts: false, staging: false,
    recoveredFromIntent: true, setupIntentSha256: hash(fs.readFileSync(intentFile)) };
  idle(setup);
  const app = inspect(appId), net = JSON.parse(docker('network', 'inspect', NETWORK))[0];
  need(Number.isFinite(Date.parse(intent.at)) && Date.parse(app.Created) >= Date.parse(intent.at)
    && Date.parse(net.Created) >= Date.parse(intent.at)
    && equal(Object.keys(net.Containers ?? {}).sort(), [appId, PG_ID].sort()), 'EXACT_RECOVERY_RESOURCES_REQUIRED');
  write('integration-setup.json', setup);
  return { recoveredFromIntent: true, appId, networkId: expectedNetworkId, staging: false };
}

function dependencyState(setup, spec) {
  const state = JSON.parse(docker('exec', setup.appId, 'node', '-e', dependencyProgram));
  need(/^v22\./.test(state.node) && state.pnpm === JSON.parse(fs.readFileSync(path.join(ROOT, 'package.json'), 'utf8')).packageManager.split('@')[1], 'FROZEN_RUNTIME_VERSIONS_REQUIRED');
  need(equal(state.inputs, spec.dependencies), 'CACHED_DEPENDENCY_INPUTS_CHANGED_REINSTALL_REQUIRED');
  return state;
}

function verify(setup, spec, frozenDependencies) {
  checkedApp(setup); checkedPg();
  need(equal(sources(), spec.rows) && equal(DOCS.map(p => fileRow(ROOT, p)), spec.docs), 'HOST_INPUTS_CHANGED');
  const runtime = runtimeState(setup);
  need(equal(runtime.rows, spec.rows) && equal(runtime.docs, spec.docs), 'FULL_RUNTIME_INPUTS_CHANGED');
  const deps = dependencyState(setup, spec);
  need(equal(deps, frozenDependencies), 'INSTALLED_DEPENDENCIES_CHANGED');
  return { sourceFiles: spec.rows.length, sourceManifestSha256: hash(JSON.stringify(spec.rows)),
    docsSha256: hash(JSON.stringify(spec.docs)), dependencyMetadataSha256: hash(JSON.stringify(deps)),
    hostUnchanged: true, runtimeUnchanged: true, supplementalDocsUnchanged: true, installedDependenciesUnchanged: true };
}

async function setupRuntime() {

  const pg = checkedPg();
  need(!docker('network', 'ls', '--format', '{{.Name}}').split(/\r?\n/).includes(NETWORK)
    && !docker('ps', '-a', '--format', '{{.Names}}').split(/\r?\n/).includes(APP), 'FRESH_RESOURCE_NAMES_REQUIRED');
  write('integration-setup-intent.json', { ticket: 'SCRUM-637', run: LABELS.run, at: new Date().toISOString(),
    root: ROOT, network: NETWORK, app: APP, image: IMAGE, pg, staging: false });
  const networkId = docker('network', 'create', '--internal', ...Object.entries(LABELS).flatMap(([k, v]) => ['--label', `oto.agent.${k}=${v}`]), NETWORK);
  need(/^[a-f0-9]{64}$/.test(networkId), 'NETWORK_ID_REQUIRED');

  const appId = docker('run', '-d', '--name', APP, '--network', NETWORK, '--workdir', '/workspace', '--log-driver', 'none',
    ...Object.entries(LABELS).flatMap(([k, v]) => ['--label', `oto.agent.${k}=${v}`]), '--entrypoint', 'bash', IMAGE, '-c', 'exec tail -f /dev/null');
  need(/^[a-f0-9]{64}$/.test(appId), 'APP_ID_REQUIRED');
  const setup = { ticket: 'SCRUM-637', run: LABELS.run, at: new Date().toISOString(), networkId, network: NETWORK,
    appId, app: APP, image: IMAGE, pg, appHostPorts: false, staging: false };

async function syncRuntime() {
  const setup = read(path.join(OUT, 'integration-setup.json')); idle(setup);
  const rows = sources(), docs = DOCS.map(p => fileRow(ROOT, p));
  need(rows.some(r => r.path === '.env.example') && rows.some(r => r.path === '.gitignore'), 'EXAMPLE_AND_IGNORE_INPUTS_REQUIRED');
  const stamp = Date.now(), prefix = `combined-source-${stamp}`, spec = { ticket: 'SCRUM-637', rows, docs, dependencies: dependencies(rows) };
  write(prefix + '.json', spec);
  docker('cp', path.join(OUT, prefix + '.json'), setup.appId + ':' + SPEC);
  // Reusing cached node_modules is valid only while its original package/lock
  // inputs still match. Replacing those files first would hide required installs.
  const originalDependencies = dependencyState(setup, spec);
  write(prefix + '-intent.json', { at: new Date().toISOString(), appId: setup.appId, freshFullCopy: true,
    sourceManifestSha256: hash(JSON.stringify(rows)), sourceFiles: rows.length, dependencyMetadataSha256: hash(JSON.stringify(originalDependencies)) });
  const pruned = JSON.parse(docker('exec', setup.appId, 'node', '-e', pruneProgram));
  const files = [...rows.map(r => r.path), ...DOCS];
  fs.writeFileSync(path.join(OUT, prefix + '.txt'), files.join('\n') + '\n', { flag: 'wx' });
  call('tar', ['-cf', path.join(OUT, prefix + '.tar'), '-T', path.join(OUT, prefix + '.txt')]);
  docker('cp', path.join(OUT, prefix + '.tar'), setup.appId + ':/tmp/admin637-combined-source.tar');
  docker('exec', setup.appId, 'tar', '-xf', '/tmp/admin637-combined-source.tar', '-C', '/workspace');
  const indexProgram = `const fs=require('fs'),cp=require('child_process'),s=JSON.parse(fs.readFileSync('${SPEC}','utf8'));
fs.writeFileSync('${INDEX_LIST}',s.rows.map(r=>r.path).join('\\0')+'\\0');
for(const args of [['init','-q'],['config','core.autocrlf','false'],['config','core.filemode','false'],['add','-f','--pathspec-from-file=${INDEX_LIST}','--pathspec-file-nul']]){
const r=cp.spawnSync('git',args,{cwd:'/workspace',encoding:'utf8'});if(r.status!==0)throw Error('RUNTIME_INDEX_FAILED');}`;
  docker('exec', setup.appId, 'node', '-e', indexProgram);
  const verified = verify(setup, spec, originalDependencies);
  const binding = { stamp, specFile: prefix + '.json', frozenDependencies: originalDependencies, ...verified };
  write(prefix + '-verified.json', { ...binding, pruned, archiveSha256: hash(fs.readFileSync(path.join(OUT, prefix + '.tar'))) });
  // A later sync is another full copy, never a delta inferred from a previous container.
  fs.writeFileSync(path.join(OUT, 'combined-current.json'), JSON.stringify(binding, null, 2) + '\n');
  return { ...verified, freshFullCopy: true, stamp };
}
async function runGate(name, command) {
  need(/^[a-z0-9][a-z0-9-]{0,79}$/.test(name ?? '') && typeof command === 'string' && command.trim().length > 0
    && command.length < 32_000 && !/[\0\r|]/.test(command) && !/postgres(?:ql)?:\/\//i.test(command), 'UNPIPED_NAMED_COMMAND_REQUIRED');
  need(!['json', 'log', 'progress.jsonl', 'intent.json'].some(ext => fs.existsSync(path.join(OUT, `${name}.${ext}`))), 'PRESERVE_PRIOR_GATE');
  const setup = read(path.join(OUT, 'integration-setup.json')), current = read(path.join(OUT, 'combined-current.json'));
  const spec = read(path.join(OUT, current.specFile)), before = verify(setup, spec, current.frozenDependencies), startedAt = new Date().toISOString();
  const privateDb = new URL('postgres:' + '/'.repeat(2) + 'pg/postgres'); privateDb.username = 'oto'; privateDb.port = '5432';
  write(name + '.intent.json', { name, command, startedAt, before, appId: setup.appId, staging: false });
  fs.writeFileSync(path.join(OUT, name + '.progress.jsonl'), '', { flag: 'wx' });
  const result = await new Promise(resolve => {
    const args = ['exec', '-e', 'TEST_DATABASE_URL=' + privateDb.href, '-e', 'DATABASE_URL=' + privateDb.href,
      '-e', 'CONSOLE_E2E_SERVER_URL=' + privateDb.href, '-e', 'CONSOLE_DIGEST_ENABLED=false', '-e', 'ALERT_WEBHOOK_URL=',
      setup.appId, 'bash', '-lc', 'set -eo pipefail\n' + command];
    const child = spawn('docker', args, { stdio: ['ignore', 'pipe', 'pipe'] });
    let stdout = '', stderr = '', tail = '', files = 0, outputLimitExceeded = false, childError = false, progressWriteError = false;
    const cap = () => { if (stdout.length + stderr.length > 256 * 1024 * 1024) { outputLimitExceeded = true; child.kill('SIGTERM'); } };
    child.stdout.on('data', bytes => {
      stdout += bytes.toString(); tail += bytes.toString();
      const lines = tail.split('\n'); tail = lines.pop() ?? '';
      for (const line of lines) {
        const clean = line.replace(/\u001b\[[0-?]*[ -/]*[@-~]/g, '');
        const match = /^\s*[✓❯]\s+(test\/[A-Za-z0-9_./-]+\.test\.ts)\s+\((\d+) tests?(?:[^)]*)\)\s+(\d+)ms/.exec(clean);
        if (match) { const progress = { name, at: new Date().toISOString(), completedFiles: ++files, file: match[1], tests: Number(match[2]), failureObserved: /\d+ failed/.test(clean) };
          try { fs.appendFileSync(path.join(OUT, name + '.progress.jsonl'), JSON.stringify(progress) + '\n'); }
          catch { progressWriteError = true; }
          if (files % 10 === 0 || progress.failureObserved) process.stdout.write(JSON.stringify(progress) + '\n'); }
      } cap();
    });
    child.stderr.on('data', bytes => { stderr += bytes.toString(); cap(); });
    child.on('error', () => { childError = true; });
    child.on('close', (exitCode, signal) => resolve({ stdout, stderr, exitCode, signal, outputLimitExceeded, childError, progressWriteError }));
  });
  const log = redact(result.stdout + result.stderr); fs.writeFileSync(path.join(OUT, name + '.log'), log, { flag: 'wx' });
  let after = null, verificationError = null;
  try { after = verify(setup, spec, current.frozenDependencies); } catch (error) { verificationError = safeCode(error); }
  const passed = result.exitCode === 0 && result.signal === null && !result.childError && !result.outputLimitExceeded && !result.progressWriteError && !verificationError;
  const receipt = { ticket: 'SCRUM-637', run: LABELS.run, name, command, startedAt, completedAt: new Date().toISOString(),
    exitCode: result.exitCode, signal: result.signal, childError: result.childError, outputLimitExceeded: result.outputLimitExceeded, progressWriteError: result.progressWriteError,
    gateTerminalKnown: !result.childError && !result.outputLimitExceeded && result.signal === null && result.exitCode !== null,
    passed, verificationError, logSha256: hash(log), sourceManifestSha256: before.sourceManifestSha256, before, after,
    appId: setup.appId, image: IMAGE, digestDefaultOff: true, staging: false };
  write(name + '.json', receipt); return receipt;
}

export async function main(argv = process.argv.slice(2)) {
  const [mode, name, command] = argv;
  need(['setup', 'sync', 'run'].includes(mode) && argv.length === (mode === 'run' ? 3 : 1), 'EXPLICIT_MODE_REQUIRED');
  need(fs.realpathSync(ROOT).replaceAll('\\', '/').toLowerCase() === ROOT.toLowerCase(), 'HOST_ROOT_REQUIRED');
  need(fs.existsSync(OUT), 'OWNER_RUNTIME_DIRECTORY_REQUIRED');
  const result = mode === 'setup' ? await setupRuntime() : mode === 'sync' ? await syncRuntime() : await runGate(name, command);
  process.stdout.write(JSON.stringify(result) + '\n');
  if (mode === 'run' && !result.passed) process.exitCode = Number.isInteger(result.exitCode) && result.exitCode > 0 ? result.exitCode : 2;
  return result;
}
if (process.argv[1] && path.resolve(process.argv[1]).toLowerCase() === path.resolve(fileURLToPath(import.meta.url)).toLowerCase()) {
  main().catch(error => { process.stderr.write(safeCode(error) + '\n'); process.exitCode = 2; });
}
