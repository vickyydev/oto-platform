S.phase = 'Normal card approval on a simulated terminal';
const p = S.till;
S.receipt.numbers.paid = S.free[0];
await S.walkIn(p, S.free[0]);
await S.ticket(p, 1, 1);
const sale = await S.saleWritten('sale-card');
S.paid = { sale: sale.id };
await S.expect(S.paymentStep(p)).toBeVisible();
await p.getByRole('heading', { name: 'Card', exact: true }).first().click();
const next = p.waitForResponse(r => r.request().method() === 'POST' && new URL(r.url()).pathname === '/api/payments/attempts');
await p.getByRole('button', { name: 'Start card payment', exact: true }).click();
const response = await next;
S.assert.equal(response.status(), 200);
S.paid.attempt = (await response.json()).attempt.id;
await p.getByText('Payment Successful', { exact: true }).waitFor({ timeout: 120000 });
await S.shot(p, '01-card-paid.png', [p.locator('img'), p.locator('span.font-mono')]);
const detail = await S.sale(sale.id);
S.assert.equal(detail.sale.status, 'finalised');
S.assert.equal(detail.attempts.length, 1);
S.assert.equal(detail.attempts[0].status, 'approved');
const commands = await S.termCommandsFor(S.paid.attempt);
S.assert.equal(commands.length, 1);
const audit = await S.audit(S.paid.attempt);
S.paid.commands = commands;
S.paid.auditRecoveryCount = (audit.body.events ?? audit.body.entries ?? audit.body.rows ?? []).filter(x => x.action === 'payment.attempt.result_from_command').length;
S.receipt.reads.paid = { ...S.paid, status: detail.sale.status, attempts: detail.attempts.map(a => ({ method: a.method, status: a.status, amountSatang: a.amountSatang })) };
S.point('normal-card', 'PASS', ['01-card-paid.png'], S.receipt.reads.paid);
return S.receipt.reads.paid;
