// Step 7: clean up through the screens. Archive the ZZ TEST event with the OTO App's own archive button,
// then mark the ZZ TEST conversation handled in the inbox. Record the chip's archived state.
import { api, assert, adminConsole, persona, hand, INBOX, APP, loadState, saveState, writeReceipt, chromium, shot, signOut } from './lib.mjs';
const state = loadState();
const note = (name, detail = {}) => { state.log.push({ step: name, ...detail, at: new Date().toISOString() }); saveState(state); console.log('OK', name, JSON.stringify(detail).slice(0, 900)); };
const EV = state.event.id;
state.cleanup ??= {};
const browser = await chromium.launch({ channel: 'msedge' });
const admin = await adminConsole(browser);
let ag;
try {
  ag = await persona(browser, admin, state.agent);
  if (!state.cleanup.archived) {
    const a = await hand(ag.ctx, 'oto_app');
    await a.setViewportSize({ width: 1920, height: 1000 });
    await a.goto(APP + '/studio/events/kanban');
    const card = a.locator(`[data-testid="card-event-${EV}"]`);
    await card.waitFor({ timeout: 60000 });
    await card.click();
    await a.getByTestId('button-archive-beo').waitFor({ timeout: 60000 });
    await a.waitForTimeout(1500);
    await shot(a, 's775-st-10-beo-editor-archive-button.png');
    const [resp] = await Promise.all([
      a.waitForResponse((r) => new URL(r.url()).pathname === `/api/admin/events/${EV}/archive` && r.request().method() === 'POST', { timeout: 60000 }),
      a.getByTestId('button-archive-beo').click(),
    ]);
    assert.equal(resp.status(), 200, 'archive ' + resp.status());
    await a.waitForTimeout(2500);
    await shot(a, 's775-st-11-board-after-archive-tentative-column-stays.png');
    const cols = (await a.locator('h3').allTextContents()).map((c) => c.trim()).filter(Boolean);
    state.cleanup.archived = { at: new Date().toISOString(), status: resp.status(), columnsAfter: cols.slice(0, 10) };
    saveState(state);
    note('Clean-up: archived the ZZ TEST event with the BEO editor\'s archive button', state.cleanup.archived);
  }
  const p = ag.page;
  await p.goto(INBOX + '/?c=' + state.conv.id);
  await p.locator('[data-testid="event-chip"]').waitFor({ timeout: 60000 });
  await p.waitForTimeout(1500);
  const chip = (await p.locator('[data-testid="event-chip"]').textContent())?.trim();
  const archivedButton = await p.getByRole('button', { name: /Event archived/ }).count();
  await shot(p, 's775-st-12-inbox-chip-event-archived.png');
  state.cleanup.chip = { chip, eventArchivedButton: archivedButton };
  saveState(state);
  note('The inbox chip after the archive', state.cleanup.chip);
  assert.ok(/· Archived$/.test(chip ?? ''), 'chip not archived');
  if (!state.cleanup.handled) {
    await p.getByRole('button', { name: /^Mark as Handled$/ }).first().click();
    const d = p.getByRole('dialog');
    await d.getByRole('button', { name: 'Yes, mark as handled' }).waitFor({ timeout: 30000 });
    await shot(p, 's775-st-13-mark-handled-confirm.png');
    const [resp] = await Promise.all([
      p.waitForResponse((r) => /\/api\/inbox\/conversations\/[^/]+\/handle$/.test(new URL(r.url()).pathname) && r.request().method() === 'POST', { timeout: 60000 }),
      d.getByRole('button', { name: 'Yes, mark as handled' }).click(),
    ]);
    assert.equal(resp.status(), 200, 'handle ' + resp.status());
    await p.waitForTimeout(2000);
    state.cleanup.handled = { at: new Date().toISOString(), status: resp.status() };
    saveState(state);
    note('Clean-up: marked the ZZ TEST conversation handled', state.cleanup.handled);
  }
  await p.goto(INBOX + '/?tab=handled&c=' + state.conv.id);
  await p.locator('[data-testid="event-chip"]').waitFor({ timeout: 60000 });
  await p.waitForTimeout(1500);
  await shot(p, 's775-st-14-handled-conversation-keeps-archived-event.png');
  const c = await api(ag.ctx, INBOX, 'GET', '/inbox/conversations/' + state.conv.id);
  state.cleanup.convRead = { status: (c.conversation ?? c).status, handledAt: (c.conversation ?? c).handledAt ?? null, event: c.event ?? (c.conversation ?? c).event ?? null };
  saveState(state);
  note('The conversation after clean-up', state.cleanup.convRead);
} catch (e) {
  console.error('FAIL', e.message.split('\n')[0]);
  if (ag) for (const pg of ag.ctx.pages()) await pg.screenshot({ path: 'C:/Users/waqar/AppData/Local/Temp/claude/9d4c56906b777e8e/317fcfce-2754-44c9-bec5-9a762ab18d03/scratchpad/s775p/fail-p5-' + ag.ctx.pages().indexOf(pg) + '.png' }).catch(() => {});
  process.exitCode = 1;
} finally {
  await signOut(ag);
  await signOut(admin);
  await browser.close();
}
