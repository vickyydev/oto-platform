// Step 4: End of Day shows the counts and refuses to close over the part-paid sale; the close route answers 409 PART_PAID_SALES.
export default async function (S) {
  const { sleep } = S.lib;
  const f = S.st.fixture;
  const page = S.pages.A.page;
  await page.goto(S.POS + '/end-of-day', { waitUntil: 'domcontentloaded' });
  const card = page.getByTestId('eod-open-sales');
  await S.expect(card).toBeVisible({ timeout: 30000 });
  await card.scrollIntoViewIfNeeded();
  await sleep(1200);
  const counts = { partPaid: await page.getByTestId('eod-part-paid-count').innerText(), held: await page.getByTestId('eod-held-count').innerText() };
  const reason = await page.getByTestId('close-day-blocked-reason').innerText();
  const disabled = await page.getByRole('button', { name: 'Close Day' }).isDisabled();
  await S.shot('A', 's774-st-11-end-of-day-counts');
  await page.getByRole('button', { name: 'Close Day' }).scrollIntoViewIfNeeded();
  await sleep(800);
  await S.shot('A', 's774-st-12-close-day-shut-with-its-reason');
  // The close route, called only when the day reads open with this run's part-paid sale on it: it must refuse.
  const date = f.businessDate;
  const eod = await S.adminApi('GET', `/branches/${f.branchId}/end-of-day?date=${date}`);
  const mine = (eod.body.openSales?.rows ?? []).find((r) => r.saleId === f.ben.saleId && r.kind === 'part_paid');
  if (eod.body.status !== 'open' || !mine) throw new Error('guard: the day is not open with this run\'s part-paid sale on it; the close route was not called');
  const req = await (await import('node:module')).createRequire(S.LANE + '/apps/pos/package.json')('@playwright/test').request.newContext({ baseURL: S.POS });
  const si = await req.post('/api/auth/sign-in', { data: { phone: S.need('STAGING_MANAGER_PHONE'), password: S.need('STAGING_MANAGER_PASSWORD') } });
  if (si.status() !== 200) throw new Error('manager sign-in ' + si.status());
  const res = await req.post(`/api/branches/${f.branchId}/end-of-day/close`, { data: { date, actuals: [], countedSatang: null, floatLeftSatang: null }, headers: { 'idempotency-key': S.randomUUID() } });
  const body = await res.json().catch(() => null);
  await req.post('/api/auth/sign-out', { headers: { 'idempotency-key': S.randomUUID() } }).catch(() => {});
  await req.dispose();
  const after = await S.adminApi('GET', `/branches/${f.branchId}/end-of-day?date=${date}`);
  S.st.steps.s4 = { at: new Date().toISOString(), date, counts, reason, closeDayDisabled: disabled, apiOpenSales: eod.body.openSales, close: { by: 'Khun Lek (Manager)', status: res.status(), code: body?.error?.code, message: body?.error?.message, details: body?.error?.details }, dayStillOpen: after.body.status };
  S.save();
  return S.st.steps.s4;
}
