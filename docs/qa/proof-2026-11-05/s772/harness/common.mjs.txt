const localPhone = async (field, phone) => {
  const prefix = await field.evaluate((el) => /\+(\d{1,4})/.exec(el.parentElement?.textContent ?? '')?.[1] ?? null).catch(() => null);
  if (prefix && phone.startsWith(`+${prefix}`)) return prefix === '66' ? `0${phone.slice(3)}` : phone.slice(prefix.length + 1);
  return phone.replace(/^\+66/, '0');
};

export async function consoleSession(browser, viewport = { width: 1500, height: 1000 }) {
  const phone = need('STAGING_ADMIN_PHONE');
  const password = need('STAGING_ADMIN_PASSWORD');
  const ctx = await browser.newContext({ viewport, timezoneId: 'Asia/Bangkok' });
  ctx.setDefaultTimeout(30000);
  const page = await ctx.newPage();
  await page.goto(CONSOLE, { waitUntil: 'domcontentloaded' });
  await page.waitForTimeout(4000);
  if (await page.locator('input[type="password"]').count()) {
    const field = page.locator('input[type="tel"], input[inputmode="tel"]').last();
    await field.fill(await localPhone(field, phone));
    await page.locator('input[type="password"]').last().fill(password);
    await page.getByRole('button', { name: /sign in/i }).first().click();
    await page.waitForTimeout(6000);
  }
  if (await page.locator('input[type="password"]').count()) throw new Error('Console still on the sign-in screen');
  return { ctx, page };
}

/** Sign in on the POS lock screen as reception and pick a station by exact name. */
export async function tillSession(browser, stationName, viewport = { width: 1600, height: 1050 }) {
  const ctx = await browser.newContext({ viewport, timezoneId: 'Asia/Bangkok' });
  ctx.setDefaultTimeout(30000);
  const page = await ctx.newPage();
  await page.goto(POS + '/', { waitUntil: 'domcontentloaded' });
  await page.getByRole('heading', { name: 'Oto POS is locked' }).waitFor({ timeout: 60000 });
  const field = page.locator('input[inputmode="tel"], input[type="tel"]').first();
  await field.fill(await localPhone(field, need('STAGING_RECEPTION_PHONE')));
  await page.locator('input[type="password"]').first().fill(need('STAGING_RECEPTION_PASSWORD'));
  await page.getByRole('button', { name: 'Sign in', exact: true }).click();
  const picker = page.getByRole('heading', { name: 'Which station are you on?' });
  await picker.waitFor({ timeout: 30000 });
  await page.getByRole('button', { name: stationName }).first().click();
  await picker.waitFor({ state: 'hidden', timeout: 30000 });
  return { ctx, page };
}

/** fetch from inside a signed-in page, with an Idempotency-Key on every write. */
export const api = (page, method, path, body) =>
  page.evaluate(async ({ method, path, body }) => {
    const headers = body === undefined ? {} : { 'content-type': 'application/json' };
    if (method !== 'GET') headers['idempotency-key'] = crypto.randomUUID();
    const r = await fetch('/api' + path, { method, credentials: 'include', headers, body: body === undefined ? undefined : JSON.stringify(body) });
    const text = await r.text();
    let json = null;
    try { json = JSON.parse(text); } catch { json = text.slice(0, 400); }
    return { status: r.status, body: json };
  }, { method, path, body });

export const ok = (res, what) => {
  if (res.status !== 200 && res.status !== 201) throw new Error(what + ' status ' + res.status + ' ' + JSON.stringify(res.body).slice(0, 400));
  return res.body;
};

/** Type the member's phone on the till keypad and look them up, as reception does. */
export async function lookup(page, phone) {
  await page.getByText('Membership Check', { exact: true }).first().waitFor({ timeout: 30000 });
  const digits = phone.replace(/^\+66/, '');
  for (const d of digits) await page.getByRole('button', { name: d, exact: true }).last().click();
  await page.getByRole('button', { name: 'Find my membership' }).click();
  const skip = page.getByRole('button', { name: 'Skip for now' });
  const type = page.getByText('Select Customer Type');
  await Promise.race([skip.waitFor({ timeout: 20000 }).catch(() => {}), type.waitFor({ timeout: 20000 }).catch(() => {})]);
  if (await skip.isVisible().catch(() => false)) await skip.click();
  await type.waitFor({ timeout: 20000 });
}
