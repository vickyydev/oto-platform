const until = async (label: string, run: () => Promise<any>, ms = 90000) => {
  const end = Date.now() + ms;
  let last: any;
  while (Date.now() < end) {
    try { const v = await run(); if (v) return v; } catch (e) { last = e; }
    await sleep(500);
  }
  throw Error('Timed out: ' + label + (last ? ' (' + String((last as Error).message ?? last).slice(0, 200) + ')' : ''));
};
const listen = (server: any, port = 0) => new Promise<number>((resolve, reject) => {
  server.once('error', reject);
  server.listen(port, '127.0.0.1', () => resolve(server.address().port));
});
const free = async () => { const s = createServer(); const p = await listen(s); await new Promise<void>((r) => s.close(() => r())); return p; };

// The counter box's link to the platform: a local proxy the step files can cut.
let cloudDown = false;
const proxy = createServer((req, res) => {
  if (cloudDown) { req.socket.destroy(); return; }
  const up = httpsRequest(API + req.url, { method: req.method, headers: { ...req.headers, host: new URL(API).host } }, (r) => {
    res.writeHead(r.statusCode ?? 502, r.headers);
    r.pipe(res);
  });
  up.on('error', () => { if (!res.headersSent) res.writeHead(502); res.end(); });
  req.on('aborted', () => up.destroy());
  res.on('close', () => up.destroy());
  req.pipe(up);
});

// Console session of the staging administrator, through the launcher hand-off, as a person signs in.
S.consoleIn = async () => {
  if (S.admin) return S.admin;
  const ctx = await browser.newContext({ viewport: { width: 1440, height: 1000 } });
  ctx.setDefaultTimeout(30000);
  assert.equal((await ctx.request.post(LAUNCHER + '/api/auth/sign-in', { data: { phone: need('STAGING_ADMIN_PHONE'), password: need('STAGING_ADMIN_PASSWORD') } })).status(), 200);
  const hand = await (await ctx.request.post(LAUNCHER + '/api/auth/handoff', { data: { app: 'console' } })).json();
  const page = await ctx.newPage();
  await page.goto(hand.launchUrl);
  await page.waitForURL('**/overview', { timeout: 60000 });
  st.launcherSignOut = (await ctx.request.post(LAUNCHER + '/api/auth/sign-out', { data: {} })).status();
  save();
  S.admin = { ctx, page };
  return S.admin;
};
S.apiRaw = async (method: string, path: string, body?: any) => {
  const { ctx } = await S.consoleIn();
  const r = await ctx.request.fetch(CONSOLE + '/api' + path, { method, data: body, headers: method === 'GET' ? {} : { 'Idempotency-Key': randomUUID() } });
  const b = await r.json().catch(() => null);
  return { status: r.status(), body: b };
};
S.api = async (method: string, path: string, body?: any, okStatuses = [200, 201, 204]) => {
  const r = await S.apiRaw(method, path, body);
  assert.ok(okStatuses.includes(r.status), `${method} ${path.split('?')[0]}: ${r.status} ${r.body?.error?.code ?? ''} ${String(r.body?.error?.message ?? '').slice(0, 200)}`);
  return r.body;
};
S.newTill = async (label: string) => {
  if (S.tills[label]) return S.tills[label];
  const ctx = await browser.newContext({ viewport: { width: 1440, height: 1000 } });
  ctx.setDefaultTimeout(25000);
  // What a person answers "Allow" to when the browser asks to reach apps on this device (the box on the counter network).
  await ctx.grantPermissions(['local-network-access'], { origin: POS });
  const page = await ctx.newPage();
  const log: any[] = [];
  const short = (u: URL) => u.pathname.replace(/[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}/g, ':id');
  page.on('response', (res: any) => {
    const u = new URL(res.url());
    if (/\/(channel|scans)$/.test(u.pathname) || u.pathname.startsWith('/box/v1/')) {
      log.push({ at: new Date().toISOString(), method: res.request().method(), host: u.hostname.endsWith('localhost') ? 'counter-box' : u.hostname, path: short(u), status: res.status() });
      if (log.length > 600) log.shift();
    }
  });
  page.on('requestfailed', (req: any) => {
    const u = new URL(req.url());
    if (/\/(channel|scans)$/.test(u.pathname) || u.pathname.startsWith('/box/v1/')) {
      log.push({ at: new Date().toISOString(), method: req.method(), host: u.hostname.endsWith('localhost') ? 'counter-box' : u.hostname, path: short(u), failed: req.failure()?.errorText ?? 'failed' });
    }
  });
  const till: any = { ctx, page, log };
  till.activity = setInterval(() => { void page.keyboard.press('Shift').catch(() => {}); }, 15000);
  S.tills[label] = till;
  return till;
};

};

const server = createServer(async (req, res) => {
  let body = '';
  for await (const c of req) body += c;
  if (body === 'STOP') {
    res.end('stopping');
    for (const t of Object.values(S.tills) as any[]) clearInterval(t.activity);
    await browser.close().catch(() => {});
    if (S.runner) await S.runner.stop().catch(() => {});
    proxy.closeAllConnections();
    proxy.close();
    server.close();
    st.driverStoppedAt = new Date().toISOString();
    save();
    process.exit(0);
  }
  const file = body.trim();
  const started = Date.now();
  const name = file.split(/[\\/]/).pop();
  try {
    const mod = await import(pathToFileURL(file).href + '?v=' + Date.now());
    const result = await mod.default(S);
    st.steps[name!] = { at: new Date().toISOString(), ms: Date.now() - started, ok: true };
    save();
    res.writeHead(200, { 'content-type': 'application/json' });
    res.end(JSON.stringify({ ok: true, ms: Date.now() - started, result }, null, 1));
  } catch (e: any) {
    st.steps[name!] = { at: new Date().toISOString(), ms: Date.now() - started, ok: false, error: String(e?.message ?? e).slice(0, 600) };
    save();
    res.writeHead(500, { 'content-type': 'application/json' });
    res.end(JSON.stringify({ ok: false, error: String(e?.message ?? e).slice(0, 1500), stack: String(e?.stack ?? '').split('\n').slice(0, 6).join('\n') }));
  }
});
