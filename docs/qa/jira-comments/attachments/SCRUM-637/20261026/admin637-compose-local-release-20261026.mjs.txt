function archiveBytes(bytes, filename) {
  if (!filename.endsWith('.log')) return bytes;
  need(Buffer.from(bytes.toString('utf8'), 'utf8').equals(bytes), 'DISPLAY_LOG_UTF8_REQUIRED');
  return Buffer.from(bytes.toString('utf8').replace(/[ \t]+(?=\r?$)/gm, ''), 'utf8');
}
function sourcePath(p) {
  if (!p || !/^[A-Za-z0-9_./-]+$/.test(p) || p.split('/').includes('..')) return false;
  if (/^(?:docs|imports|services)\//.test(p) || /(?:^|\/)(?:node_modules|dist|\.git)(?:\/|$)/.test(p)) return false;
  if (p === '.env.example') return true;
  if (/(?:^|\/)\.env(?:\.|\/|$)/.test(p)) return false;
  return path.posix.basename(p) === '.gitignore' || /^(?:apps|packages|scripts)\//.test(p) || /^\.github\/workflows\//.test(p)
    || /^(?:package\.json|pnpm-lock\.yaml|pnpm-workspace\.yaml|turbo\.json|eslint\.config\.mjs|tsconfig[^/]*\.json|render\.yaml|\.npmrc|\.pnpmfile\.cjs)$/.test(p);
}
function hostSource() {
  const result = spawnSync('git', ['ls-files', '--cached', '--others', '--exclude-standard', '-z'], { cwd: ROOT, encoding: 'utf8', maxBuffer: 32 * 1024 * 1024 });
  need(result.status === 0 && result.signal === null, 'HOST_SOURCE_CENSUS_FAILED');
  return [...new Set(result.stdout.split('\0').filter(sourcePath))].sort().map(p => {
    const file = path.join(ROOT, p);
    need(fs.lstatSync(file).isFile() && fs.realpathSync(file).startsWith(fs.realpathSync(ROOT) + path.sep), 'HOST_SOURCE_BOUNDARY');
    return { path: p, sha256: sha(file) };
  });
}
function noSecrets(text) {
  need(!/postgres(?:ql)?:\/\/[^\s"'<>]+/i.test(text)
    && !/\beyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\b/.test(text)
    && !/\bBearer\s+[A-Za-z0-9._~-]{12,}/i.test(text), 'UNSAFE_ARTIFACT_REFUSED');
}
function counts(text) {
  const value = { passed: 0, failed: 0, skipped: 0, todo: 0 };
  for (const match of text.matchAll(/([\d,]+)\s+(passed|failed|skipped|todo)/g)) value[match[2]] += Number(match[1].replaceAll(',', ''));
  const declared = /\(([\d,]+)\)\s*$/.exec(text);
  return { ...value, total: declared ? Number(declared[1].replaceAll(',', '')) : Object.values(value).reduce((a, b) => a + b, 0) };
}
function statsOf(log) {
  const clean = log.replace(/\u001b\[[0-?]*[ -/]*[@-~]/g, ''), vitest = [], tap = [], browser = [];
  let files = null, block = null;
  for (const line of clean.split(/\r?\n/)) {
    const f = /\bTest Files\s+(.+)$/.exec(line); if (f) files = counts(f[1]);
    const t = /\bTests\s+(.+)$/.exec(line); if (t && /\d+\s+(?:passed|failed|skipped|todo)/.test(t[1])) { vitest.push({ files, tests: counts(t[1]) }); files = null; }
    const n = /(?:^|:\s*)# (tests|pass|fail|cancelled|skipped|todo) (\d+)\s*$/.exec(line);
    if (n) { if (n[1] === 'tests') { if (block) tap.push(block); block = {}; } if (block) block[n[1]] = Number(n[2]); }
    const b = /^\s*(\d+)\s+(passed|failed|skipped|flaky)(?:\s+\([^)]*\))?\s*$/.exec(line);
    if (b) browser.push({ [b[2]]: Number(b[1]) });
  }
  if (block) tap.push(block);
  const tasks = [...clean.matchAll(/Tasks:\s+(\d+) successful,\s+(\d+) total/g)].at(-1);
  const typed = [...new Set([...clean.matchAll(/^([^\s:]+):typecheck:/gm)].map(m => m[1]))];
  const built = [...new Set([...clean.matchAll(/^([^\s:]+):build:/gm)].map(m => m[1]))];
  return { vitest, tap, browser, turbo: tasks ? { successful: Number(tasks[1]), total: Number(tasks[2]), typecheckPackages: typed, buildPackages: built } : null,
    typeScriptDiagnosticHeaders: [...clean.matchAll(/\berror TS\d+:/g)].length };
}

export function compose({ explicitCompose = false } = {}) {
  need(explicitCompose === true, 'EXPLICIT_COMPOSE_REQUIRED');
  const prepared = prepareComposition(), directory = TARGET + '/' + SUB, top = TARGET + '/local-release.json';
  need(!fs.existsSync(directory) && !fs.existsSync(top), 'PRESERVE_EXISTING_EVIDENCE');
  need(fs.realpathSync(TARGET).startsWith(fs.realpathSync(ROOT) + path.sep), 'REPOSITORY_EVIDENCE_BOUNDARY');
  need(equal(hostSource(), prepared.expectedRows), 'HOST_CHANGED_BEFORE_COPY');
  fs.mkdirSync(directory);
  for (const item of prepared.files) {
    const original = fs.readFileSync(item.from);
    need(hash(original) === item.originalSourceSha256, 'ARTIFACT_CHANGED_BEFORE_COPY');
    const expected = archiveBytes(original, item.to), target = directory + '/' + item.to;
    need(hash(expected) === item.copiedArchiveSha256, 'DISPLAY_ARCHIVE_DERIVATION_CHANGED');
    if (item.formattingNormalization === 'none') fs.copyFileSync(item.from, target, fs.constants.COPYFILE_EXCL);
    else fs.writeFileSync(target, expected, { flag: 'wx' });
    need(sha(target) === item.copiedArchiveSha256 && sha(item.from) === item.originalSourceSha256, 'ARTIFACT_COPY_HASH_MISMATCH');
  }
  need(equal(hostSource(), prepared.expectedRows), 'HOST_CHANGED_AFTER_COPY');
  fs.writeFileSync(directory + '/file-list.json', JSON.stringify(prepared.summary.artifacts, null, 2) + '\n', { flag: 'wx' });
  fs.writeFileSync(top, JSON.stringify(prepared.summary, null, 2) + '\n', { flag: 'wx' });
  return { ticket: 'SCRUM-637', localReleaseGreen: true, sourceManifestSha256: MANIFEST, copiedFiles: prepared.files.length,
    summaryPath: posix(top), summarySha256: sha(top) };
}
export function main(argv = process.argv.slice(2)) {
  need(argv.length === 1 && ['--compose', '--check'].includes(argv[0]), 'EXPLICIT_COMPOSE_OR_CHECK_REQUIRED');
  let result;
  if (argv[0] === '--check') {
    const prepared = prepareComposition();
    result = { ticket: 'SCRUM-637', validationPassed: true, sourceManifestSha256: MANIFEST,
      gateCount: prepared.summary.gateCount, plannedArtifacts: prepared.files.length, repositoryWrites: false };
  } else result = compose({ explicitCompose: true });
  process.stdout.write(JSON.stringify(result) + '\n'); return result;
}
