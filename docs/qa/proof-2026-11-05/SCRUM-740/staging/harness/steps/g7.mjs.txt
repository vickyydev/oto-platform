export default async function (S) {
  const { text, sleep } = S.lib;
  const page = S.pages.T740.page;
  const btn = page.getByRole('button', { name: 'Continue to payment' });
  const enabled = await btn.isEnabled();
  await S.shot('T740', 'SCRUM-740/staging/st-01a-two-children-registered-for-drop-off');
  if (!enabled) return { enabled };
  await btn.click();
  await sleep(4000);
  await S.look('T740');
  return { enabled, text: (await text(page, 4000)).slice(300, 3500) };
}
