/** Shared by direct HTTP and browser dispatch; no business write, handoff or aggregate fan-out. */
export function requestAllowed({ url, method, body, branchIds = [], auth = false, health = false, assetPaths = [], document = false }) {
  let u; try { u = new URL(url); } catch { return false; }
  if (u.origin !== ORIGIN || u.username || u.password || u.hash || u.href !== url || u.href !== u.origin + u.pathname + u.search
    || u.href.includes('%2f') || u.href.includes('%5c')) return false;
  const p = u.pathname, q = u.searchParams;
  if (method === 'POST') return auth && !u.search && (p === '/api/auth/sign-in'
    ? exact(body, ['phone', 'password']) && ['phone', 'password'].every(k => typeof body[k] === 'string' && body[k].length > 0 && body[k].length < 200)
    : p === '/api/auth/sign-out' && body === undefined);
  if (method !== 'GET' || body !== undefined) return false;
  if (['/api/me', '/api/me/permissions', '/api/ready'].includes(p)) return !u.search;
  if (p === '/api/overview') return !u.search || q.size === 1 && q.has('branchId') && uuid(q.get('branchId')) && branchIds.includes(q.get('branchId'));
  if (p === '/api/ops/health') return health && !u.search;
  if (p === '/api/ops/failures') return health && q.size === 2 && q.getAll('windowHours').length === 1 && q.get('windowHours') === '24' && q.getAll('limit').length === 1 && q.get('limit') === '50';
  return !u.search && ((p === '/overview' && document) || publicPath(p) && assetPaths.includes(p));
}

export function validateOverview(body, { availableIds, selectedIds, operatorWide, after, observedAt }) {

  need(Number.isFinite(generated) && generated > Date.parse(after) && generated <= observed + 5_000 && generated >= observed - 30_000, 'FRESH_POSTDEPLOY_OVERVIEW_REQUIRED');

  need(body.parks.every(p => body.branches.find(b => b.id === p.branchId)?.businessDate === p.businessDate), 'NATIVE_BUSINESS_DATES_REQUIRED');
  const day = dayOf(new Date(body.generatedAt)); let metricBuckets = 0;
  for (const key of GROUPS) {
    const g = body.aggregates[key];
    need(g.generatedAt === body.generatedAt && eqIds(g.branches.map(p => p.id), availableIds) && same(g.branches, body.branches)
      && eqIds(g.selectedBranchIds, selectedIds) && eqIds(g.parks.map(p => p.branchId), selectedIds), 'EXACT_GROUP_ENVELOPES_REQUIRED');
    if (key === 'fleet') need(g.asOf === body.generatedAt && g.basis === 'current-snapshot', 'FLEET_CURRENT_ASOF_REQUIRED');
    else {
      need(g.period.from === day && g.period.to === day && g.parks.every(p => p.rows.every(r => r.date === day)), 'SERVER_BANGKOK_CALENDAR_TODAY_REQUIRED');
      if (key.endsWith('Usage')) {
        need(operatorWide ? g.operatorWide !== null : g.operatorWide === null, 'OPERATOR_BUCKET_REACH_REQUIRED');
        need(g.operatorWide === null || g.operatorWide.rows.every(r => r.date === day), 'OPERATOR_BUCKET_DATE_REQUIRED');
        metricBuckets += currentAi(g);
        for (const metric of ['count', 'inputTokens', 'outputTokens', 'cacheReadTokens', 'totalTokens', 'unpricedTokens'])
          need(g.totals[metric] === g.parks.reduce((n, p) => n + p.totals[metric], 0) + (g.operatorWide?.totals[metric] ?? 0), 'AI_SCOPED_SUM_REQUIRED');
      }
    }
  }
  need(body.decisions.generatedAt === body.generatedAt && eqIds(body.decisions.parks.map(p => p.branchId), selectedIds), 'DECISION_DATE_REACH_REQUIRED');
  for (const items of [body.decisions.items, ...body.decisions.parks.map(p => p.items)]) {

    for (const key of ['erasureRequests', 'approvalRequests']) need(items.find(i => i.key === key)?.status === 'not-built', 'UNBUILT_QUEUE_REQUIRED');
    const stranded = items.find(i => i.key === 'strandedOccupants');
    need(stranded.label === 'End-of-day review rows still inside' && stranded.basis === 'current-business-date-native-end-of-day-rows', 'NATIVE_END_OF_DAY_MEANING_REQUIRED');
  }
  need(typeof body.statusSummary === 'string' && body.statusSummary.startsWith('Admin status summary\n') && body.statusSummary.includes(`Checked: ${body.generatedAt}`)
    && body.statusSummary.includes('Price unavailable: no_rate_configured (USD estimate)') && body.statusSummary.includes('unpriced tokens; excludes unpersisted cache writes.'), 'CURRENT_SERVER_COPY_TEXT_REQUIRED');
  return { generatedAt: body.generatedAt, calendarDate: day, availableParkCount: availableIds.length, selectedParkCount: selectedIds.length,
    selectedIdsSha256: digest(sorted(selectedIds)), totals: body.totals, groupTotals: Object.fromEntries(GROUPS.map(k => [k, body.aggregates[k].totals])),
    decisions: body.decisions.items.map(({ key, status, count, basis }) => ({ key, status, count, basis })), metricBuckets, statusSummarySha256: hash(body.statusSummary) };
}

export function verifyLocalSource(plan, { io = fs, git, root = WORKTREE, nodeVersion = process.version } = {}) {

  };
  const paths = git(['ls-files', '--cached', '--others', '--exclude-standard', '-z']).split('\0').filter(Boolean).filter(selected).sort();
  const tree = git(['ls-tree', '-r', '-z', '--full-tree', plan.sourceSha]).split('\0').filter(Boolean).map(line => {
    const m = /^(\d{6}) (blob|commit) ([a-f0-9]{40})\t(.+)$/.exec(line); need(m, 'FEATURE_TREE_ENTRY_REQUIRED');
    return { mode: m[1], type: m[2], blobId: m[3], path: m[4] };
  }).filter(r => selected(r.path)).sort((a, b) => a.path < b.path ? -1 : a.path > b.path ? 1 : 0);
  need(same(paths, tree.map(r => r.path)), 'FEATURE_SOURCE_PATHS_DIFFER');
  const rows = paths.map((p, i) => {

      && io.lstatSync(path.join(root, p)).isFile(), 'REGULAR_SOURCE_REQUIRED');
    const b = Buffer.from(io.readFileSync(path.join(root, p)));
    need(createHash('sha1').update(Buffer.from(`blob ${b.length}\0`)).update(b).digest('hex') === tree[i].blobId, 'FEATURE_SOURCE_BLOB_DIFFERS');
    return { path: p, sha256: hash(b) };

    '.env.example', '.gitignore', ':(glob)**/.gitignore'])
    .split('\0').filter(Boolean).filter(selected).length === 0, 'CURRENT_FEATURE_RUNTIME_DIFFERS');
  const normalized = b => b.toString().replaceAll('\r\n', '\n').trim();
  need(normalized(io.readFileSync(path.join(root, 'pnpm-lock.yaml'))) === normalized(io.readFileSync(path.join(root, 'node_modules/.pnpm/lock.yaml'))), 'INSTALLED_LOCK_DIFFERS');
  need(runtimeBindingsSha256Of(root, io, nodeVersion) === plan.runtimeBindingsSha256, 'INSTALLED_RUNTIME_BINDING_DIFFERS');

}

/** Always probe the original cookie independently, even after a logout timeout. No uncertainty is cleared here. */
export async function closeSession(context, request, record) {
  const original = context.originalCookie || context.cookie, row = { actor: context.actor, signOutStatus: null, meStatus: null,
    noOriginalCookie: typeof original !== 'string' || original.length === 0, confirmed: false };
  context.originalCookie = original; context.cookie = original;
  let logout = false;
  try { const r = await request(context, 'POST', '/api/auth/sign-out', { cleanup: true }); row.signOutStatus = r.status; logout = r.status === 200 && r.body?.ok === true; }
  catch (e) { row.errorCode = codeOf(e); }
  try { context.cookie = original; const r = await request(context, 'GET', '/api/me', { cleanup: true }); row.meStatus = r.status; }
  catch (e) { row.errorCode ??= codeOf(e); }
  finally { context.cookie = ''; row.confirmed = !row.noOriginalCookie && logout && row.meStatus === 401 && !row.errorCode; context.closed = row.confirmed;
    if (row.confirmed) context.originalCookie = ''; record(row); }
  return row.confirmed;
}
async function boundedJson(response) {
  need(response.body && Number(response.headers.get('content-length') ?? 0) <= MAX_BODY, 'JSON_BODY_LIMIT');
  const reader = response.body.getReader(), chunks = []; let size = 0;
  try { for (;;) { const r = await reader.read(); if (r.done) break; size += r.value.byteLength;
    need(size <= MAX_BODY, 'JSON_BODY_LIMIT'); chunks.push(Buffer.from(r.value)); }
    try { return JSON.parse(Buffer.concat(chunks).toString('utf8')); } catch { throw new ProofError('JSON_BODY_REQUIRED'); }
  } finally { await reader.cancel().catch(() => {}); reader.releaseLock(); }
}
