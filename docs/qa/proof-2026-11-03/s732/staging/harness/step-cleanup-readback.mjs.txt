  const { st, save, adminApi, admin } = S;
  const f = st.fixture;
  const c = st.cleanup;
  const rb = (c.readback = { at: new Date().toISOString() });
  const stations = (await adminApi('GET', `/branches/${f.branchId}/stations?includeArchived=true`)).body?.stations ?? [];
  const mine = stations.find((s) => s.id === f.stationId);
  rb.station = { archivedAt: mine?.archivedAt ?? null };
  const devices = (await adminApi('GET', `/boxes/${f.boxId}/devices?includeArchived=true`)).body?.devices ?? [];
  rb.terminal = { archivedAt: devices.find((d) => d.id === f.terminalId)?.archivedAt ?? null };

  rb.ownEventsStillListed = JSON.stringify(events.body ?? {}).split('ZZ TEST SCRUM-732 ' + f.stamp).length - 1;
  const spec = { member: f.memberId, children: f.children.map((k) => k.id), station: f.stationId, terminal: f.terminalId, box: f.boxId, since: st.steps.passQr.startedAt, ids: Object.values(f.events).map((e) => e.id) };
  const program = `(async()=>{const {Client}=require('pg');const s=${JSON.stringify(spec)},c=new Client({connectionString:process.env.DATABASE_URL});await c.connect();try{await c.query('BEGIN READ
