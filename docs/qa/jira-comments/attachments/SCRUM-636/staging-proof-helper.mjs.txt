/** Pure and deliberately refuses pending planner choices. Schemas are imported from released source, never copied here. */
export function validatePlan(plan) {
  need(exact(plan, ['approved', 'ticket', 'purpose', 'sourceSha', 'sourceManifestSha256', 'ciRunId', 'apiDeployment',
    'helperSha256', 'routeModuleSha256', 'localReleaseSha256', 'runtimeBindingsSha256', 'actors', 'period', 'rulings', 'outputDirectory']), 'EXACT_REVIEWED_PLAN_REQUIRED');
  need(plan.approved === true && plan.ticket === 'SCRUM-636' && plan.purpose === 'read-only-aggregate-api-proof', 'REVIEWED_636_PLAN_REQUIRED');
  need(/^[a-f0-9]{40}$/.test(plan.sourceSha ?? '') && plan.sourceSha !== '0'.repeat(40) && /^\d+$/.test(String(plan.ciRunId))
    && [plan.sourceManifestSha256, plan.helperSha256, plan.routeModuleSha256, plan.localReleaseSha256, plan.runtimeBindingsSha256].every(hex), 'FUTURE_FEATURE_BINDINGS_REQUIRED');
  const dep = plan.apiDeployment;
  need(exact(dep, ['name', 'serviceId', 'id', 'finishedAt']) && dep.name === 'oto-api-staging' && /^srv-[A-Za-z0-9]+$/.test(dep.serviceId ?? '')
    && /^dep-[A-Za-z0-9]+$/.test(dep.id ?? '') && iso(dep.finishedAt), 'EXACT_LIVE_API_DEPLOYMENT_REQUIRED');
  need(exact(plan.actors, ACTORS) && ACTORS.every(actor => exact(plan.actors[actor], ['accountId', 'operatorId'])
    && plan.actors[actor].accountId === DEMOS[actor] && plan.actors[actor].operatorId === DEMOS.operatorId), 'THREE_EXACT_EXISTING_DEMOS_REQUIRED');
  need(exact(plan.period, ['from', 'to']) && realDay(plan.period.from) && realDay(plan.period.to), 'EXPLICIT_REAL_DATES_REQUIRED');
  const days = (Date.parse(plan.period.to) - Date.parse(plan.period.from)) / 86_400_000 + 1;
  need(days >= 1 && days <= 31, 'BOUNDED_ORDERED_PERIOD_REQUIRED');
  need(exact(plan.rulings, ['Q17', 'Q18']) && ['Q17', 'Q18'].every(key => exact(plan.rulings[key], ['resolved', 'reference'])
    && plan.rulings[key].resolved === true && typeof plan.rulings[key].reference === 'string'
    && /^[A-Za-z0-9 .:/#_-]{8,240}$/.test(plan.rulings[key].reference)), 'Q17_Q18_APPROVED_CONTRACT_REQUIRED');
  need(/^C:\/w\/admin636-staging-[A-Za-z0-9_-]{1,60}$/.test(plan.outputDirectory ?? ''), 'EXCLUSIVE_EXTERNAL_ATTEMPT_REQUIRED');
  return plan;
}

/** Pure outbound grammar. Invalid query values are allowed only in the explicitly armed negative-query cases. */
export function validateRequest({ method, pathname, query = '', body, negative = false }) {
  if (method === 'POST') return !query && (pathname === '/auth/sign-in'
    ? exact(body, ['phone', 'password']) && typeof body.phone === 'string' && typeof body.password === 'string' && body.phone.length > 0 && body.password.length > 0
    : pathname === '/auth/sign-out' && body === undefined);
  if (method !== 'GET' || body !== undefined) return false;
  if (['/me', '/me/permissions'].includes(pathname)) return !query;
  if (!PATHS.includes(pathname) || query.length > 400 || /[\r\n#]/.test(query)) return false;
  const params = new URLSearchParams(query);
  if (negative) return [...params.keys()].every(key => ['from', 'to', 'branchId', 'operatorId'].includes(key)) && params.size <= 5;
  const keys = [...params.keys()];
  if (new Set(keys).size !== keys.length || keys.some(key => !['from', 'to', 'branchId'].includes(key))) return false;
  if (params.has('branchId') && !uuid(params.get('branchId'))) return false;
  if (pathname === '/aggregates/fleet') return keys.every(key => key === 'branchId');
  if (!realDay(params.get('from')) || !realDay(params.get('to'))) return false;
  const days = (Date.parse(params.get('to')) - Date.parse(params.get('from'))) / 86_400_000 + 1;
  return days >= 1 && days <= 31;
}

export function overviewReach(permissions, operatorId) {
  need(Array.isArray(permissions), 'FRESH_PERMISSIONS_REQUIRED');
  const grants = permissions.filter(row => row?.permission === 'console:overview:read');
  return grants.some(row => row.scopeType === 'operator' && (row.scopeId === null || row.scopeId === operatorId))
    ? { kind: 'operator', branchIds: [] }
    : { kind: 'branches', branchIds: [...new Set(grants.filter(row => row.scopeType === 'branch' && uuid(row.scopeId)).map(row => row.scopeId))].sort() };
}

/** SCRUM-636 has no configured rates. A future schema-valid price is not proof of this release's contract. */
export function validateCurrentAiSpend(body) {
  need(body?.currency === 'USD' && body.excludesCacheWrites === true, 'CURRENT_RELEASE_USD_SCOPE_REQUIRED');
  need(Array.isArray(body.parks) && body.parks.every(park => Array.isArray(park.rows))
    && (body.operatorWide === null || Array.isArray(body.operatorWide?.rows)), 'CURRENT_RELEASE_METRIC_BUCKETS_REQUIRED');
  const metrics = [body.totals, ...body.parks.flatMap(park => [park.totals, ...park.rows]),
    ...(body.operatorWide === null ? [] : [body.operatorWide.totals, ...body.operatorWide.rows])];
  for (const metric of metrics) {
    need(metric && metric.spendEstimate === null && metric.reason === 'no_rate_configured', 'CURRENT_RELEASE_UNAVAILABLE_SPEND_REQUIRED');
    const tokens = [metric.inputTokens, metric.outputTokens, metric.cacheReadTokens];
    need(tokens.every(value => Number.isSafeInteger(value) && value >= 0), 'CURRENT_RELEASE_NATIVE_TOKENS_REQUIRED');
    const total = tokens.reduce((sum, value) => sum + value, 0);
    need(Number.isSafeInteger(total) && metric.totalTokens === total && metric.unpricedTokens === total, 'CURRENT_RELEASE_UNPRICED_REMAINDER_REQUIRED');
  }
  return { currency: body.currency, excludesCacheWrites: body.excludesCacheWrites, priceReason: 'no_rate_configured', metricBuckets: metrics.length };
}

async function boundedJson(response) {
  need(response.body && Number(response.headers.get('content-length') ?? 0) <= MAX_BODY, 'RESPONSE_BODY_LIMIT');
  const reader = response.body.getReader(), chunks = []; let bytes = 0;
  try {
    for (;;) { const value = await reader.read(); if (value.done) break; bytes += value.value.byteLength;
      need(bytes <= MAX_BODY, 'RESPONSE_BODY_LIMIT'); chunks.push(Buffer.from(value.value)); }
    try { return JSON.parse(Buffer.concat(chunks).toString('utf8')); } catch { throw new ProofError('RESPONSE_JSON_REQUIRED'); }
  } finally { await reader.cancel().catch(() => {}); reader.releaseLock(); }
}

      need(iso(body.generatedAt) && Date.parse(body.generatedAt) > Date.parse(plan.apiDeployment.finishedAt)
        && Date.parse(body.generatedAt) >= Date.parse(receipt.startedAt) - 5_000 && Date.parse(body.generatedAt) <= Date.now() + 5_000, 'FRESH_POSTDEPLOY_RESULT_REQUIRED');
      if (pathname !== '/aggregates/fleet') need(same(body.period, plan.period), 'EXACT_PERIOD_REQUIRED');
      else need(body.asOf === body.generatedAt, 'CURRENT_FLEET_ASOF_REQUIRED');
      if (pathname.includes('usage')) need(operatorWide ? body.operatorWide !== null && typeof body.operatorWide === 'object'
        : body.operatorWide === null, 'OPERATOR_BUCKET_AUTHORITY_MISMATCH');
      const spend = pathname.includes('usage') ? validateCurrentAiSpend(body) : null;
      Object.assign(observed, { generatedAt: body.generatedAt, period: body.period ?? null, basis: body.basis, selectedParkCount: selected.length,
        availableParkCount: available.length, rowCount: body.parks.reduce((sum, park) => sum + (park.rows?.length ?? 0), 0),
        safeTotals: body.totals, selectedBranchIdsSha256: digest([...selected].sort()), operatorWideVisible: body.operatorWide != null, ...(spend ?? {}) });
      receipt.reads.push(observed); save(); return body;
    }
    need(result.body?.error && typeof result.body.error.code === 'string', 'STRUCTURED_REFUSAL_REQUIRED');
    receipt.reads.push(observed); save(); return null;
  }
  async function closeContext(context) {
    const row = { actor: context.actor, signOutStatus: null, meStatus: null, confirmed: false };
    const originalCookie = context.cookie;
    let logoutOk = false;
    try { const out = await request(context, 'POST', '/auth/sign-out', { cleanup: true }); row.signOutStatus = out.status;
      logoutOk = out.status === 200 && out.body?.ok === true; }
    catch (error) { row.errorCode = codeOf(error); }
    // Independent best effort even after logout timeout/body failure. This never clears unresolvedAuth.
    try { context.cookie = originalCookie;
      const me = await request(context, 'GET', '/me', { cleanup: true }); row.meStatus = me.status; }
    catch (error) { row.errorCode ??= codeOf(error); }
    finally { row.confirmed = logoutOk && row.meStatus === 401 && !row.errorCode;
      context.cookie = ''; context.cleanupConfirmed = row.confirmed; receipt.authCleanup.push(row); try { save(); } catch {} }
    return row.confirmed;
  }

  try {
    need(Date.parse(receipt.startedAt) > Date.parse(plan.apiDeployment.finishedAt), 'POSTDEPLOY_START_REQUIRED');
    await sourceGate();
    const state = {};
    for (const actor of ACTORS) {
      const context = newContext(actor), signed = await request(context, 'POST', '/auth/sign-in', {
        body: { phone: secret(`STAGING_${actor}_PHONE`), password: secret(`STAGING_${actor}_PASSWORD`) } });
      need(signed.status === 200 && signed.body?.accountId === plan.actors[actor].accountId && signed.body.mustChangePassword === false && context.cookie, 'EXPECTED_DEMO_SIGN_IN_REQUIRED');
      const snapshot = await census(context); context.baseline = snapshot.hash; context.permissions = snapshot.reach;
      state[actor] = context; receipt.authority.push({ actor, beforeSha256: snapshot.hash, afterSha256: null, reachKind: snapshot.reach.kind }); save();
    }
    check('ADMIN_GENUINE_OPERATOR_REACH', state.ADMIN.permissions.kind === 'operator');
    check('MANAGER_GENUINE_BRANCH_REACH', state.MANAGER.permissions.kind === 'branches' && state.MANAGER.permissions.branchIds.length > 0);
    check('RECEPTION_HAS_NO_OVERVIEW_REACH', state.RECEPTION.permissions.kind === 'branches' && state.RECEPTION.permissions.branchIds.length === 0);
    await sourceGate();
    let ownedIds;
    for (const pathname of PATHS) {
      const body = await aggregate(state.ADMIN, pathname, { expectedIds: ownedIds, operatorWide: true });
      ownedIds ??= body.branches.map(row => row.id);
      need(ownedIds.length > 0, 'LIVE_OWNED_PICKER_REQUIRED');
    }
    const managerIds = ownedIds.filter(id => state.MANAGER.permissions.branchIds.includes(id));
    check('MANAGER_HAS_LIVE_OWNED_REACH', managerIds.length > 0);
    for (const pathname of PATHS) {
      await aggregate(state.MANAGER, pathname, { expectedIds: managerIds, operatorWide: false });
      await aggregate(state.MANAGER, pathname, { branchId: managerIds[0], expectedIds: [managerIds[0]], availableIds: managerIds, operatorWide: false });
      await aggregate(state.RECEPTION, pathname, { expectedStatus: 403 });
    }
    for (const pathname of PATHS.filter(value => value.includes('usage'))) await aggregate(state.ADMIN, pathname,
      { branchId: managerIds[0], expectedIds: [managerIds[0]], availableIds: ownedIds, operatorWide: false });
    const outside = ownedIds.find(id => !managerIds.includes(id));
    if (outside) for (const pathname of PATHS) await aggregate(state.MANAGER, pathname, { branchId: outside, expectedStatus: 403 });
    receipt.checks.push({ name: 'OWNED_UNREACHABLE_PARK', passed: outside ? true : null, available: Boolean(outside),
      outcome: outside ? 'passed' : 'not-available-with-existing-grants', at: stamp() });
    const absentId = randomUUID(), anonymous = { actor: 'ANONYMOUS', cookie: '' };
    for (const pathname of PATHS) { await aggregate(anonymous, pathname, { expectedStatus: 401 });
      await aggregate(state.ADMIN, pathname, { branchId: absentId, expectedStatus: 404 }); }
    for (const query of ['from=2026-02-30&to=2026-03-01', 'from=2026-10-02&to=2026-10-01',
      'from=2026-10-01&to=2026-11-01', `from=${plan.period.from}&to=${plan.period.to}&branchId=${managerIds[0]}&branchId=${managerIds[0]}`,
      `from=${plan.period.from}&to=${plan.period.to}&operatorId=${plan.actors.ADMIN.operatorId}`])
      await aggregate(state.ADMIN, '/aggregates/members', { query, negative: true, expectedStatus: 400 });
    await aggregate(state.ADMIN, '/aggregates/fleet', { query: `from=${plan.period.from}&to=${plan.period.to}`, negative: true, expectedStatus: 400 });
    for (const context of sessions) { const after = await census(context); check('AUTHORITY_UNCHANGED_' + context.actor, after.hash === context.baseline);
      receipt.authority.find(row => row.actor === context.actor).afterSha256 = after.hash; }
  } catch (error) { failed = true; receipt.errorCode = codeOf(error); }
  finally {
    // Always attempt every independent context, including an uncertain sign-in; never erase the original uncertainty.
    for (const context of sessions) if (context.attempted && !(await closeContext(context))) failed = true;
    receipt.completedAt = stamp(); try { save(); } catch { failed = true; }
  }
  try {
    safe(); need(!failed && sessions.length === 3 && sessions.every(row => row.cleanupConfirmed), 'ALL_AUTH_CLEANUP_REQUIRED');
    await sourceGate();
    const { launch, card } = await import(pathToFileURL(ROOT + '/scripts/agent/evidence.mjs').href);
    browser = await launch(); const page = await browser.newPage(); let pageRequests = 0, pageErrors = 0, consoleErrors = 0;
    page.on('pageerror', () => { pageErrors += 1; }); page.on('console', message => { if (message.type() === 'error') consoleErrors += 1; });
    await page.route('**/*', route => { pageRequests += 1; return route.abort(); }); await page.goto('about:blank');
    const positive = PATHS.map(pathname => receipt.reads.find(row => row.actor === 'ADMIN' && row.path === pathname && row.status === 200 && !row.negative));
    need(positive.every(Boolean), 'FIVE_ACTUAL_RESULTS_REQUIRED');
    const captureStartedAt = stamp(), filename = path.join(plan.outputDirectory, PNG);
    await card(page, { title: 'SCRUM-636 — actual staging API test run', subtitle: 'API evidence; Overview UI follows in SCRUM-637',
      identity: 'Existing seeded ADMIN / MANAGER / RECEPTION; personal details omitted',
      deploy: `Source ${plan.sourceSha} | CI ${plan.ciRunId} | API ${plan.apiDeployment.id} live ${plan.apiDeployment.finishedAt}`,
      rows: [...positive.map(row => ({ ok: true, tag: '200', head: 'GET ' + row.path,
        note: `${row.selectedParkCount} parks; ${row.rowCount} daily rows; generated ${row.generatedAt}`, body: JSON.stringify(row.safeTotals) })),
      { ok: true, tag: 'LIMIT', head: 'AI price unavailable in USD', note: 'no_rate_configured; estimate excludes cache writes.',
        body: positive.filter(row => row.path.includes('usage')).map(row => `${row.path}: spendEstimate=null; ${row.safeTotals.unpricedTokens} unpriced tokens.`).join(' | ') },
      { ok: true, tag: 'PASS', head: 'Current scopes and refusals', note: 'Fresh grants; Manager park scope; Reception403; anonymous401; bounded queries; absent park404.' },
      { ok: true, tag: 'PASS', head: 'Cleanup and source', note: 'Three logout200/current-user401; branch/preferences/grants unchanged; no business writes or fixtures.' },
      { ok: true, tag: 'LIMIT', head: 'API test-run evidence', note: `${plan.period.from} through ${plan.period.to}; no native Overview screen, private detail lists or payment reads.` }], images: [] }, filename);
    need(pageRequests === 0 && pageErrors === 0 && consoleErrors === 0, 'EVIDENCE_PAGE_NETWORK_OR_ERROR');
    receipt.screenshots.push({ filename: PNG, sha256: hash(fs.readFileSync(filename)), captureStartedAt, capturedAt: stamp(), pageRequests, pageErrors, consoleErrors });
    await browser.close(); browser = null; await sourceGate();
    receipt.completedAt = stamp(); receipt.complete = true; save();
  } catch (error) { receipt.complete = false; receipt.errorCode ??= codeOf(error); }
  finally { if (browser) try { await browser.close(); } catch { receipt.complete = false; receipt.errorCode ??= 'EVIDENCE_BROWSER_CLOSE_FAILED'; }
    try { save(); } catch { receipt.complete = false; receipt.errorCode ??= 'RECEIPT_WRITE_FAILED'; } }
  return receipt;
}
