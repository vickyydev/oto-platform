import assert from 'node:assert/strict';
import {readFileSync,writeFileSync} from 'node:fs';
import {job} from './qa-render-job.mjs';
const out=process.argv[2],r=JSON.parse(readFileSync(out+'/receipt.json','utf8')),f=r.fixtures;
assert.ok(r.name.startsWith('ZZ TEST older proof ')&&f.walletId);
const spec={branchId:f.walletBranch,walletId:f.walletId,saleId:f.walletSale,actorId:r.identity.accountId,name:r.name,amount:r.wallet.amountSatang};
const program=`(async()=>{
 require('tsx/cjs');const assert=require('node:assert/strict'),{Pool}=require('pg'),{drizzle}=require('drizzle-orm/node-postgres'),{expireWalletsForDay}=require('./src/services/wallet.ts'),{audit}=require('./src/services/audit.ts');
 const s=${JSON.stringify(spec)},pool=new Pool({connectionString:process.env.DATABASE_URL,max:2}),db=drizzle(pool),c=await pool.connect();
 try {
  await c.query('BEGIN');await c.query("SET LOCAL statement_timeout='15s'");
  const br=(await c.query('SELECT * FROM core.branch WHERE id=$1 FOR UPDATE',[s.branchId])).rows[0];assert.equal(br.name,s.name+' wallet branch');assert.equal(br.archived_at,null);
  const ws=(await c.query('SELECT id,holder_name,balance_satang,status FROM pos.wallet WHERE branch_id=$1 FOR UPDATE',[s.branchId])).rows;assert.equal(ws.length,1);assert.equal(ws[0].id,s.walletId);assert.equal(Number(ws[0].balance_satang),s.amount);assert.equal(ws[0].status,'active');
  const p=(await c.query("SELECT method,status,amount_satang FROM pos.payment_attempt WHERE sale_id=$1",[s.saleId])).rows;assert.equal(p.length,1);assert.equal(p[0].method,'cash');assert.equal(p[0].status,'approved');
  const es=(await c.query('SELECT id,kind,source,sale_id,amount_satang,expires_at,business_date FROM pos.wallet_entry WHERE wallet_id=$1 FOR UPDATE',[s.walletId])).rows;assert.equal(es.length,1);assert.equal(es[0].kind,'grant');assert.equal(es[0].sale_id,s.saleId);assert.equal(Number(es[0].amount_satang),s.amount);
  const day=new Date(Date.now()-2*86400000).toISOString().slice(0,10),expiresAt=new Date(day+'T00:00:00Z');
  await c.query('UPDATE pos.wallet SET holder_name=$2 WHERE id=$1',[s.walletId,s.name+' wallet']);
  await c.query('UPDATE pos.wallet_entry SET business_date=$2,expires_at=$3 WHERE id=$1',[es[0].id,day,expiresAt]);
  await audit.record(drizzle(c),{actorAccountId:s.actorId,operatorId:br.operator_id,branchId:s.branchId,action:'qa.wallet.age',entityType:'wallet',entityId:s.walletId,sourceApp:'console',origin:'job',reason:'ZZ TEST expiry proof; only its grant date is aged, no money changed',before:{expiresAt:es[0].expires_at,businessDate:es[0].business_date},after:{expiresAt:expiresAt.toISOString(),businessDate:day,holderName:s.name+' wallet'}});
  await c.query('COMMIT');
  const result=await expireWalletsForDay(db,s.branchId,day,new Date());assert.equal(result.expiredWallets,1);assert.equal(result.expiredSatang,s.amount);
  const after=(await pool.query('SELECT id,holder_name,balance_satang,status FROM pos.wallet WHERE id=$1',[s.walletId])).rows[0];assert.equal(after.status,'expired');assert.equal(Number(after.balance_satang),0);
  const ledger=(await pool.query('SELECT kind,source,amount_satang,balance_after,business_date FROM pos.wallet_entry WHERE wallet_id=$1 ORDER BY created_at,id',[s.walletId])).rows;assert.deepEqual(ledger.map(e=>e.kind),['grant','expire']);
  console.log(JSON.stringify({event:'old_wallet_expiry',at:new Date().toISOString(),testFixtureAging:true,scope:s.branchId,walletId:s.walletId,originalGrant:{expiresAt:es[0].expires_at,businessDate:es[0].business_date},agedBusinessDate:day,result,wallet:after,ledger,cashPayments:p,passed:true}));
 }catch(e){await c.query('ROLLBACK');throw e;}finally{c.release();await pool.end();}
})().catch(e=>{console.error(JSON.stringify({event:'old_wallet_expiry_error',name:e.name,code:e.code??'ASSERTION',message:e.message}));process.exitCode=1;});`;
const result=await job(program,'old_wallet_expiry',r.deployments.find(d=>d.name==='oto-api-staging').sha);
writeFileSync(out+'/wallet-expiry-job.json',JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify(result));
