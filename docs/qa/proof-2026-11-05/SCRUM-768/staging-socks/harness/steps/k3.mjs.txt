export default async function (S) {
  const { sleep, text } = S.lib;
  const page = S.pages.CON.page;
  const dlg = page.getByRole('dialog');
  const mark = S.pages.CON.net.length;
  await dlg.getByRole('button', { name: 'Add ticket type', exact: true }).click();
  await sleep(3500);
  const net = S.pages.CON.net.slice(mark).map((n) => ({ at: n.at, method: n.method, path: n.path, status: n.status, id: n.response?.package?.id ?? n.response?.id ?? null, err: n.response?.error ?? null }));
  const created = net.find((n) => /ticket-packages$/.test(n.path) && n.method === 'POST');
  const f = S.st.fixture;
  if (created?.status < 300) {
    f.ticket = { id: created.id, name: 'ZZ TEST 768 Socks Staging' };
    S.st.created.push({ kind: 'ticket-package', id: created.id, name: 'ZZ TEST 768 Socks Staging', at: new Date().toISOString() });
    S.save();
  }
  const pk = await S.adminApi('GET', `/branches/${f.branchId}/ticket-packages?includeArchived=true`);
  const rows = Array.isArray(pk.body) ? pk.body : (pk.body.packages ?? []);
  const mine = rows.find((p) => p.id === f.ticket?.id);
  S.st.steps.ticketSaved = { at: new Date().toISOString(), net, name: mine?.name, active: mine?.active, freebies: mine?.freebies, prices: mine?.prices ?? null };
  S.save();
  return S.st.steps.ticketSaved;
}
