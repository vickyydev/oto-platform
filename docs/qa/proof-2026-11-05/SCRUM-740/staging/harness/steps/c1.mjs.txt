export default async function (S) {
  const { consoleSignIn, sleep, text } = S.lib;
  const page = await consoleSignIn(S);
  await page.goto(S.CONSOLE + '/watch/activity', { waitUntil: 'domcontentloaded' });
  await page.getByText('checkin.second_charge_refused').first().waitFor({ timeout: 30000 });
  await sleep(1500);
  await S.look('CON');
  return { text: (await text(page, 5000)).slice(0, 3500), rows: await page.getByText('checkin.second_charge_refused').count() };
}
