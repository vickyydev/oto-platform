const pages = {};
const newPage = async (label, viewport) => {
  if (pages[label]) return pages[label];
  const ctx = await browser.newContext({ viewport, timezoneId: 'Asia/Bangkok', locale: 'en-GB' });
  ctx.setDefaultTimeout(30000);
  const page = await ctx.newPage();
  page.on('pageerror', (e) => { (st.pageErrors = st.pageErrors ?? []).push({ at: new Date().toISOString(), label, message: String(e.message).slice(0, 300) }); save(); });
  pages[label] = { ctx, page };
  return pages[label];
};
const closePage = async (label) => { if (!pages[label]) return; await pages[label].ctx.close(); delete pages[label]; };

const call = async (ctxReq, method, path, data) => {
  const action = randomUUID();
  const res = await ctxReq.fetch(POS + '/api' + path, { method, ...(data !== undefined ? { data } : {}), headers: method === 'GET' ? {} : { 'idempotency-key': action, 'x-oto-action-id': action } });
  let body = null; try { body = await res.json(); } catch {}
  return { status: res.status(), body };
};
const adminApi = (m, p, d) => call(admin, m, p, d);
const pageApi = (label, m, p, d) => call(pages[label].page.request, m, p, d);
const raw = async (method, path) => { const res = await admin.fetch(POS + '/api' + path, { method }); return { status: res.status(), buf: await res.body() }; };
const shot = async (label, name, opts = {}) => {
  const page = pages[label].page;
  const file = OUT + '/' + name + '.png';
  await page.mouse.move(2, 2).catch(() => {});
  await page.evaluate(async () => { await document.fonts.ready; await new Promise((done) => requestAnimationFrame(() => requestAnimationFrame(done))); });
  await page.waitForTimeout(opts.wait ?? 500);
  await page.screenshot({ path: file, animations: 'disabled', mask: [page.locator('input[type=password]'), ...(opts.mask ?? [])], ...(opts.fullPage ? { fullPage: true } : {}), ...(opts.clip ? { clip: opts.clip } : {}) });
  st.images.push({ file: REL + '/' + name + '.png', at: new Date().toISOString(), page: label, url: page.url(), viewport: page.viewportSize() });
  save();
  return file;
};
const S = { st, save, pages, newPage, closePage, browser, admin, adminApi, pageApi, call, raw, shot, expect, need, POS, CONSOLE, LANE, REL, OUT, SOURCE, randomUUID };

  }
  save();
  res.setHeader('content-type', 'application/json');
  res.end(JSON.stringify(out, null, 2));
});
