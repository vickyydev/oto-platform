const command = (exe, args, cwd = WORKTREE) => { const r = spawnSync(exe, args, { cwd, encoding: 'utf8', timeout: TIMEOUT, maxBuffer: 5_000_000, stdio: ['ignore', 'pipe', 'pipe'] }); need(r.status === 0 && !r.signal, 'BOUNDED_METADATA_COMMAND_FAILED'); return r.stdout; };
async function json(response) { need(response.status === 200 && response.body, 'METADATA_200_REQUIRED'); const chunks = []; let n = 0;
  for await (const part of response.body) { n += part.length; need(n <= 5_000_000, 'METADATA_BYTE_LIMIT'); chunks.push(part); }
  try { return JSON.parse(Buffer.concat(chunks)); } catch { throw Error('METADATA_JSON_REQUIRED'); }
}

/** Same TypeScript AST rules as the reviewed634/637 collectors: permission strings/API templates are not imports/assets. */
export function jsReferences(text, ts) {
  const root = ts.createSourceFile('public.js', text, ts.ScriptTarget.Latest, true, ts.ScriptKind.JS); need(root.parseDiagnostics.length === 0, 'PUBLIC_JS_PARSE_FAILED'); const refs = [];
  const literal = n => n && (ts.isStringLiteral(n) || ts.isNoSubstitutionTemplateLiteral(n));
  const visit = n => {
    if (ts.isImportDeclaration(n) || ts.isExportDeclaration(n) && n.moduleSpecifier) { need(literal(n.moduleSpecifier), 'STATIC_MODULE_LITERAL_REQUIRED'); refs.push({ value: n.moduleSpecifier.text, viteRoot: false }); }
    if (ts.isCallExpression(n) && n.expression.kind === ts.SyntaxKind.ImportKeyword) { need(n.arguments.length >= 1 && n.arguments.length <= 2 && literal(n.arguments[0]), 'COMPUTED_MODULE_REFUSED'); refs.push({ value: n.arguments[0].text, viteRoot: false }); }
    if (ts.isNewExpression(n) && ts.isIdentifier(n.expression) && n.expression.text === 'URL' && n.arguments?.length === 2) {
      const base = n.arguments[1], meta = ts.isPropertyAccessExpression(base) && base.name.text === 'url' && ts.isMetaProperty(base.expression) && base.expression.keywordToken === ts.SyntaxKind.ImportKeyword;
      if (meta) { need(literal(n.arguments[0]), 'COMPUTED_ASSET_REFUSED'); refs.push({ value: n.arguments[0].text, viteRoot: false }); }
    }
    if (literal(n) && (/^\/?assets\//.test(n.text) || /^(?:\/|\.\.?\/|https?:\/\/)/.test(n.text) && /\.(?:js|css|woff2?|ttf|svg|png|webp|jpe?g|gif|avif|ico|wasm)(?:[?#].*)?$/.test(n.text))) refs.push({ value: n.text, viteRoot: true });
    if (ts.isTemplateExpression(n) && /^\/?assets\//.test(n.head.text)) need(false, 'COMPUTED_ASSET_REFUSED');
    ts.forEachChild(n, visit);
  }; visit(root); return refs;
}
export function resolvePublic(raw, base, viteRoot = false) {
  if (typeof raw === 'string' && (raw.startsWith('data:') || raw.startsWith('#'))) return null;
  need(typeof raw === 'string' && raw.length > 0 && raw.length < 2048 && !/[\\%\s]/.test(raw) && !raw.split(/[/?#]/).includes('..'), 'PUBLIC_REFERENCE_REFUSED');
  const u = new URL(viteRoot && raw.startsWith('assets/') ? '/' + raw : raw, ORIGIN + base);
  need(u.origin === ORIGIN && !u.username && !u.password && !u.search && !u.hash && publicPath(u.pathname) && u.pathname !== '/', 'PUBLIC_REFERENCE_REFUSED'); return u.pathname;
}
export async function collectAssets(input, outputFile, { explicitExecute = false } = {}) {
