assert.ok(['enable','restore'].includes(action));assert.match(sha??'',/^[a-f0-9]{40}$/);
assert.equal(process.env.S728_WINDOW_RELEASED,'true');
const root='C:/oto-lanes/codex-box-outage-release-20261104';

const out=root+'/docs/qa/proof-2026-11-02/s728';mkdirSync(out,{recursive:true});
const hash=value=>createHash('sha256').update(value).digest('hex');
const read=async id=>{const rows=await render('GET',`/services/${id}/env-vars?limit=100`);assert.ok(Array.isArray(rows)&&rows.length<100);return Object.fromEntries(rows.map(row=>{const v=row.envVar??row;assert.equal(typeof v.value,'string');return[v.key,v.value];}));};
const values={
 'oto-api-staging':{BOX_BRIDGE_ZONE:'localhost',BOX_BRIDGE_ALLOW_LOOPBACK:'true',STAFF_OFFLINE_SIGN_IN:'true'},
 'oto-pos-web-staging':{VITE_BOX_BRIDGE_ZONE:'localhost',VITE_BOX_BRIDGE_ALLOW_LOOPBACK:'true'},
};
const baseline=action==='restore'?JSON.parse(readFileSync(out+'/settings-enable.json','utf8')):null;
const receipt={action,source:sha,ciRun,startedAt:new Date().toISOString(),services:[]};
const save=()=>writeFileSync(out+`/settings-${action}.json`,JSON.stringify(receipt,null,2)+'\n');
for(const [name,changes] of Object.entries(values)) {
 const s=await service(name);assert.ok(s?.id);
 const live=(await render('GET',`/services/${s.id}/deploys?limit=1`))[0]?.deploy;
 assert.equal(live.status,'live');assert.equal(live.commit.id,sha);
 const before=await read(s.id);
 if(name==='oto-api-staging') {
   assert.ok(Object.keys(before).length>=10);
   assert.ok(before.STAFF_TOKEN_PRIVATE_KEY?.trim(),'Existing signing key must remain available');
 } else {
   // The forwarding service has four baseline settings, unlike the API.
   for(const key of ['API_UPSTREAM','NODE_ENV','NODE_VERSION','VITE_LAUNCHER_URL'])
     assert.ok(before[key]?.trim(),`Missing required POS setting: ${key}`);
 }
 const prior=baseline?.services.find(x=>x.name===name);
 if(prior)assert.deepEqual(Object.fromEntries(Object.entries(before).map(([k,v])=>[k,hash(v)])),prior.afterFingerprints,'Environment changed since proof; inspect before restoration');
 const record={name,id:s.id,countBefore:Object.keys(before).length,namesBefore:Object.keys(before).sort(),beforeFingerprints:Object.fromEntries(Object.entries(before).map(([k,v])=>[k,hash(v)])),changedNames:[],restore:{}};
 receipt.services.push(record);save();
 for(const [key,target] of Object.entries(changes)) {
   assert.notEqual(key,'BOX_AGENT_TEST_SLOT');assert.notEqual(key,'STAFF_TOKEN_PRIVATE_KEY');
   const old=before[key];
   if(action==='enable') {
     // Only absent/empty test-zone settings or an explicit boolean are eligible.
     assert.ok(old===undefined||old===''||(!key.endsWith('_ZONE')&&['true','false'].includes(old)),'Existing policy needs inspection');
     record.restore[key]={present:old!==undefined,value:old??null};save();
   }
   const restore=prior?.restore[key];
   const next=action==='enable'?target:restore?.value;
   assert.equal(await envValue(s.id,key),old??null,'Named read disagrees with full read');
   if(action==='restore'&&!restore.present)await render('DELETE',`/services/${s.id}/env-vars/${key}`);
   else if(old!==next)await render('PUT',`/services/${s.id}/env-vars/${key}`,{value:next});
   assert.equal(await envValue(s.id,key),action==='restore'&&!restore.present?null:next,'Named write did not read back');
   record.changedNames.push(key);save();
 }
 const after=await read(s.id);
 const untouched=Object.keys(before).filter(k=>!(k in changes));
 assert.ok(untouched.every(k=>before[k]===after[k]));
 assert.ok(Object.keys(after).every(k=>k in before||k in changes));
 if(action==='restore')assert.deepEqual(Object.fromEntries(Object.entries(after).map(([k,v])=>[k,hash(v)])),prior.beforeFingerprints);
 record.countAfter=Object.keys(after).length;record.untouchedValuesByteIdentical=true;
 record.afterFingerprints=Object.fromEntries(Object.entries(after).map(([k,v])=>[k,hash(v)]));save();
 const dep=await render('POST',`/services/${s.id}/deploys`,{commitId:sha});assert.ok(dep.id,'Deploy not created');record.deployId=dep.id;save();
 console.log(JSON.stringify({name,action,changedNames:record.changedNames,countBefore:record.countBefore,countAfter:record.countAfter,deployId:dep.id,untouchedValuesByteIdentical:true}));
}
receipt.settingsWrittenAt=new Date().toISOString();save();
