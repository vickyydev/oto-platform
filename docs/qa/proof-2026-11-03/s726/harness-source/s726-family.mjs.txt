import assert from 'node:assert/strict';import {randomUUID} from 'node:crypto';
export async function makeFamily(rig){
 const f=rig.f,ctx=rig.admin;await rig.api(ctx,'PUT','/me/session/station',{stationId:f.stations[0].id});const config=await rig.api(ctx,'GET','/checkin/config?branchId='+f.branchId),packages=(await rig.api(ctx,'GET','/branches/'+f.branchId+'/ticket-packages')).packages;const pkg=packages.find(x=>x.name==='2 Hours Play'&&x.active&&!x.archivedAt);assert.ok(pkg);
 const registrationId=randomUUID(),checkinId=randomUUID(),saleId=randomUUID(),name=f.name+' child '+((f.previousFamilies?.length??0)+1);f.family={registrationId,checkinId,saleId,childName:name};rig.save();
 await rig.api(ctx,'POST','/checkin/registrations',{id:registrationId,branchId:f.branchId,stationId:f.stations[0].id,guardianName:f.name+' guardian',contactChannel:'whatsapp',consentAcknowledged:true,acknowledgedConfirmationIds:config.policy.confirmations.map(x=>x.id),children:[{checkinId,name,ageYears:7,service:'drop_off',foodProvision:{mode:'none',paidSatang:0}}]});
 const pg=await rig.browser.newPage({viewport:{width:640,height:480}});await pg.setContent('<body style="background:#163345;color:white;font:30px sans-serif;padding:100px">ZZ TEST SCRUM-726<br>Synthetic image<br>No person photographed</body>');const photo=await pg.screenshot();await pg.close();const photoId=randomUUID();f.family.photoId=photoId;rig.save();await rig.api(ctx,'POST','/files',{id:photoId,contentType:'image/png',ownerEntityType:'registration',ownerEntityId:registrationId,filename:'zz-test-s726.png'});assert.ok((await ctx.put('https://oto-pos-web-staging.onrender.com/api/files/'+photoId+'/content',{data:photo,headers:{'content-type':'image/png'}})).ok());await rig.api(ctx,'POST','/checkin/registrations/'+registrationId+'/photo',{fileId:photoId,checkinIds:[checkinId]});
 await rig.api(ctx,'POST','/sales',{id:saleId,stationId:f.stations[0].id,lines:[{id:checkinId,packageId:pkg.id,kids:1,adults:0,serviceFee:{label:'ZZ TEST drop-off service',amountSatang:config.pricing.oneTimeFee.weekday}}]});const paid=await rig.api(ctx,'POST','/sales/'+saleId+'/finalise',{method:'cash',kind:'cash'});assert.equal(paid.sale.status,'finalised');f.family.grossSatang=paid.sale.totals.grossSatang;f.family.receiptNumber=paid.sale.receiptNumber;rig.save();await rig.api(ctx,'POST','/checkin/leave-as-booked',{saleId,entries:[{checkinId}]});await rig.api(ctx,'POST','/checkin/check-in-booked',{entries:[{checkinId}],consentAcknowledged:true});const reg=await rig.api(ctx,'GET','/checkin/registrations/'+registrationId),stay=reg.children.find(x=>x.id===checkinId);assert.equal(stay.status,'in_park');const detail=await rig.api(ctx,'GET','/sales/'+saleId),band=detail.bands.find(x=>x.id===stay.bandId);assert.ok(band?.shortCode);f.family.bandId=band.id;f.family.inPark=true;rig.save();rig.bandCode=band.shortCode;rig.photo=photo;return f.family;
}
export async function retireFamily(rig){const f=rig.f;if(!f.family?.inPark)return;await rig.api(rig.admin,'PUT','/me/session/station',{stationId:f.stations[0].id});const pickupId=randomUUID();f.family.pickupPhotoId=pickupId;rig.save();await rig.api(rig.admin,'POST','/files',{id:pickupId,contentType:'image/png',ownerEntityType:'registration',ownerEntityId:f.family.registrationId,filename:'zz-test-s726-pickup.png'});assert.ok((await rig.admin.put('https://oto-pos-web-staging.onrender.com/api/files/'+pickupId+'/content',{data:rig.photo,headers:{'content-type':'image/png'}})).ok());const result=await rig.api(rig.admin,'POST','/checkin/pickups/stays/'+f.family.checkinId+'/release',{id:randomUUID(),collector:{kind:'dropper_off'},pickupPhotoFileId:pickupId,stationId:f.stations[0].id});const ended=await rig.api(rig.admin,'GET','/checkin/registrations/'+f.family.registrationId);assert.equal(ended.children.find(x=>x.id===f.family.checkinId).status,'out');f.family.inPark=false;f.family.releasedAt=new Date().toISOString();f.family.releaseId=result.id??result.release?.id;rig.save();}
