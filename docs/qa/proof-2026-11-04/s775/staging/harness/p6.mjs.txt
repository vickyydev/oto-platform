// Retire the ZZ TEST S775 login (inactive, no grants, unlinked from the OTO App) and read everything back.
import { api, raw, assert, adminConsole, CONSOLE, SHA, APP_TENANT, loadState, saveState, writeReceipt, chromium, signOut, readJob, liveDeploy } from './lib.mjs';
const state = loadState();
const note = (name, detail = {}) => { state.log.push({ step: name, ...detail, at: new Date().toISOString() }); saveState(state); console.log('OK', name, JSON.stringify(detail).slice(0, 900)); };
state.cleanup ??= {};
const browser = await chromium.launch({ channel: 'msedge' });
const admin = await adminConsole(browser);
try {
  const A = state.agent;
  if (!state.cleanup.login) {
    await api(admin.ctx, CONSOLE, 'PATCH', '/accounts/' + A.accountId, { status: 'inactive' });
    const perms = await api(admin.ctx, CONSOLE, 'GET', '/accounts/' + A.accountId + '/permissions');
    for (const g of perms.assignments ?? []) await api(admin.ctx, CONSOLE, 'DELETE', '/accounts/' + A.accountId + '/role-assignments/' + g.id);
    const un = await raw(admin.ctx, CONSOLE, 'DELETE', '/admin/apps/oto_app/users/' + A.accountId);
    assert.ok([200, 404].includes(un.status), 'unlink ' + un.status);
    state.cleanup.login = { at: new Date().toISOString(), grantsRemoved: (perms.assignments ?? []).length, unlink: un.status };
    saveState(state);
    note('Clean-up: the ZZ TEST S775 login is inactive, with no grants, and unlinked from the OTO App', state.cleanup.login);
  }
} finally {
  await signOut(admin);
  await browser.close();
}
const C = state.conv.id, E = state.event.id, A = state.agent;
const r = await readJob('s775_cleanup_read', `
const C=${JSON.stringify(C)},E=${JSON.stringify(E)},A=${JSON.stringify(A.accountId)},U=${JSON.stringify(A.appUserId)};
const ev=await q("SELECT id,status,is_archived,(archived_at IS NOT NULL) archived,updated_at FROM otoapp.core_events WHERE id=$1",[E]);
const eventsForConv=await q("SELECT count(*)::int n FROM otoapp.core_events WHERE inbox_conversation_id=$1",[C]);
const linkedAll=await q("SELECT count(*)::int n FROM otoapp.core_events WHERE inbox_conversation_id IS NOT NULL");
const tentative=await q("SELECT name,value,color,sort_order FROM otoapp.event_statuses WHERE tenant_id=$1 AND value='tentative'",[${JSON.stringify(APP_TENANT)}]);
const conv=await q("SELECT status,(handled_at IS NOT NULL) handled,handled_by_account_id=$2 handled_by_login,event_order_ref FROM inbox.conversation WHERE id=$1",[C,A]);
const books=await q("SELECT count(*)::int n FROM core.audit_log WHERE entity_id=$1 AND action LIKE 'inbox.conversation.book_date%'",[C]);
const beoHandoffs=await q("SELECT count(*)::int n FROM core.audit_log WHERE action='auth.handoff_issue' AND after->>'purpose'='inbox.create_beo'");
const account=await q("SELECT a.status,(SELECT count(*)::int FROM core.role_assignment ra WHERE ra.account_id=a.id AND ra.archived_at IS NULL) grants,(SELECT count(*)::int FROM core.session s WHERE s.account_id=a.id AND s.revoked_at IS NULL AND s.expires_at>now()) sessions,(SELECT count(*)::int FROM core.app_identity i WHERE i.account_id=a.id) links FROM core.account a WHERE a.id=$1",[A]);
const appUser=await q("SELECT is_active FROM otoapp.users WHERE id=$1",[U]);
return {ev,eventsForConv,linkedAll,tentative,conv,books,beoHandoffs,account,appUser};`, SHA);
const x = r.result;
console.log(JSON.stringify(x));
assert.equal(x.ev[0].is_archived, true);
assert.equal(x.ev[0].status, 'tentative');
assert.equal(x.eventsForConv[0].n, 1);
assert.equal(x.tentative.length, 1);
assert.equal(x.conv[0].status, 'handled');
assert.equal(x.conv[0].event_order_ref, E);
assert.equal(x.books[0].n, 1);
assert.equal(x.beoHandoffs[0].n, 1);
assert.equal(x.account[0].status, 'inactive');
assert.equal(x.account[0].grants, 0);
assert.equal(x.account[0].sessions, 0);
assert.equal(x.account[0].links, 0);
state.cleanup.readBack = { jobId: r.jobId, ...x };
const live = [];
for (const n of ['oto-api-staging', 'oto-inbox-staging', 'oto-app-staging', 'oto-pos-staging', 'oto-launcher-staging', 'oto-console-staging']) live.push(await liveDeploy(n));
state.liveAfter = { at: new Date().toISOString(), live };
saveState(state);
for (const d of live) assert.equal(d.sha, SHA, d.name + ' moved');
writeReceipt('cleanup.json', { source: SHA, steps: state.log.filter((l) => /Clean-up|chip|after clean-up/i.test(l.step)), readBack: state.cleanup.readBack });
writeReceipt('live-after.json', state.liveAfter);
note('Clean-up read back; services still on the SHA', { readBack: x, deploysUnchanged: live.every((d, i) => d.deployId === state.preflight.live[i].deployId) });
