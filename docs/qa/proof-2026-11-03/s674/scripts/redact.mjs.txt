// Paint solid boxes over credentials in a saved screenshot (the wallet credit QR codes), in place.
import { launch } from './common.mjs';
import { readFileSync, writeFileSync } from 'node:fs';
const [file, boxesJson] = process.argv.slice(2);
const boxes = JSON.parse(boxesJson);
const browser = await launch();
try {
  const page = await browser.newPage();
  const src = 'data:image/png;base64,' + readFileSync(file).toString('base64');
  const out = await page.evaluate(async ({ src, boxes }) => {
    const img = new Image(); img.src = src; await img.decode();
    const c = document.createElement('canvas'); c.width = img.naturalWidth; c.height = img.naturalHeight;
    const g = c.getContext('2d'); g.drawImage(img, 0, 0); g.fillStyle = '#ff00ff';
    for (const [x, y, w, h] of boxes) g.fillRect(x, y, w, h);
    return c.toDataURL('image/png').split(',')[1];
  }, { src, boxes });
  writeFileSync(file, Buffer.from(out, 'base64'));
  console.log('redacted', boxes.length);
} finally { await browser.close(); }
