export default async function (S) {
  const { LOOK, text } = S.lib;
  const page = S.pages.A.page;
  await page.getByRole('heading', { name: '2 Hours Play', exact: true }).click();
  await page.waitForTimeout(2500);
  await page.screenshot({ path: LOOK });
  return { text: (await text(page, 4000)).replace(/\+?\d{8,}/g, '[phone]').split('Glow Band')[1] };
}
