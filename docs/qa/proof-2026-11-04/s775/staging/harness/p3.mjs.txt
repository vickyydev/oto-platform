// Step 2: the ZZ TEST birthdays-team login opens the conversation, presses Book Date, checks the
// pre-filled confirm, adds the child and presses Book date twice quickly.
import { api, assert, adminConsole, persona, INBOX, loadState, saveState, writeReceipt, chromium, shot, signOut } from './lib.mjs';
const state = loadState();
const note = (name, detail = {}) => { state.log.push({ step: name, ...detail, at: new Date().toISOString() }); saveState(state); console.log('OK', name, JSON.stringify(detail).slice(0, 900)); };
const browser = await chromium.launch({ channel: 'msedge' });
const admin = await adminConsole(browser);
let ag;
try {
  ag = await persona(browser, admin, state.agent);
  const me = ag.me;
  state.agentMe = { name: me.viewer?.name ?? me.name ?? null, assign: me.capabilities?.assign, writes: me.capabilities?.writes, teams: (me.teams ?? []).map((t) => ({ kind: t.kind, name: t.name, member: t.member })) };
  saveState(state);
  note('The birthdays-team login in the inbox', state.agentMe);
  const p = ag.page;
  const posts = [];
  p.on('request', (r) => { if (r.method() === 'POST' && /\/api\/inbox\/conversations\/[^/]+\/book-date$/.test(new URL(r.url()).pathname)) posts.push({ at: new Date().toISOString() }); });
  p.on('response', (r) => { if (r.request().method() === 'POST' && /\/book-date$/.test(new URL(r.url()).pathname)) posts.push({ status: r.status(), replay: r.headers()['x-oto-replay'] ?? null }); });
  await p.goto(INBOX + '/?c=' + state.conv.id);
  await p.getByRole('button', { name: /^Book Date$/ }).first().waitFor({ timeout: 60000 });
  await p.waitForTimeout(1500);
  const bookEnabled = await p.getByRole('button', { name: /^Book Date$/ }).first().isEnabled();
  const beoEnabled = await p.getByRole('button', { name: /^Create BEO$/ }).first().isEnabled();
  assert.ok(bookEnabled && beoEnabled, 'Book Date / Create BEO not enabled');
  await shot(p, 's775-st-02-birthday-chat-book-date-and-create-beo-live.png');
  await p.getByRole('button', { name: /^Book Date$/ }).first().click();
  const dialog = p.getByRole('dialog');
  await dialog.locator('input[aria-label="Date"]').waitFor({ timeout: 30000 });
  await p.waitForTimeout(600);
  const values = {
    title: (await dialog.locator('h2, h3').first().textContent())?.trim(),
    date: await dialog.locator('input[aria-label="Date"]').inputValue(),
    startTime: await dialog.locator('input[aria-label="Start time"]').inputValue(),
    park: await dialog.locator('select[aria-label="Park"]').evaluate((el) => el.selectedOptions[0]?.textContent ?? ''),
    kids: await dialog.locator('input[aria-label="Kids"]').inputValue(),
    adults: await dialog.locator('input[aria-label="Adults"]').inputValue(),
    bookingName: await dialog.locator('input[aria-label="Booking name"]').inputValue(),
    phoneShape: (await dialog.locator('input[aria-label="Phone"]').inputValue()).replace(/\d(?=\d{4})/g, '•'),
    childName: await dialog.locator('input[aria-label="Child\'s name"]').inputValue(),
    age: await dialog.locator('input[aria-label="Turning age"]').inputValue(),
    dialogText: (await dialog.textContent())?.replace(/\s+/g, ' ').slice(0, 700),
  };
  state.dialog = values; saveState(state);
  await shot(p, 's775-st-03-book-date-confirm-prefilled.png');
  note('Step 2: the confirm read from the chat', values);
  assert.equal(values.date, '2027-03-21');
  assert.equal(values.startTime, '14:00');
  assert.equal(values.kids, '12');
  assert.equal(values.age, '6');
  assert.ok(/Central Floresta/.test(values.park), 'park');
  assert.equal(values.bookingName, 'ZZ TEST S775 Birthday');
  await dialog.locator('input[aria-label="Child\'s name"]').fill('ZZ TEST Child');
  await p.waitForTimeout(300);
  await shot(p, 's775-st-03b-confirm-with-child-before-save.png');
  await dialog.getByRole('button', { name: /^Book date$/ }).dblclick({ delay: 30 });
  const outcome = await Promise.race([
    p.waitForSelector('[data-testid="event-chip"]', { timeout: 60000 }).then(() => 'chip'),
    p.waitForSelector('[role="dialog"] [role="alert"]', { timeout: 60000 }).then(() => 'alert'),
  ]);
  if (outcome !== 'chip') {
    state.dialogAlert = await p.textContent('[role="dialog"] [role="alert"]');
    saveState(state);
    await shot(p, 'fail-book-date-refused.png');
    throw new Error('Book Date refused: ' + state.dialogAlert);
  }
  await p.waitForTimeout(2500);
  state.bookedAt = new Date().toISOString();
  state.uiPosts = posts; saveState(state);
  const chip = (await p.locator('[data-testid="event-chip"]').textContent())?.trim();
  const booked = await p.getByRole('button', { name: /Date Booked/ }).first().isDisabled();
  await p.locator('[data-testid="event-chip"]').scrollIntoViewIfNeeded();
  await shot(p, 's775-st-04-date-booked-and-event-chip.png');
  const pill = p.locator('text=Date blocked in the Events calendar').first();
  await pill.waitFor({ timeout: 30000 });
  const pillText = (await pill.textContent())?.trim();
  await pill.scrollIntoViewIfNeeded();
  await shot(p, 's775-st-05-timeline-pill-date-blocked.png');
  state.strip = { chip, dateBookedDisabled: booked, pill: pillText };
  saveState(state);
  note('Step 2: booked with a double press', { ...state.strip, uiPosts: posts });
  const c = await api(ag.ctx, INBOX, 'GET', '/inbox/conversations/' + state.conv.id);
  state.convAfter = { event: c.event ?? c.conversation?.event ?? null };
  saveState(state);
  writeReceipt('step2-book-date-ui.json', { conversationId: state.conv.id, login: state.agentMe, dialog: values, childAdded: 'ZZ TEST Child', doublePress: 'dblclick on Book date', uiPostsSeen: posts, strip: state.strip, conversationEventAsRead: state.convAfter.event, bookedAt: state.bookedAt });
} catch (e) {
  console.error('FAIL', e.message.split('\n')[0]);
  if (ag) await shot(ag.page, 'fail-p3.png').catch(() => {});
  process.exitCode = 1;
} finally {
  await signOut(ag);
  await signOut(admin);
  await browser.close();
}
