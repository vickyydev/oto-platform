// The Console's Support page, as a person uses it: type into "Search", press Search, read what comes back.
import { CONSOLE, OUT } from './common.mjs';
export const grouped = (code) => `${code.slice(0, 2)} ${code.slice(2, 6)} ${code.slice(6, 10)} ${code.slice(10)}`;
export async function openSupport(page) {
  await page.goto(CONSOLE + '/watch/support', { waitUntil: 'domcontentloaded' });
  await page.getByText('Find anything', { exact: true }).first().waitFor({ timeout: 30000 });
  await page.waitForTimeout(1000);
}
const card = (page) => page.locator('section').filter({ has: page.getByText('Find anything', { exact: true }) }).first();
/** One search; answers the server's own answer (sections only, never the typed value) and, when asked, a screenshot. */
export async function search(page, q, { file = null, mask = false, kind = null } = {}) {
  const c = card(page);
  const input = c.locator('input').first();
  await input.fill('');
  await input.fill(q);
  if (kind) await c.locator('select').first().selectOption(kind);
  const wait = page.waitForResponse((r) => r.url().includes('/api/console/support/lookup') && r.request().method() === 'GET', { timeout: 60000 });
  await c.getByRole('button', { name: 'Search', exact: true }).click();
  const res = await wait;
  const body = await res.json().catch(() => null);
  await page.waitForTimeout(800);
  await c.locator('[data-testid="lookup-results"], text=Nothing to search for').first().waitFor({ timeout: 30000 }).catch(() => {});
  const answer = { status: res.status(), at: new Date().toISOString(), kinds: body?.kinds ?? null, masked: body?.masked ?? null,
    sections: (body?.sections ?? []).map((s) => ({ key: s.key, kind: s.kind, status: s.status, rowCount: s.rowCount, masked: s.masked, permission: s.permission, note: s.note, rows: s.rows })) };
  if (file) {
    await c.scrollIntoViewIfNeeded();
    await c.screenshot({ path: `${OUT}/${file}`, mask: mask ? [input] : [] });
    answer.file = file;
  }
  return answer;
}
