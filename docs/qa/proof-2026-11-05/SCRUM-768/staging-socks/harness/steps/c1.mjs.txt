export default async function (S) {
  const { sleep, text, consoleSignIn } = S.lib;
  const page = await consoleSignIn(S);
  await page.goto(S.CONSOLE + '/catalogue/inventory', { waitUntil: 'domcontentloaded' });
  await sleep(5000);
  await S.look('CON');
  return { url: page.url(), t: (await text(page, 6000)).slice(0, 3500) };
}
