/** One request through a browser context's cookie jar. Refuses an unexpected status. */
export const api = async (ctx, origin, method, path, body, status = 200) => {
  const headers = method === 'GET' ? {} : { 'Idempotency-Key': randomUUID() };
  const r = await ctx.request.fetch(origin + '/api' + path, { method, data: body, headers });
  const text = await r.text();
  let b = null;
  try { b = JSON.parse(text); } catch { b = text.slice(0, 200); }
  const want = Array.isArray(status) ? status : [status];
  assert.ok(want.includes(r.status()), 'HTTP ' + r.status() + ' ' + method + ' ' + path + ' ' + (b?.error?.code ?? b?.reason ?? b?.message ?? b?.error?.message ?? ''));
  return b;
};
/** The same, answering status, headers and body without refusing. */
export const raw = async (ctx, origin, method, path, body, headers = {}) => {
  const h = method === 'GET' ? headers : { 'Idempotency-Key': randomUUID(), ...headers };
  const r = await ctx.request.fetch(origin + '/api' + path, { method, data: body, headers: h });
  let b = null;
  try { b = await r.json(); } catch { b = null; }
  return { status: r.status(), headers: r.headers(), body: b };
};

export const newContext = (browser, extra = {}) =>
  browser.newContext({ viewport: { width: 1440, height: 1000 }, timezoneId: 'Asia/Bangkok', locale: 'en-GB', ...extra });

const arrive = async (ctx, url, oto) => {
  const p = await ctx.newPage();
  const exchange = p.waitForResponse((r) => new URL(r.url()).pathname === (oto ? '/api/auth/handoff' : '/api/auth/handoff/exchange') && r.request().method() === 'POST', { timeout: 90000 });
  await p.goto(url);
  const response = await exchange;
  assert.equal(response.status(), 200, 'Native handoff refused');
  await response.finished();
  await p.waitForURL((u) => !u.hash.includes('handoff'), { timeout: 60000 });
  return p;
};
export const hand = async (ctx, app) => {
  const h = await api(ctx, LAUNCHER, 'POST', '/auth/handoff', { app });
  return arrive(ctx, h.launchUrl, app === 'oto_app');
};

/** The staging administrator, signed in to the Console (Console writes and the simulator panel). */
export const adminConsole = async (browser) => {
  const ctx = await newContext(browser);
  await api(ctx, LAUNCHER, 'POST', '/auth/sign-in', { phone: need('STAGING_ADMIN_PHONE'), password: need('STAGING_ADMIN_PASSWORD') });
  const page = await hand(ctx, 'console');
  await page.waitForURL('**/overview', { timeout: 60000 });
  const me = await api(ctx, CONSOLE, 'GET', '/me');
  return { ctx, page, me, actorId: me.account?.id ?? me.accountId, operatorId: me.operatorContext.effectiveOperatorId };
};

export const findAccount = async (admin, accountId, name) =>
  (await api(admin.ctx, CONSOLE, 'GET', '/accounts?q=' + encodeURIComponent(name))).accounts.find((x) => x.id === accountId);

/**
 * A ZZ TEST persona in its own browser: a fresh temporary password from the
 * Console, the forced change to a random password nobody keeps, then the
 * launcher's hand-off into the inbox. No SMS is involved.
 */
export const persona = async (browser, admin, { accountId, name }, extra = {}) => {
  const acct = await findAccount(admin, accountId, name);
  const phone = acct?.phone;
  assert.ok(phone, 'no phone on the account read');
  const ctx = await newContext(browser, extra);
  const temp = await api(admin.ctx, CONSOLE, 'POST', '/accounts/' + accountId + '/temp-password', {});
  await api(ctx, LAUNCHER, 'POST', '/auth/sign-in', { phone, password: temp.temporaryPassword });
  await api(ctx, LAUNCHER, 'POST', '/auth/change-password', { currentPassword: temp.temporaryPassword, password: randomBytes(30).toString('base64url') });
  delete temp.temporaryPassword;
  const page = await hand(ctx, 'inbox');
  const me = await api(ctx, INBOX, 'GET', '/inbox/me');
  return { ctx, page, me };
};
