/** No caller may copy on import or implicit invocation. The complete manifest is revalidated before any copy. */
export function copyEvidence(input, { explicitExecute = false } = {}) {
  need(explicitExecute === true, 'EXPLICIT_COPY_REQUIRED');
  const manifest = buildCopyManifest(input); need(!fs.existsSync(TARGET), 'PRESERVE_EXISTING_EVIDENCE');
  const contents = manifest.files.map(row => { const value = bytes(row.source); need(hash(value) === row.sha256 && value.length === row.bytes, 'EVIDENCE_CHANGED_BEFORE_COPY'); return value; });
  fs.mkdirSync(TARGET);
  for (const [index, row] of manifest.files.entries()) {
    const target = path.join(TARGET, row.relative); fs.mkdirSync(path.dirname(target), { recursive: true });
    fs.writeFileSync(target, contents[index], { flag: 'wx', flush: true }); need(hash(bytes(target)) === row.sha256, 'EVIDENCE_COPY_READBACK_DIFFERS');
  }
  fs.writeFileSync(path.join(TARGET, 'copy-manifest.json'), JSON.stringify(manifest, null, 2) + '\n', { flag: 'wx', flush: true });
  return { target: TARGET, files: manifest.files.length, resultSha256: manifest.resultSha256, sourceSha: SOURCE, ciRunId: CI, jiraOrDocsChanged: false };
}

if (process.argv[1] && path.resolve(process.argv[1]) === path.resolve(fileURLToPath(import.meta.url))) {
  try {
    need(process.argv.length === 4 && ['--manifest', '--copy'].includes(process.argv[2]), 'USAGE_MANIFEST_OR_COPY_REVIEWED_INPUT');
    const input = json(process.argv[3]);
    const output = process.argv[2] === '--copy' ? copyEvidence(input, { explicitExecute: true }) : buildCopyManifest(input);
    process.stdout.write(JSON.stringify(output) + '\n');
  } catch (error) { process.stderr.write(/^[A-Z][A-Z0-9_]+$/.test(error.message ?? '') ? error.message + '\n' : 'EVIDENCE_NOT_READY\n'); process.exitCode = 1; }
