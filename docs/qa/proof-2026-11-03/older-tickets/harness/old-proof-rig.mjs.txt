const {chromium,expect}=createRequire(runtime+'/apps/pos/package.json')('@playwright/test');
const CONSOLE='https://oto-console-staging.onrender.com',POS='https://oto-pos-web-staging.onrender.com',LAUNCHER='https://oto-launcher-staging.onrender.com';
export async function rig() {
 const stamp=Date.now().toString(36),name='ZZ TEST older proof '+stamp;
 const out=root+'/docs/qa/proof-2026-11-03/older-tickets/'+stamp;mkdirSync(out,{recursive:true});
 const receipt={startedAt:new Date().toISOString(),name,deployments:[],checks:[],fixtures:{},screens:[],outcome:'incomplete'};
 const save=()=>writeFileSync(out+'/receipt.json',JSON.stringify(receipt,null,2)+'\n');
 async function live() {
  const rows=[];
  for(const id of ['srv-danh2ljtqb8s73bt9en0','srv-dbk7qu8n74is73bm894g','srv-danh2mf40ujc73bu80j0','srv-danng2p42hec73f38e3g']) {
   const s=await render('GET',`/services/${id}`),d=(await render('GET',`/services/${id}/deploys?limit=1`))[0]?.deploy;

   rows.push({id,name:s.name,sha:d.commit.id,deployId:d.id,liveAt:d.finishedAt});
  }
  return rows;
 }
 receipt.deployments=await live();
 receipt.ci=JSON.parse(execFileSync('gh',['run','list','--branch','main','--limit','15','--json','databaseId,headSha,status,conclusion'],{cwd:root,encoding:'utf8'})).filter(r=>receipt.deployments.some(d=>d.sha===r.headSha));
 assert.ok(receipt.ci.length&&receipt.ci.every(r=>r.status==='completed'&&r.conclusion==='success'));save();
 const browser=await chromium.launch({channel:'msedge'});
 const ctx=await browser.newContext({viewport:{width:1500,height:1100}});ctx.setDefaultTimeout(20000);
 const p=await ctx.newPage();
 const sign=await ctx.request.post(LAUNCHER+'/api/auth/sign-in',{data:{phone:need('STAGING_ADMIN_PHONE'),password:need('STAGING_ADMIN_PASSWORD')}});assert.equal(sign.status(),200);
 const h=await ctx.request.post(LAUNCHER+'/api/auth/handoff',{data:{app:'console'}});assert.equal(h.status(),200);
 await p.goto((await h.json()).launchUrl);await p.waitForURL('**/overview');
 const api=async(method,path,body)=>{

  const data=await r.json().catch(()=>null);assert.equal(r.status(),200,`${method} ${path}: ${r.status()} ${data?.error?.code??''}`);return data;
 };
 const me=await api('GET','/me');receipt.identity={accountId:me.account?.id,branchId:me.operatorContext.branchId};save();

  await page.screenshot({path,fullPage:true,mask:[page.locator('input[type="password"]'),page.getByTestId('display-pairing-code')]});

  await page.getByText('Membership Check',{exact:true}).waitFor();

  return {ctx:c,p:page};
 }
 return {browser,ctx,p,api,me,receipt,save,check,capture,pos,live,out,name,stamp,expect,POS,CONSOLE};
}
