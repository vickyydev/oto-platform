// Read-only: the branch, Virtual box 1 and its simulated devices, stations by name, ZZ TEST members.
export default async function (S) {
  const { adminApi, st, save } = S;
  const me = await adminApi('GET', '/me');
  const branches = (await adminApi('GET', '/branches')).body.branches;
  const br = branches.find((b) => b.code === 'hkt-central');
  st.fixture.branchId = br.id; st.fixture.branchName = br.name; st.fixture.branchCode = br.code;
  const boxes = (await adminApi('GET', `/branches/${br.id}/boxes`)).body.boxes;
  const vb = boxes.find((b) => b.role === 'virtual' && b.slot === 'virtual-1' && !b.archived);
  const stations = (await adminApi('GET', `/branches/${br.id}/stations?includeArchived=true`)).body.stations;
  const devs = await adminApi('GET', `/branches/${br.id}/devices`);
  save();
  return {
    me: { status: me.status, name: me.body?.employee?.name ?? me.body?.account?.name },
    branch: { id: br.id, name: br.name },
    boxes: boxes.map((b) => ({ id: b.id, name: b.name, role: b.role, slot: b.slot, status: b.status, archived: b.archived })),
    stations: stations.filter((s) => !s.archived && !s.archivedAt).map((s) => ({ id: s.id, name: s.name, kind: s.kind, box: s.boxId, prefix: s.codePrefix, devices: (s.devices ?? []).map((d) => d.role + ':' + d.label) })),
    usedPrefixes: stations.map((s) => s.codePrefix).filter((p) => /^Z/.test(p ?? '')),
    devices: devs.status === 200 ? (devs.body.devices ?? []).map((d) => ({ id: d.id, label: d.label ?? d.name, kind: d.kind, transport: d.transport, box: d.boxId })) : devs.status,
  };
}
