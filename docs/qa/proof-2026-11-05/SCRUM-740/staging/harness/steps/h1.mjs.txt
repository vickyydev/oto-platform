export default async function (S) {
  const { text, sleep, stepTo } = S.lib;
  const page = S.pages.T740.page;
  const next = page.getByRole('button', { name: 'Start New Sale' });
  if (await next.isVisible().catch(() => false)) await next.click();
  await sleep(1500);
  await page.getByRole('button', { name: /Skip.*walk-in/ }).click();
  await page.getByRole('heading', { name: 'Tourist', exact: true }).click();
  await page.getByRole('heading', { name: '2 Hours Play', exact: true }).click();
  await page.getByRole('heading', { name: /^Configure:/ }).waitFor({ timeout: 15000 });
  await stepTo(page, 'Kids', 1);
  await stepTo(page, 'Adults', 0);
  await page.getByRole('button', { name: 'Continue', exact: true }).click();
  await page.getByText('Children playing alone').waitFor({ timeout: 15000 });
  await page.getByPlaceholder("Child's name").first().fill('ZZ TEST 740 Nok');
  await page.getByRole('button', { name: /Tap to set age/ }).first().click();
  const picker = page.getByRole('dialog');
  await picker.getByRole('button', { name: '5', exact: true }).click(); await sleep(300);
  await picker.getByRole('button', { name: 'Sep', exact: true }).click(); await sleep(300);
  await picker.getByRole('button', { name: '20', exact: true }).click(); await sleep(300);
  const confirm = picker.getByRole('button', { name: 'Confirm', exact: true });
  if (await confirm.isVisible().catch(() => false)) await confirm.click();
  await sleep(800);
  await page.getByRole('button', { name: 'Start camera', exact: true }).first().click();
  const take = page.getByRole('button', { name: /Take photo/ }).first();
  await take.waitFor({ timeout: 15000 });
  await sleep(1200);
  await take.click();
  await sleep(2500);
  await page.getByPlaceholder('Your full name').fill('ZZ TEST 740 Second Family');
  await page.getByRole('button', { name: /I authorize Oto Play Park staff/ }).click();
  for (const c of ['I will remain within 15 minutes of the venue', 'I understand early pickup does not qualify for refund', 'I acknowledge the emergency evacuation point']) { await page.getByRole('button', { name: c }).click(); await sleep(300); }
  await sleep(1200);
  const btn = page.getByRole('button', { name: 'Continue to payment' });
  const enabled = await btn.isEnabled();
  await S.look('T740');
  const t = await text(page, 4000);
  const i = t.indexOf('Children playing alone');
  return { enabled, gate: t.slice(i, i + 500) };
}
