// SCRUM-746 regression: a genuine ZZ TEST voucher typed into the same F&B band field still opens its tab, held.
import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';
const SECRET_FILE = 'C:/Users/waqar/AppData/Local/Temp/claude/9d4c56906b777e8e/317fcfce-2754-44c9-bec5-9a762ab18d03/scratchpad/s747/secrets.json';
export default async function (S) {
  const { st, save, pages, expect, shot, POS, adminApi } = S;
  const f = st.fixture, page = pages.four.page;
  const code = JSON.parse(readFileSync(SECRET_FILE, 'utf8')).voucherCode;
  const v = (f.voucher = f.voucher ?? {});
  v.lastFour = code.slice(-4);
  if (!v.definitionId) {
    const d = await adminApi('POST', '/voucher-definitions', { code: 'zz-test-s747-' + f.stamp, nameEn: 'ZZ TEST SCRUM-746 ' + f.stamp + ' voucher', kind: 'free_item', productId: '01a0cb02-13a9-790f-85a1-73d0ef92ae52', codeMode: 'fixed', fixedCode: code, active: true, singleUse: true, expiryDays: 1, offlinePolicy: 'refuse', costSatang: 0 });
    assert.ok([200, 201].includes(d.status), JSON.stringify(d.body?.error ?? d.status));
    const def = d.body.definition ?? d.body;
    v.definitionId = def.id; v.definitionName = def.nameEn; v.product = 'Bottled Water'; v.createdAt = new Date().toISOString();
    save();
  }
  const step = (st.steps.typedVoucher = { startedAt: new Date().toISOString(), till: f.tills.four.name, lastFour: v.lastFour });
  await page.goto(POS + '/order-station');
  const field = page.getByPlaceholder('Wristband code e.g. 1001');
  await field.fill(code);
  const held = page.waitForResponse((res) => res.request().method() === 'POST' && /\/api\/sales\/[^/]+\/vouchers$/.test(new URL(res.url()).pathname), { timeout: 60000 });
  await page.getByRole('button', { name: 'Load Tab', exact: true }).click();
  const hold = await held;
  const hb = await hold.json().catch(() => null);
  step.holdStatus = hold.status();
  step.saleId = new URL(hold.url()).pathname.split('/')[3];
  step.voucherId = hb?.voucher?.id ?? hb?.id ?? null;
  step.holdError = hb?.error?.code ?? null;
  save();
  assert.equal(hold.status(), 200, 'the genuine voucher is held: ' + JSON.stringify(hb?.error ?? null));
  await expect(page.getByRole('button', { name: 'Remove voucher', exact: true })).toBeVisible({ timeout: 30000 });
  await page.waitForTimeout(1200);
  step.invalidCodeShown = await page.getByText(/Invalid code/).count();
  const chars = [...code].join('[\s-]*');
  const codeText = page.getByText(new RegExp(chars, 'i'));
  step.codeVisibleCount = await codeText.count();
  save();
  await shot('four', 's746-03-genuine-voucher-typed-opens-its-tab-held', { mask: [codeText, field] });
  assert.equal(step.invalidCodeShown, 0);
  // Let it go again: Remove voucher releases the hold on a cart that was never rung up.
  const released = page.waitForResponse((res) => res.request().method() === 'DELETE' && /\/api\/sales\/[^/]+\/vouchers\/[^/]+$/.test(new URL(res.url()).pathname), { timeout: 60000 });
  await page.getByRole('button', { name: 'Remove voucher', exact: true }).click();
  const rel = await released;
  step.releaseStatus = rel.status();
  step.releasedAt = new Date().toISOString();
  save();
  assert.equal(rel.status(), 200, 'release');
  await expect(page.getByRole('button', { name: 'Remove voucher', exact: true })).toHaveCount(0, { timeout: 30000 });
  return step;
}
