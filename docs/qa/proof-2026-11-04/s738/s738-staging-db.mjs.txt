export async function job(program,event,expectedSha){

 const live=(await render('GET',`/services/${api.id}/deploys?limit=1`))[0]?.deploy;assert.equal(live?.status,'live');assert.equal(live.commit.id,expectedSha);

export async function readProof(spec,sha,run=job){
 const event='s738_proof_readback';
 return run(program(spec,event,`
 await c.query('BEGIN READ ONLY');await c.query("SET LOCAL statement_timeout='15s'");const one=async(q,p)=>(await c.query(q,p)).rows;
 const station=(await one('SELECT name FROM core.station WHERE id=$1',[s.stationId]))[0];assert.equal(station.name,s.name+' till');
 const sales=await one('SELECT id,status,gross_satang FROM pos.sale WHERE station_id=$1',[s.stationId]);
 const partyPayments=await one('SELECT p.otoapp_event_id,a.amount_satang FROM pos.party_payment p JOIN pos.payment_attempt a ON a.id=p.payment_attempt_id WHERE p.otoapp_event_id=ANY($1::uuid[])',[[s.events.party.id,s.events.mobile.id]]);
 const attendees=(await one("SELECT count(*)::int n,count(*) FILTER (WHERE parent_phone IS NOT NULL AND parent_phone<>'')::int phones FROM otoapp.event_attendees WHERE event_id::text=$1",[s.events.pass.id]))[0];
 const lease=(await one('SELECT lease_id IS NULL AS released FROM edge.station_session WHERE station_id=$1',[s.stationId]))[0];
 const attempts=await one('SELECT id,status,provider FROM pos.payment_attempt WHERE station_id=$1',[s.stationId]);

export async function readCleanup(spec,sha,run=job){
 const event='s738_cleanup_readback';
 return run(program(spec,event,`
 await c.query('BEGIN READ ONLY');await c.query("SET LOCAL statement_timeout='15s'");const one=async(q,p)=>(await c.query(q,p)).rows;
 const station=(await one('SELECT name,archived_at IS NOT NULL AS archived FROM core.station WHERE id=$1',[s.stationId]))[0];assert.equal(station.name,s.name+' till');assert.equal(station.archived,true);
 const credential=(await one('SELECT label,revoked_at IS NOT NULL AS revoked FROM core.device_credential WHERE id=$1 AND station_id=$2',[s.displayId,s.stationId]))[0];assert.equal(credential.label,s.name+' display');assert.equal(credential.revoked,true);
 const sessions=(await one('SELECT count(*)::int n FROM core.session WHERE station_id=$1 AND expires_at>now() AND revoked_at IS NULL',[s.stationId]))[0].n;assert.equal(sessions,0);
 const lease=(await one('SELECT lease_id IS NULL AS released FROM edge.station_session WHERE station_id=$1',[s.stationId]))[0];assert.equal(lease.released,true);
 const events=(await one('SELECT count(*)::int n,count(*) FILTER (WHERE is_archived)::int archived FROM otoapp.core_events WHERE id::text=ANY($1::text[])',[Object.values(s.events).map(e=>e.id)]))[0];assert.deepEqual(events,{n:3,archived:3});
 const pending=(await one("SELECT count(*)::int n FROM pos.payment_attempt WHERE station_id=$1 AND status NOT IN ('approved','declined','cancelled','not_found')",[s.stationId]))[0].n;assert.equal(pending,0);
 const pendingHolds=(await one("SELECT count(*)::int n FROM pos.event_pass_hold WHERE station_id=$1 AND status='pending'",[s.stationId]))[0].n;assert.equal(pendingHolds,0);
 await c.query('COMMIT');console.log(JSON.stringify({event:'${event}',at:new Date().toISOString(),station,credential,sessions,lease,events,pending,pendingHolds,financialHistoryRetained:true,passed:true}));`),event,sha);
}
