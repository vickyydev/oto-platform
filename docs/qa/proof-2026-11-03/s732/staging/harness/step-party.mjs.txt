import assert from 'node:assert/strict';
import { lastNet } from './lib.mjs?v=4';
export default async function (S) {
  const { st, save, page, expect, shot, tillApi, adminApi, POS } = S;
  const f = st.fixture, party = f.events.party, date = f.eventsJob.date;
  const step = (st.steps.party = { startedAt: new Date().toISOString() });
  const bill = async () => { const r = await tillApi('GET', `/parties/${party.id}?branchId=${f.branchId}`); return { status: r.status, outstandingSatang: r.body?.event?.party?.bill?.outstandingSatang ?? null, payments: (r.body?.event?.party?.bill?.payments ?? r.body?.event?.party?.payments ?? []).map((p) => ({ amountSatang: p.amountSatang ?? null, method: p.method ?? p.tender?.method ?? null })) }; };
  const eod = async () => { const r = await tillApi('GET', `/branches/${f.branchId}/end-of-day?date=${date}`); return { status: r.status, partyPrepaySatang: (r.body?.lines ?? []).find((l) => l.channel === 'party_prepay')?.expectedSatang ?? 0, promptpaySatang: (r.body?.lines ?? []).find((l) => l.channel === 'promptpay')?.expectedSatang ?? 0, cardSatang: (r.body?.lines ?? []).find((l) => l.channel === 'card')?.expectedSatang ?? 0, channels: (r.body?.lines ?? []).map((l) => l.channel) }; };
  step.before = { at: new Date().toISOString(), bill: await bill(), eod: await eod() };
  save();
  await page.goto(POS + '/parties');
  await page.waitForLoadState('networkidle').catch(() => undefined);
  await page.getByText(party.title, { exact: true }).first().click({ timeout: 30000 });
  await page.getByRole('button', { name: /Take balance ฿/ }).click({ timeout: 30000 });
  await expect(page.getByText('How much to collect?')).toBeVisible();
  await page.getByRole('button', { name: /Partial amount/ }).click();
  await page.getByPlaceholder(/Up to ฿/).fill('5000');
  await page.getByRole('button', { name: 'PromptPay', exact: true }).click();
  await page.getByRole('button', { name: /Take ฿5000 by PromptPay/ }).click();
  await expect(page.getByText('PromptPay payment').first()).toBeVisible();
  await shot('staging-09-party-qr-before-start');
  await page.getByRole('button', { name: 'Show payment QR' }).click();
  await expect(page.getByTestId('payment-qr')).toHaveCount(2, { timeout: 30000 });
  step.qrPendingAt = new Date().toISOString();
  await shot('staging-10-party-qr-pending-real-code');
  const tender = lastNet(S, /\/parties\/[^/]+\/payments\/tender$/);
  assert.ok(tender && tender.status === 200, 'party tender call');
  step.qrAttemptId = tender.response?.attempt?.id; step.qrPaymentId = tender.request?.paymentId;
  step.qrAttempt = { status: tender.response?.attempt?.status ?? null, method: tender.response?.attempt?.method ?? null, provider: tender.response?.attempt?.provider ?? null, amountSatang: tender.response?.attempt?.amountSatang ?? null };
  step.qrPending = { at: new Date().toISOString(), bill: await bill(), eod: await eod() };
  save();
  step.qrPaidPressAt = new Date().toISOString();
  const paid = await adminApi('POST', '/webhooks/2c2p/simulator', { attemptId: step.qrAttemptId, event: 'paid' });
  step.qrPaidPress = { status: paid.status, body: paid.body };
  save();
  assert.equal(paid.status, 200, JSON.stringify(paid.body));
  await expect(page.getByRole('heading', { name: 'Payment recorded' })).toBeVisible({ timeout: 60000 });
  await shot('staging-11-party-qr-paid-recorded');
  step.qrPaid = { at: new Date().toISOString(), bill: await bill(), eod: await eod() };
  save();
  await page.getByRole('button', { name: 'Done', exact: true }).click();
  await page.getByRole('button', { name: /Take balance ฿/ }).click({ timeout: 30000 });
  await page.getByRole('button', { name: 'Card', exact: true }).click();
  await page.getByRole('button', { name: /Take ฿4000 by Card/ }).click();
  await shot('staging-12-party-card-before-start');
  await page.getByRole('button', { name: 'Start card payment' }).click();
  step.cardStartedAt = new Date().toISOString();
  await page.waitForTimeout(800);
  const cardTender = lastNet(S, /\/parties\/[^/]+\/payments\/tender$/);
  step.cardAttemptId = cardTender?.response?.attempt?.id; step.cardPaymentId = cardTender?.request?.paymentId;
  step.cardPending = { at: new Date().toISOString(), bill: await bill() };
  save();
  await expect(page.getByRole('heading', { name: 'Payment recorded' })).toBeVisible({ timeout: 120000 });
  await shot('staging-13-party-card-recorded');
  step.cardPaid = { at: new Date().toISOString(), bill: await bill(), eod: await eod() };
  save();
  await page.getByRole('button', { name: 'Done', exact: true }).click();
  await page.waitForTimeout(1500);
  await shot('staging-14-party-bill-balance-zero', { fullPage: true });
  return step;
}
