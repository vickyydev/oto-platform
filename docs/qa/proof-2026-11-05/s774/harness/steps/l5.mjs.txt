export default async function (S) {
  const { LOOK, text } = S.lib;
  const page = S.pages.A.page;
  await page.getByRole('button', { name: 'Add one Kids' }).first().click();
  await page.waitForTimeout(1200);
  await page.getByRole('button', { name: 'Continue', exact: true }).click();
  await page.waitForTimeout(2500);
  await page.screenshot({ path: LOOK });
  return { text: (await text(page, 4000)).replace(/\+?\d{8,}/g, '[phone]').split('Review →')[1] };
}
