// Hide the credit voucher's code on the F&B tab card: whoever holds it can spend the credit.
import { readFileSync, writeFileSync } from 'node:fs';
import { launch } from './common.mjs';
const src = process.argv[2], out = process.argv[3];
const [x, y, w, h] = process.argv.slice(4, 8).map(Number);
const browser = await launch();
try {
  const page = await (await browser.newContext()).newPage();
  const data = 'data:image/png;base64,' + readFileSync(src).toString('base64');
  const b64 = await page.evaluate(async ({ data, x, y, w, h }) => {
    const img = new Image(); img.src = data; await img.decode();
    const c = document.createElement('canvas'); c.width = img.width; c.height = img.height;
    const g = c.getContext('2d'); g.drawImage(img, 0, 0);
    g.fillStyle = '#3b3b3b'; g.fillRect(x, y, w, h);
    g.fillStyle = '#ffffff'; g.font = 'bold 11px sans-serif'; g.textBaseline = 'middle';
    g.fillText('voucher code hidden', x + 6, y + h / 2);
    return c.toDataURL('image/png').split(',')[1];
  }, { data, x, y, w, h });
  writeFileSync(out, Buffer.from(b64, 'base64'));
} finally { await browser.close(); }
