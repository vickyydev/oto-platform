// SCRUM-729 staging proof, F&B in Russian: the ZZ TEST dish twice, per-unit line and tax row. No payment.
import { launch, posSignIn } from 'file:///C:/Users/waqar/OneDrive/Desktop/Projects/oto-pos/scripts/agent/evidence.mjs';
import { readFileSync, writeFileSync } from 'node:fs';

const DIR = 'C:/oto-lanes/pos-small-20261103/docs/qa/proof-2026-11-03/s729';
const FIX = DIR + '/staging-fixture.json';
const rec = JSON.parse(readFileSync(FIX, 'utf8'));
const fx = rec.fixtures;
const F = (rec.fnb = { startedAt: new Date().toISOString() });
const save = () => writeFileSync(FIX, JSON.stringify(rec, null, 2) + '\n');
const browser = await launch();
const { page } = await posSignIn(browser, 'ADMIN', { width: 1700, height: 1000 });
const side = (border) =>
  page.evaluate((border) => [...document.querySelectorAll('div')].filter((d) => d.classList.contains('w-1/2') && d.classList.contains('h-full') && d.classList.contains('border-r') === border).map((e) => e.innerText).join('\n'), border);
const PILL = ['Change language', 'Сменить язык', 'เปลี่ยนภาษา', '更改语言', 'Changer de langue'];
async function setLang(native) {
  await page.locator(PILL.map((l) => `button[aria-label="${l}"]`).join(', ')).last().click();
  await page.getByRole('button', { name: native, exact: true }).last().click();
  await page.waitForTimeout(2500);
}
try {
  const heading = page.getByRole('heading', { name: 'Which station are you on?' });
  if (await heading.isVisible().catch(() => false)) {
    await page.getByRole('button', { name: new RegExp(fx.stationName) }).click();
    await heading.waitFor({ state: 'hidden', timeout: 20000 });
  }
  await page.getByText('Membership Check', { exact: true }).first().waitFor({ timeout: 30000 });
  F.headerStation = await page.getByText(/ZZ TEST SCRUM-729/).first().innerText().catch(() => null);
  await page.getByRole('button', { name: 'F&B' }).first().click();
  await page.waitForTimeout(3000);
  F.url = new URL(page.url()).pathname;
  const guest = page.getByRole('button', { name: /No wristband.*continue as guest/ });
  if (await guest.isVisible().catch(() => false)) await guest.click();
  const search = page.getByPlaceholder('Search dish…');
  await search.waitFor({ timeout: 20000 });
  await search.fill(fx.productName);
  await page.getByText(fx.productName, { exact: true }).first().waitFor({ timeout: 20000 });
  await page.getByText(fx.productName, { exact: true }).first().click();
  await page.waitForTimeout(1200);
  await page.getByText(fx.productName, { exact: true }).first().click();
  await page.waitForTimeout(3000);
  await setLang('Русский');
  const display = await side(false);
  const staff = await side(true);
  F.check = {
    at: new Date().toISOString(),
    perUnitRussian: /за шт\./.test(display),
    taxRowRussian: display.includes('Включая НДС'),
    dishShown: display.includes(fx.productName),
    staffEnglish: !/за шт\.|Включая НДС|Итого/.test(staff),
  };
  F.displayText = display.slice(0, 1200);
  await page.screenshot({ path: `${DIR}/s729-05-fnb-russian-per-unit-and-tax.png` });
  save();
  // Back to English and clear the dish: nothing is written.
  await setLang('English');
  for (let i = 0; i < 3; i += 1) {
    const remove = page.getByRole('button', { name: `Remove ${fx.productName}` });
    if (!(await remove.count())) break;
    await remove.first().click();
    await page.waitForTimeout(1200);
  }
  F.cartCleared = (await page.getByRole('button', { name: `Remove ${fx.productName}` }).count()) === 0;
  F.finishedAt = new Date().toISOString();
  save();
  console.log(JSON.stringify(F, null, 2));
} catch (e) {
  F.error = String(e?.message ?? e).slice(0, 500);
  save();
  await page.screenshot({ path: `${DIR}/s729-error-fnb.png` }).catch(() => {});
  console.log('ERROR', F.error);
} finally {
  await browser.close();
}
