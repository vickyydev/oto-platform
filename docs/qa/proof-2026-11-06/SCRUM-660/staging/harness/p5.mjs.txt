// Step 9: the two AI helpers the tester saw fail, with no AI setting saved anywhere on staging:
// the BEO editor's paste AI Fill (on the ZZ TEST "none" event; reviewed, never loaded) and the Events
// page's AI Event Assistant (a draft for a new birthday; never applied, nothing created).
// Usage: node p5.mjs <paste|assist>
import { adminConsole, persona, hand, APP, loadState, saveState, writeReceipt, chromium, shot, signOut, SCRATCH, OUT } from './lib.mjs';
import { writeFileSync } from 'node:fs';
const which = process.argv[2];
const state = loadState();
const note = (name, detail = {}) => { state.log.push({ step: name, ...detail, at: new Date().toISOString() }); saveState(state); console.log('OK', name, JSON.stringify(detail).slice(0, 1500)); };
state.helpers ??= {};
state.modelCalls ??= [];
const browser = await chromium.launch({ channel: 'msedge' });
const admin = await adminConsole(browser);
let ag;
try {
  if (state.helpers[which]?.done) throw new Error('already done: ' + which);
  ag = await persona(browser, admin, state.agent);
  const a = await hand(ag.ctx, 'oto_app');
  await a.setViewportSize({ width: 1920, height: 1000 });
  const calls = [];
  const path = which === 'paste' ? '/api/ai/parse-beo' : '/api/ai/extract-event';
  a.on('request', (r) => { if (r.method() === 'POST' && new URL(r.url()).pathname === path) calls.push({ startAt: Date.now() }); });
  a.on('response', async (r) => {
    if (r.request().method() === 'POST' && new URL(r.url()).pathname === path) {
      const row = calls[calls.length - 1] ?? {};
      row.status = r.status(); row.endAt = Date.now();
      try { row.body = await r.json(); } catch { row.body = null; }
    }
  });
  const H = (state.helpers[which] = { startedAt: new Date().toISOString() });
  if (which === 'paste') {
    const EV = state.team.none.pick.eventId;
    H.eventId = EV;
    await a.goto(APP + '/studio/events/kanban?beo=' + EV);
    await a.getByTestId('button-ai-fill-beo').waitFor({ timeout: 90000 });
    await a.waitForTimeout(2500);
    await a.getByTestId('button-ai-fill-beo').click();
    const text = 'ZZ TEST pasted note: Saturday 10 April 2027 from 14:00 to 16:00, 12 kids and 5 adults, food ZZ TEST nuggets and fries, cake at 15:30, balloons in blue.';
    H.text = text;
    await a.getByTestId('textarea-ai-fill-beo').fill(text);
    await shot(a, 's660-st-13-paste-ai-fill-text.png');
    await a.getByTestId('button-ai-parse-beo').click();
    const outcome = await Promise.race([
      a.getByTestId('button-ai-fill-confirm').waitFor({ timeout: 240000 }).then(() => 'review'),
      a.locator('[role="dialog"] p.text-destructive').waitFor({ timeout: 240000 }).then(() => 'error'),
    ]);
    await a.waitForTimeout(1200);
    H.outcome = outcome;
    if (outcome === 'error') H.error = (await a.locator('[role="dialog"] p.text-destructive').textContent())?.trim();
    await shot(a, 's660-st-14-paste-ai-fill-answers.png');
    // Leave without loading and without saving.
    await a.keyboard.press('Escape');
    await a.waitForTimeout(800);
    await a.getByTestId('button-cancel').click().catch(() => {});
    await a.waitForTimeout(1000);
  } else {
    await a.goto(APP + '/studio/events/kanban');
    await a.getByTestId('button-create-event').waitFor({ timeout: 90000 });
    await a.waitForTimeout(1500);
    await a.getByTestId('button-create-event').click();
    await a.getByTestId('button-create-birthday').click();
    await a.getByTestId('button-ai-assist').waitFor({ timeout: 30000 });
    await a.getByTestId('button-ai-assist').click();
    const notes = 'ZZ TEST birthday for ZZ TEST Child Assist turning 5 on 24 April 2027 at 15:00, 10 kids and 6 adults, ZZ TEST pizza and juice.';
    H.text = notes;
    await a.getByTestId('textarea-ai-notes').fill(notes);
    await a.getByTestId('button-ai-generate').click();
    const outcome = await Promise.race([
      a.getByTestId('button-ai-apply').waitFor({ timeout: 240000 }).then(() => 'draft'),
      a.getByText('AI Extraction Failed').first().waitFor({ timeout: 240000 }).then(() => 'error'),
    ]);
    await a.waitForTimeout(1200);
    H.outcome = outcome;
    if (outcome === 'error') H.error = (await a.getByText('AI Extraction Failed').first().locator('..').textContent())?.trim();
    await shot(a, 's660-st-15-ai-event-assistant-answers.png');
    // Leave: no field applied, no event created.
    if (outcome === 'draft') await a.getByTestId('button-ai-cancel').click();
    await a.waitForTimeout(600);
    await a.keyboard.press('Escape');
    await a.waitForTimeout(800);
  }
  const c = calls[0] ?? {};
  H.status = c.status ?? null;
  H.seconds = c.endAt && c.startAt ? +((c.endAt - c.startAt) / 1000).toFixed(1) : null;
  H.calls = calls.length;
  if (c.body) writeFileSync(OUT + '/helper-answer-' + which + '.json', JSON.stringify({ route: path, status: c.status, seconds: H.seconds, text: H.text, answer: c.body }, null, 2) + '\n');
  if (c.startAt) state.modelCalls.push({ which, kind: which === 'paste' ? 'paste AI Fill' : 'AI Event Assistant', at: new Date(c.startAt).toISOString(), status: c.status ?? null, seconds: H.seconds });
  H.done = true;
  saveState(state);
  note('Step 9: ' + which, { outcome: H.outcome, status: H.status, seconds: H.seconds, error: H.error ?? null });
} catch (e) {
  console.error('FAIL', e.message.split('\n')[0]);
  if (ag) for (const pg of ag.ctx.pages()) await pg.screenshot({ path: SCRATCH + '/fail-p5-' + which + '-' + ag.ctx.pages().indexOf(pg) + '.png' }).catch(() => {});
  saveState(state);
  process.exitCode = 1;
} finally {
  await signOut(ag);
  await signOut(admin);
  await browser.close();
}
