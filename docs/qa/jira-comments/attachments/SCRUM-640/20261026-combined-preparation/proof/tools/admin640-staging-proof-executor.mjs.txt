export function requestAllowed({ url, method, body, auth = false, health = false, arm = null, assets = [], document = false }) {
  let u; try { u = new URL(url); } catch { return false; }
  if (u.origin !== ORIGIN || u.username || u.password || u.hash || u.href !== url || /[%\\]/.test(u.pathname)) return false;
  if (method === 'POST') return auth && !u.search && (u.pathname === '/api/auth/sign-in'
    ? exact(body, ['phone', 'password']) && ['phone', 'password'].every(k => typeof body[k] === 'string' && body[k].length > 0 && body[k].length < 200)
    : u.pathname === '/api/auth/sign-out' && body === undefined);
  if (method === 'PUT') return !u.search && u.pathname === '/api/me/session/operator' && arm !== null && exact(body, ['operatorId', 'expectedRevision'])
    && uuid(body.operatorId) && Number.isSafeInteger(body.expectedRevision) && body.expectedRevision >= 0 && body.operatorId === arm.operatorId && body.expectedRevision === arm.expectedRevision;
  if (method !== 'GET' || body !== undefined) return false;
  if (['/api/me', '/api/me/permissions', '/api/operators', '/api/branches', '/api/branches/oto-app', '/api/ready'].includes(u.pathname)) return !u.search;
  if (u.pathname === '/api/ops/health') return health && !u.search;
  if (u.pathname === '/api/ops/failures') return health && u.searchParams.size === 2 && u.searchParams.getAll('windowHours').length === 1
    && u.searchParams.get('windowHours') === '24' && u.searchParams.getAll('limit').length === 1 && u.searchParams.get('limit') === '50';
  return !u.search && (document && u.pathname === '/branches' || publicPath(u.pathname) && assets.includes(u.pathname));
}

/** Acknowledgement requires the expected status and validated body, not a later GET. */
export class DispatchLedger {
  constructor(persist, save = () => {}) { this.persist = persist; this.save = save; this.rows = []; this.unresolved = 0; this.pending = 0; this.broken = false; this.failed = null; }
  fail(code) { this.failed ??= code; }
  safe() { need(!this.broken && !this.failed && this.unresolved === 0 && this.pending === 0, 'UNSETTLED_PROOF_REFUSED'); }
  write(row) { try { this.persist(row); this.save(); } catch { this.broken = true; this.fail('RECEIPT_WRITE_FAILED'); throw new ProofError('RECEIPT_WRITE_FAILED'); } }
  async send(description, transport, validate, { cleanup = false } = {}) {
    if (!cleanup) this.safe();
    const row = { ...description, sequence: this.rows.length + 1, intentAt: NOW(), state: 'intent', terminal: false, status: null, cleanup };
    this.rows.push(row);
    try { this.write(row); } catch (e) { if (!cleanup) throw e; }
    // Cleanup always attempts revocation even when the journal has failed.
    this.pending++; row.state = 'dispatching'; let acknowledged = false;
    try {
      const response = await transport(); row.status = response.status;
      if (response.status >= 400 && response.status < 500) { row.state = 'refused'; row.terminal = true; acknowledged = true; }
      const accepted = validate(response);
      need(accepted === true, 'EXPECTED_NATIVE_ACKNOWLEDGEMENT_REQUIRED');
      row.state = 'acknowledged'; row.terminal = true; acknowledged = true; return response;
    } catch (e) {
      if (!acknowledged) { row.state = 'effect_unknown'; row.terminal = false; this.unresolved++; }
      if (!cleanup) this.fail(codeOf(e)); row.errorCode = codeOf(e); throw e;
    } finally {
      row.completedAt = NOW(); this.pending--;
      try { this.write(row); } catch (e) { if (!cleanup) throw e; }
    }
  }
}

export async function closeAuth(c, request, record) {
  const row = { actor: c.actor, signOutStatus: null, meStatus: null, confirmed: false };
  const cookie = c.originalCookie;
  try { need(typeof cookie === 'string' && cookie.startsWith('oto_session='), 'ORIGINAL_OWN_COOKIE_REQUIRED');
    try { const r = await request(c, 'POST', '/api/auth/sign-out', { cleanup: true, originalCookie: cookie }); row.signOutStatus = r.status; row.logoutOk = r.status === 200 && r.body?.ok === true; } catch (e) { row.errorCode = codeOf(e); }
    try { const r = await request(c, 'GET', '/api/me', { cleanup: true, originalCookie: cookie }); row.meStatus = r.status; } catch (e) { row.errorCode ??= codeOf(e); }
    row.confirmed = row.logoutOk === true && row.meStatus === 401 && !row.errorCode;
  } catch (e) { row.errorCode = codeOf(e); }
  c.closed = row.confirmed; c.cookie = ''; if (c.closed) c.originalCookie = '';
  try { record(row); } catch { /* Revocation already attempted; the caller retains its broken-journal latch. */ }
  return row.confirmed;
}
export async function finishOwnedSession(c, { closeContext, request, record, fail }) {
  let contextClosed = true;
  try { await closeContext(c); } catch { contextClosed = false; fail('BROWSER_CONTEXT_CLOSE_FAILED'); }
  const ok = await closeAuth(c, request, record);
  let extrasClosed = true;
  for (const cookie of c.extraCookies) {
    const closed = await closeAuth({ actor: c.actor, originalCookie: cookie }, request, row => record({ ...row, observedCookie: true }));
    if (!closed) extrasClosed = false;
  }
  if (!ok || !extrasClosed) fail('AUTH_CLEANUP_UNCONFIRMED');
  return ok && contextClosed && extrasClosed;
}

export async function performSessionMatrix(d) {
  const admin = await d.login('ADMIN');
  const before = await d.handshake(admin); need(before.platform && before.switchGrant, 'REAL_SWITCH_AUTHORITY_BEFORE_OFFER_REQUIRED');
  await d.open(admin, false); await d.capture(admin, 0);
  await d.select(admin, d.target, before.revision); const working = await d.handshake(admin);
  await d.checkDirectory(admin); await d.capture(admin, 1);
  await d.select(admin, DEMOS.homeOperatorId, working.revision); const restored = await d.handshake(admin);
  validateJourney({ before, working, restored, targetOperatorId: d.target });
  need(await d.finish(admin), 'ADMIN_AUTH_NOT_CLOSED'); d.safe();
  const reception = await d.login('RECEPTION'), rBefore = await d.handshake(reception);
  await d.open(reception, true); const refused = await d.refuse(reception, rBefore.revision), rAfter = await d.handshake(reception);

  await d.capture(reception, 2); need(await d.finish(reception), 'RECEPTION_AUTH_NOT_CLOSED'); d.safe();
  return { before, working, restored, receptionBefore: rBefore, receptionAfter: rAfter };
}

async function boundedJson(response) {
  need(response.body && Number(response.headers.get('content-length') ?? 0) <= MAX_JSON, 'JSON_LIMIT_REQUIRED');
  const reader = response.body.getReader(), chunks = []; let size = 0;
  try { for (;;) { const { done, value } = await reader.read(); if (done) break; size += value.length; need(size <= MAX_JSON, 'JSON_LIMIT_REQUIRED'); chunks.push(Buffer.from(value)); }
    try { return JSON.parse(Buffer.concat(chunks).toString('utf8')); } catch { throw new ProofError('NATIVE_JSON_REQUIRED'); }
  } finally { await reader.cancel().catch(() => {}); reader.releaseLock(); }
}
