import { spawn } from 'node:child_process';
import { createTestDatabase, stopTestServer } from './codex-box-outage-final-20261103/packages/db/src/testing.ts';
const database = await createTestDatabase();
let code = 1;
try {
  const server = new URL(database.url); server.pathname = '/postgres';
  code = await new Promise<number>((resolve, reject) => {
    const child = spawn('pnpm', ['--filter', '@oto/console', 'e2e'], {
      cwd: 'C:/oto-lanes/codex-box-outage-final-20261103', shell: true, windowsHide: true, stdio: 'inherit',
      env: { ...process.env, TEST_DATABASE_URL: server.toString(), CONSOLE_E2E_BUILT: '1', VITE_BOX_BRIDGE_ZONE: 'localhost', VITE_BOX_BRIDGE_ALLOW_LOOPBACK: 'true' },
    });
    child.once('error', reject); child.once('exit', value => resolve(value ?? 1));
  });
} finally { await database.drop(); await stopTestServer(); }
process.exit(code);
