const modelCalls: Array<{ model: string; builtInAdvice: boolean; temperature: unknown; maxTokens: unknown; language: string }> = [];
const modelServer = createHttpServer((req, res) => {
  let raw = '';
  req.on('data', (c) => (raw += c));
  req.on('end', () => {
    const body = JSON.parse(raw || '{}') as { model: string; system?: string; temperature?: unknown; max_tokens?: unknown; messages: Array<{ content: string }> };
    const text = String(body.messages?.[0]?.content ?? '').split('New text to parse:').pop() ?? '';
    const language = /[\u0E00-\u0E7F]/.test(text) ? 'th' : /[\u0400-\u04FF]/.test(text) ? 'ru' : 'en';
    modelCalls.push({
      model: body.model,
      builtInAdvice: String(body.system ?? '').startsWith("You read the final confirmation message of a birthday party"),
      temperature: body.temperature ?? null,
      maxTokens: body.max_tokens ?? null,
      language,
    });
    res.writeHead(200, { 'content-type': 'application/json' });
    res.end(JSON.stringify({
      id: `msg_zz_${modelCalls.length}`, type: 'message', role: 'assistant', model: body.model,
      content: [{ type: 'text', text: JSON.stringify(ANSWER[language as 'en' | 'th' | 'ru']) }],
      stop_reason: 'end_turn', stop_sequence: null, usage: { input_tokens: 1, output_tokens: 1 },
    }));
  });
});
