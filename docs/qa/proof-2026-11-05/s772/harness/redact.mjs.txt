// Hide credit voucher QR codes in a screenshot: whoever holds one can spend the credit.
// node redact.mjs <src> <out> x,y,w,h [x,y,w,h ...]
import { readFileSync, writeFileSync } from 'node:fs';
import { launch } from './common.mjs';
const [src, out, ...rects] = process.argv.slice(2);
const boxes = rects.map((r) => r.split(',').map(Number));
const browser = await launch();
try {
  const page = await (await browser.newContext()).newPage();
  const data = 'data:image/png;base64,' + readFileSync(src).toString('base64');
  const b64 = await page.evaluate(async ({ data, boxes }) => {
    const img = new Image(); img.src = data; await img.decode();
    const c = document.createElement('canvas'); c.width = img.width; c.height = img.height;
    const g = c.getContext('2d'); g.drawImage(img, 0, 0);
    for (const [x, y, w, h] of boxes) {
      g.fillStyle = '#3b3b3b'; g.fillRect(x, y, w, h);
      g.fillStyle = '#ffffff'; g.font = 'bold 10px sans-serif'; g.textBaseline = 'middle'; g.textAlign = 'center';
      g.fillText('code', x + w / 2, y + h / 2 - 6); g.fillText('hidden', x + w / 2, y + h / 2 + 6);
    }
    return c.toDataURL('image/png').split(',')[1];
  }, { data, boxes });
  writeFileSync(out, Buffer.from(b64, 'base64'));
  console.log('redacted', boxes.length);
} finally { await browser.close(); }
