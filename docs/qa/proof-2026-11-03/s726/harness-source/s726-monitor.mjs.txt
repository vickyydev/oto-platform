export function rowDelta(a,b){const prior=new Map(a.tables.map(x=>[x.name,x]));return b.tables.map(x=>{const p=prior.get(x.name)??{inserted:0,updated:0,deleted:0};return{name:x.name,inserted:x.inserted-p.inserted,updated:x.updated-p.updated,deleted:x.deleted-p.deleted};}).filter(x=>x.inserted||x.updated||x.deleted);}
async function metricRead(path,out){
 for(let attempt=0;attempt<3;attempt++){
  let status=null,data,errorName=null,headers={};
  try{const response=await fetch('https://api.render.com/v1'+path,{headers:{authorization:'Bearer '+need('RENDER_API_KEY'),accept:'application/json'},signal:AbortSignal.timeout(15000)});status=response.status;headers=Object.fromEntries([...response.headers].filter(([k])=>/ratelimit|retry-after/i.test(k)));data=await response.json().catch(()=>null);}catch(e){errorName=e.name;}
  (out.metricReads??=[]).push({at:new Date().toISOString(),metric:path.split('?')[0],attempt:attempt+1,status,errorName,headers});
  if(status===200&&Array.isArray(data))return data;
  let delay=1000*(attempt+1);if(status===429){const raw=Number(headers['retry-after']??headers['ratelimit-reset']);if(Number.isFinite(raw)&&raw>0)delay=raw>1e8?Math.max(1000,raw*1000-Date.now()+100):raw*1000+100;else delay=10000;}
  const retryable=status===null||status===429||status>=500;
  assert.ok(retryable&&attempt<2&&delay<=10000,'Render metric GET unavailable within bounded reads: status '+status);
  await new Promise(r=>setTimeout(r,delay));
 }
}
export async function metrics(m,startTime,endTime=new Date().toISOString()){
 if(m.metricCache&&Date.now()-m.metricCacheAt<20000)return {...m.metricCache,reusedAt:new Date().toISOString()};
 const out={startTime,endTime,at:new Date().toISOString(),series:{},summary:{}};

  const data=await metricRead('/metrics/'+metric+'?'+q,out);out.series[metric]=data.map(series=>{const id=series.labels?.find(x=>x.field==='resource')?.value;assert.ok(ids.includes(id),'Metric resource label missing or unexpected');return{...series,resourceId:id};});
  for(const id of ids)assert.ok(out.series[metric].some(series=>series.resourceId===id&&(series.values??[]).some(v=>Number.isFinite(Number(v.value))&&Date.parse(v.timestamp)>=Date.parse(startTime)&&Date.parse(v.timestamp)<=Date.parse(endTime))),'Missing fresh '+metric+' for '+id);
 }
 for(const service of m.services){const s={name:service.name,plan:service.plan};for(const metric of ['cpu','cpu-limit','memory','memory-limit','active-connections']){const v=(out.series[metric]??[]).filter(x=>x.resourceId===service.id).flatMap(x=>x.values??[]).map(x=>Number(x.value)).filter(Number.isFinite);if(v.length)s[metric]={samples:v.length,min:Math.min(...v),max:Math.max(...v),mean:v.reduce((a,x)=>a+x,0)/v.length};}if(s.cpu&&s['cpu-limit'])s.cpuPercentMax=100*s.cpu.max/s['cpu-limit'].min;if(s.memory&&s['memory-limit'])s.memoryPercentMax=100*s.memory.max/s['memory-limit'].min;out.summary[service.id]=s;}
 m.metricCache=out;m.metricCacheAt=Date.now();return out;
}
