import assert from 'node:assert/strict';
import { execFileSync } from 'node:child_process';
import { writeFileSync } from 'node:fs';
import { need } from 'file:///C:/Users/waqar/OneDrive/Desktop/Projects/oto-pos/scripts/agent/env.mjs';
import { render, stagingServices } from 'file:///C:/Users/waqar/OneDrive/Desktop/Projects/oto-pos/scripts/agent/render.mjs';
const SHA = '65c22edf1832e5927536c28ea3db94222ef681e3';
const POS = 'https://oto-pos-web-staging.onrender.com';
const out = process.argv[2];
const r = { at: new Date().toISOString(), source: SHA };
const services = await stagingServices();
r.deployments = [];
for (const name of ['oto-api-staging', 'oto-pos-web-staging', 'oto-pos-staging']) {
  const s = services.find((x) => x.name === name); assert.ok(s, name);
  const d = (await render('GET', `/services/${s.id}/deploys?limit=1`))[0]?.deploy;
  assert.equal(d?.status, 'live', name); assert.equal(d.commit.id, SHA, name);
  r.deployments.push({ name, serviceId: s.id, deployId: d.id, source: d.commit.id, liveAt: d.finishedAt });
}
const ci = JSON.parse(execFileSync('gh', ['run', 'view', '41720268210', '--json', 'databaseId,headSha,status,conclusion,jobs', '-R', 'vickyydev/oto-platform'], { encoding: 'utf8' }));
assert.equal(ci.headSha, SHA); assert.equal(ci.conclusion, 'success');
r.ci = { run: ci.databaseId, headSha: ci.headSha, conclusion: ci.conclusion, jobs: ci.jobs.map((j) => ({ name: j.name, conclusion: j.conclusion })) };
const api = services.find((x) => x.name === 'oto-api-staging');
const one = await render('GET', `/services/${api.id}/env-vars/PGW_PROVIDER`);
const v = one?.value ?? one?.envVar?.value;
r.gatewayIsSimulator = v === 'simulator';
assert.equal(r.gatewayIsSimulator, true, 'PGW_PROVIDER is not the simulator');
const names = [];
let cursor;
for (let i = 0; i < 5; i++) {
  const page = await render('GET', `/services/${api.id}/env-vars?limit=100${cursor ? `&cursor=${cursor}` : ''}`);
  for (const row of page) names.push((row.envVar ?? row).key);
  if (page.length < 100) break; cursor = page[page.length - 1].cursor;
}
r.apiSettingNames = names.length;
const health = await fetch('https://oto-api-staging.onrender.com/health');
r.apiHealth = health.status;
// Admin reads through the POS web address (names and states only).
const signIn = await fetch(POS + '/api/auth/sign-in', { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify({ phone: need('STAGING_ADMIN_PHONE'), password: need('STAGING_ADMIN_PASSWORD') }) });
assert.equal(signIn.status, 200, 'admin sign-in');
const cookie = signIn.headers.getSetCookie().map((c) => c.split(';')[0]).join('; ');
const get = async (p) => { const res = await fetch(POS + '/api' + p, { headers: { cookie } }); const b = await res.json(); assert.equal(res.status, 200, p); return b; };
const me = await get('/me');
r.branchId = me.operatorContext.branchId;
assert.equal(r.branchId, '01a0bbdc-8c42-7753-b7e3-5ae19fb66d33');
const boxes = (await get(`/branches/${r.branchId}/boxes`)).boxes;
const vbox = boxes.find((b) => b.name === 'Virtual box 1');
assert.ok(vbox, 'Virtual box 1');
r.virtualBox = { id: vbox.id, name: vbox.name, role: vbox.role, status: vbox.status ?? null, lastSeenAt: vbox.lastSeenAt ?? null };
const devices = (await get(`/boxes/${vbox.id}/devices`)).devices;
const terms = devices.filter((d) => d.kind === 'terminal' && d.transport === 'simulated' && !d.archivedAt);
r.virtualBoxSimulatedTerminals = terms.map((d) => ({ id: d.id, label: d.label, protocol: d.protocol, status: d.status ?? null }));
r.virtualBoxDeviceTransports = [...new Set(devices.filter((d) => !d.archivedAt).map((d) => d.transport))];
assert.ok(terms.length > 0, 'no simulated terminal on the virtual box');
const stations = (await get(`/branches/${r.branchId}/stations?includeArchived=true`)).stations;
r.existingS732Stations = stations.filter((s) => /SCRUM-732/.test(s.name)).map((s) => ({ id: s.id, name: s.name, archivedAt: s.archivedAt }));
r.usedPrefixes = stations.map((s) => s.codePrefix);
const so = await fetch(POS + '/api/auth/sign-out', { method: 'POST', headers: { cookie, 'idempotency-key': crypto.randomUUID() } });
r.adminSignOut = so.status;
writeFileSync(out, JSON.stringify(r, null, 2) + '\n');
console.log(JSON.stringify({ ...r, usedPrefixes: r.usedPrefixes.length }, null, 1));
