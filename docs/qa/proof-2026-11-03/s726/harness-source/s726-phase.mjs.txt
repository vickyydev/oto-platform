import assert from 'node:assert/strict';import {writeFileSync} from 'node:fs';import {cpus} from 'node:os';import {performance} from 'node:perf_hooks';import {POS,quantile} from './s726-traffic.mjs';import {snapshot,rowDelta,metrics,verifySales} from './s726-monitor.mjs';
const dir='C:/oto-lanes/codex-pos-path-20261102/docs/qa/proof-2026-11-03/s726';export const sleep=ms=>new Promise(r=>setTimeout(r,ms));
export async function quiet(m,t){const s=await snapshot(m);const own=new Set(m.ownSessions?.()??[]),stations=new Set(m.ownStations?.()??[]);s.otherRecentStationSessions=(s.recentStationSessions??[]).filter(x=>!own.has(x.id)&&!stations.has(x.station_id)).length;if(s.otherRecentStationSessions)t.stop('Other staging station activity observed');s.otherRecentUnboundSessions=(s.unboundRecentSessionIds??[]).filter(id=>!own.has(id)).length;if(s.otherRecentUnboundSessions)t.stop('Other staging session activity observed');if(s.nonTestRecentStationSessions||s.nonTestRecentSales)t.stop('Non-test staging activity observed');if(s.connections.total>=m.maxConnections*.6)t.stop('Database connections reached 60%');return s;}
export async function measure(name,seconds,{m,t,warmP95=500,work=null,description=null}){
 assert.ok(!t.stopReason);const before=await quiet(m,t);assert.ok(!t.stopReason);const p={name,description,startedAt:new Date().toISOString(),targetSeconds:seconds,samples:[],metricWindows:[],outcome:'running',leaseRenewalsBefore:t.leaseRenewals,attemptsBefore:t.attempts.length};const start=Date.now();let lastDb=0,lastMetrics=0,lastPrint=0,slowWindows=0,lastLatencyWindow=0,wrongStreamSamples=0;t.phase=name;
 const hostTimes=()=>cpus().reduce((s,c)=>({idle:s.idle+c.times.idle,total:s.total+Object.values(c.times).reduce((a,v)=>a+v,0)}),{idle:0,total:0});let cpuBefore=process.cpuUsage(),hostBefore=hostTimes(),localAt=performance.now(),loopBefore=performance.eventLoopUtilization();const local=()=>{const cpu=process.cpuUsage(),host=hostTimes(),at=performance.now(),loop=performance.eventLoopUtilization();const sample={nodeMemory:process.memoryUsage(),nodeCpuPercentOfOneCore:100*(cpu.user-cpuBefore.user+cpu.system-cpuBefore.system)/1000/(at-localAt),hostCpuPercent:100*(1-(host.idle-hostBefore.idle)/(host.total-hostBefore.total)),eventLoopUtilization:performance.eventLoopUtilization(loop,loopBefore).utilization};cpuBefore=cpu;hostBefore=host;localAt=at;loopBefore=loop;return sample;};
 const save=()=>writeFileSync(dir+'/'+name+'.json',JSON.stringify({...p,traffic:p.traffic??t.summary(name),stop:t.stopReason},null,2)+'\n');const tasks=new Set();const schedule=fn=>{if(t.stopReason)return;const task=Promise.resolve().then(fn).catch(e=>{t.stop('Workload '+e.name+' '+String(e.message).split('\n')[0].replace(/https?:\/\/\S+/g,'[URL]').slice(0,180));}).finally(()=>tasks.delete(task));tasks.add(task);};
 let workRunning=false;const workTimer=work?setInterval(async()=>{if(workRunning||t.stopReason||Date.now()-start>=seconds*1000)return;workRunning=true;try{await work({elapsedMs:Date.now()-start,schedule});}catch(e){t.stop('Work scheduler '+e.name);}finally{workRunning=false;}},200):null;
 try{while(Date.now()-start<seconds*1000&&!t.stopReason){const now=Date.now();if(now-lastDb>=15000){const db=await quiet(m,t);const r=await fetch(POS+'/api/ready',{signal:AbortSignal.timeout(10000)});if(r.status!==200)t.stop('Readiness failed');const ready=await r.json();const channelCount=(ready.stationChannels??ready.checks?.stationChannels)?.connections;if(m.expectedStreams!==undefined){wrongStreamSamples=channelCount===m.expectedStreams?0:wrongStreamSamples+1;if(wrongStreamSamples>=3)t.stop('Stream count did not match the steady client count for three samples');}p.samples.push({at:new Date().toISOString(),dbSampleAt:db.at,connections:db.connections,nonTestSessions:db.nonTestRecentStationSessions,nonTestSales:db.nonTestRecentSales,otherRecentUnboundSessions:db.otherRecentUnboundSessions,otherRecentStationSessions:db.otherRecentStationSessions,stationChannels:ready.stationChannels??ready.checks?.stationChannels,expectedStreams:m.expectedStreams,pool:ready.pool??ready.checks?.pool,nativeStreams:t.nativeStreams,scriptedStreams:t.streams.size,leaseRenewals:t.leaseRenewals,loadGenerator:local()});lastDb=now;}
 if(now-lastMetrics>=60000){if(m.verifySource)await m.verifySource();const window=await metrics(m,new Date(now-90000).toISOString(),new Date().toISOString());p.metricWindows.push(window);for(const s of Object.values(window.summary)){assert.ok(Number.isFinite(s.cpuPercentMax)&&Number.isFinite(s.memoryPercentMax),'Missing resource metrics for '+s.name);if(s.cpuPercentMax>=80||s.memoryPercentMax>=80)t.stop(s.name+' reached the 80% resource stop line');if(s['active-connections']?.max>=m.maxConnections*.6)t.stop('Database connections reached 60%');}lastMetrics=now;}
 if(now-lastLatencyWindow>=60000&&now-start>=60000){const rows=t.rows.filter(x=>x.phase===name&&x.at>=now-60000),errors=rows.filter(x=>x.status<200||x.status>=400).length;if(rows.length&&errors/rows.length>.01)t.stop('Unexpected error rate exceeded 1%');const reads=rows.filter(x=>x.method==='GET'&&!x.stream);const p95=quantile(reads.map(x=>x.ms),.95);slowWindows=p95!==null&&p95>Math.max(1000,2*warmP95)?slowWindows+1:0;if(slowWindows>=2)t.stop('Read p95 exceeded the latency stop line twice');lastLatencyWindow=now;}
 if(now-lastPrint>=30000){const sum=t.summary(name);console.log(JSON.stringify({phase:name,elapsedSeconds:Math.round((now-start)/1000),calls:sum.calls,errors:sum.errors,p95Ms:sum.p95,attempts:t.attempts.length-p.attemptsBefore,streams:t.nativeStreams+t.streams.size,leases:t.leaseRenewals-p.leaseRenewalsBefore}));save();lastPrint=now;}await sleep(1000);}
 clearInterval(workTimer);await Promise.all([...tasks]);p.finishedAt=new Date().toISOString();p.leaseRenewals=t.leaseRenewals-p.leaseRenewalsBefore;t.phase=null;p.seconds=(Date.parse(p.finishedAt)-Date.parse(p.startedAt))/1000;const after=await snapshot(m);p.database={from:before.at,to:after.at,seconds:(Date.parse(after.at)-Date.parse(before.at))/1000,changes:rowDelta(before,after)};p.traffic=t.summary(name,Date.parse(p.finishedAt));p.displayUpdates=t.displaySamples.filter(x=>x.phase===name);p.displayLatency={samples:p.displayUpdates.length,p50:quantile(p.displayUpdates.map(x=>x.ms),.5),p95:quantile(p.displayUpdates.map(x=>x.ms),.95)};p.requestsPerMinute=p.traffic.calls*60/p.seconds;p.rowChanges=p.database.changes.reduce((n,x)=>n+x.inserted+x.updated+x.deleted,0);p.databaseWritesPerMinute=p.rowChanges*60/p.database.seconds;p.sales=await verifySales(m,t.attempts.filter(x=>x.phase===name));const finalWindow=await metrics(m,new Date(Date.parse(p.finishedAt)-90000).toISOString(),p.finishedAt);p.metricWindows.push(finalWindow);for(const resource of Object.values(finalWindow.summary)){assert.ok(Number.isFinite(resource.cpuPercentMax)&&Number.isFinite(resource.memoryPercentMax),'Missing final resource metrics');if(resource.cpuPercentMax>=80||resource.memoryPercentMax>=80)t.stop(resource.name+' reached the 80% resource stop line');if(resource['active-connections']?.max>=m.maxConnections*.6)t.stop('Database connections reached 60%');}if(m.verifySource)await m.verifySource();p.outcome=t.stopReason?'stopped':'passed';save();return p;
 }catch(e){t.stop('Measurement '+e.name+' '+String(e.message).split('\n')[0].replace(/https?:\/\/\S+/g,'[URL]').slice(0,160));p.outcome='stopped';save();throw e;}finally{clearInterval(workTimer);t.phase=null;await Promise.allSettled([...tasks]);writeFileSync(dir+'/traffic-'+name+'.json',JSON.stringify(t.rows.filter(x=>x.phase===name&&(!p.finishedAt||x.at<=Date.parse(p.finishedAt))))+'\n');if(t.attempts.length)writeFileSync(dir+'/sale-attempts.json',JSON.stringify(t.attempts,null,2)+'\n');}
}
