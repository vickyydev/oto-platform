import { writeFileSync } from 'node:fs';
import { render, stagingServices } from 'file:///C:/Users/waqar/OneDrive/Desktop/Projects/oto-pos/scripts/agent/render.mjs';
const services = await stagingServices();
const r = { at: new Date().toISOString(), deployments: [] };
for (const name of ['oto-api-staging', 'oto-pos-web-staging', 'oto-pos-staging']) { const s = services.find((x) => x.name === name); const d = (await render('GET', `/services/${s.id}/deploys?limit=1`))[0]?.deploy; r.deployments.push({ name, status: d.status, source: d.commit.id, deployId: d.id }); }
const api = services.find((x) => x.name === 'oto-api-staging');
const one = await render('GET', `/services/${api.id}/env-vars/PGW_PROVIDER`);
r.gatewayIsSimulator = (one?.value ?? one?.envVar?.value) === 'simulator';
const page = await render('GET', `/services/${api.id}/env-vars?limit=100`);
r.apiSettingNames = page.length;
r.apiHealth = (await fetch('https://oto-api-staging.onrender.com/health')).status;
writeFileSync(process.argv[2], JSON.stringify(r, null, 2) + '\n');
console.log(JSON.stringify(r));
