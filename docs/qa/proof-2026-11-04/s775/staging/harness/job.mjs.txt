import assert from 'node:assert/strict';
import {render,service} from 'file:///C:/Users/waqar/OneDrive/Desktop/Projects/oto-pos/scripts/agent/render.mjs';
export async function job(program,event,expectedSha){
 const api=await service('oto-api-staging');assert.ok(api?.id);
 const live=(await render('GET',`/services/${api.id}/deploys?limit=1`))[0].deploy;assert.equal(live.status,'live');if(expectedSha)assert.equal(live.commit.id,expectedSha);
 const startedAt=new Date().toISOString();
 const startCommand='cd apps/api && node -e "eval(Buffer.from(\''+Buffer.from(program).toString('base64')+'\',\'base64\').toString())"';
 const made=await render('POST',`/services/${api.id}/jobs`,{startCommand});assert.ok(made.id,JSON.stringify(made).slice(0,200));console.error(JSON.stringify({jobId:made.id,event,startedAt}));
 let status;for(let i=0;i<120;i++){const d=await render('GET',`/services/${api.id}/jobs/${made.id}`);if(['succeeded','failed','canceled'].includes(d.status)){status=d.status;break;}await new Promise(r=>setTimeout(r,3000));}
 const qs=new URLSearchParams({ownerId:api.ownerId,resource:made.id,startTime:startedAt,limit:'100',direction:'forward'});
 let rows=[];for(let i=0;i<8;i++){const logs=await render('GET',`/logs?${qs}`);rows=(logs.logs??[]).flatMap(x=>{try{return[JSON.parse(x.message)];}catch{return[];}}).filter(x=>x.event===event||x.event===event+'_error');if(rows.length)break;await new Promise(r=>setTimeout(r,2500));}
 if(status!=='succeeded')console.error(JSON.stringify({event:event+'_failed',status,errors:rows.filter(x=>x.event===event+'_error')}));
 assert.equal(status,'succeeded');const result=rows.find(x=>x.event===event);assert.ok(result,'no result row');
 return {jobId:made.id,source:live.commit.id,startedAt,status,result};
}
/** A read-only query job: body is JS using q(sql,params) that returns an object. */
export const readJob = (event, body, sha) => job(`(async()=>{const {Client}=require('pg'),c=new Client({connectionString:process.env.DATABASE_URL,options:'-c default_transaction_read_only=on -c statement_timeout=20000'});await c.connect();try{await c.query('BEGIN READ ONLY');const q=async(s,p=[])=>(await c.query(s,p)).rows;const out=await (async()=>{${body}})();await c.query('ROLLBACK');console.log(JSON.stringify({event:'${event}',at:new Date().toISOString(),readOnly:true,...out}));}finally{await c.end();}})().catch(e=>{console.error(JSON.stringify({event:'${event}_error',name:e.name,code:e.code??null,message:String(e.message).slice(0,300)}));process.exitCode=1;});`, event, sha);
