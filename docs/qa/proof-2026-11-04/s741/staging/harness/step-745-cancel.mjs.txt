// The ZZ TEST walk-in ticket is cancelled: no payment taken. The window goes back to 1440.
export default async function (S) {
  const { st, save, pages } = S;
  const page = pages.desktop.page;
  await page.setViewportSize({ width: 1440, height: 1024 });
  await page.waitForTimeout(1200);
  const order = page.getByRole('heading', { name: 'Current Order' }).locator('../..');
  await order.getByRole('button', { name: 'Cancel', exact: true }).click();
  await page.waitForTimeout(1500);
  const out = { noItems: await page.getByText('No items added yet', { exact: true }).count(), at: new Date().toISOString(), tookNoPayment: true };
  st.steps.s745 = { ...(st.steps.s745 ?? {}), cancelled: out }; save();
  return out;
}
